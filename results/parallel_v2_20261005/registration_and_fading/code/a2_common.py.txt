def require(value,message):
    if not value:raise RuntimeError(message)

def clean(value):
    if isinstance(value,dict):return {str(k):clean(v) for k,v in value.items()}
    if isinstance(value,(list,tuple)):return [clean(v) for v in value]
    if isinstance(value,np.ndarray):return value.tolist()
    if isinstance(value,np.generic):return value.item()
    if isinstance(value,Path):return str(value)
    return value

def identity(value):return hashlib.sha256(json.dumps(clean(value),sort_keys=True,separators=(',',':'),allow_nan=False).encode()).hexdigest()
def read(path):return json.loads(Path(path).read_text(encoding='utf-8-sig'))

def sha(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as f:
        for block in iter(lambda:f.read(8*1024*1024),b''):h.update(block)
    return h.hexdigest()

def write(path,value):
    path=Path(path);path.parent.mkdir(parents=True,exist_ok=True);temp=path.with_name(path.name+'.tmp')
    temp.write_text(json.dumps(clean(value),indent=2,allow_nan=False)+'\n',encoding='utf-8');os.replace(temp,path)

def seal(path,value):
    if Path(path).exists():require(read(path)==clean(value),'Immutable A2 artifact changed: '+str(path))
    else:write(path,value)

def verify(bindings):
    require(bool(bindings),'Nonempty artifact bindings required')
    for path,digest in bindings.items():require(sha(path)==digest,'Bound A2 input changed: '+str(path))

def load(alias,path):
    path=Path(path).resolve()
    if alias in sys.modules:
        result=sys.modules[alias];require(Path(result.__file__).resolve()==path,'Module alias collision');return result
    spec=importlib.util.spec_from_file_location(alias,path);result=importlib.util.module_from_spec(spec)
    sys.modules[alias]=result;spec.loader.exec_module(result);return result

def registered(path):
    value=read(path)
    require(value.get('status')=='REGISTERED' and value.get('protocol')==CONTRACT,'Exact A2 scope must be frozen before execution')
    verify(value['source_bindings']);verify(value['input_bindings'])
    return value

def array_sha(a):return hashlib.sha256(np.ascontiguousarray(a).tobytes()).hexdigest()

def rgb_sha(a):
    a=np.ascontiguousarray(a,dtype=np.float32)
    require(a.shape==(3,256,256) and np.isfinite(a).all(),'Finite original floatRGB required')
    return hashlib.sha256(b'float32:3,256,256:RGB\0'+a.tobytes()).hexdigest()

def noise(source_id,counter,role,seed=2026100502):
    require(role in ('calibration','development','qualification'),'Registered noise role required')
    key=json.dumps([CONTRACT['noise_namespace'],role,str(source_id),int(counter),int(seed)],separators=(',',':')).encode()
    state=int.from_bytes(hashlib.sha256(key).digest()[:16],'little')
    return np.random.Generator(np.random.PCG64(state)).standard_normal((1024,2))

def mask(n,counter,role):
    require(role in ('calibration','development','qualification'),'Registered mask role required')
    prefix=json.dumps([CONTRACT['mask_namespace'],role,int(counter),0],separators=(',',':')).encode()+b'|'
    blocks=[hashlib.sha256(prefix+str(i).encode('ascii')).digest() for i in range((int(n)+255)//256)]
    return np.unpackbits(np.frombuffer(b''.join(blocks),dtype=np.uint8),bitorder='big')[:n]

def transition(points):
    wanted={(c,s) for c in CONTRACT['codecs'] for s in CONTRACT['calibration_candidate_snrs']}
    index={(p['codec'],p['snr_db']):p for p in points}
    require(len(points)==len(index) and set(index)==wanted,'All ten physical calibration points are required')
    rows=[];distances={}
    for snr in CONTRACT['transition_tie_order']:
        p=[index[c,snr] for c in CONTRACT['codecs']]
        require(all(x['n_blocks']>0 and 0<=x['n_reject']<=x['n_blocks'] for x in p),'Invalid BLER counts')
        mean=sum(Fraction(x['n_reject'],x['n_blocks']) for x in p)/2
        distances[snr]=abs(mean-Fraction(1,10))
        rows.append(dict(snr_db=snr,mean_crc_reject_bler=float(mean),distance=float(distances[snr])))
    winner=min(rows,key=lambda r:(distances[r['snr_db']],CONTRACT['transition_tie_order'].index(r['snr_db'])))
    return dict(status='FROZEN_BEFORE_DEVELOPMENT',snrs_db=[13,winner['snr_db']],transition=winner,candidates=rows,
        no_per_arm_optimization=True,selection_uses_image_metrics=False,
        diagnostic='Failure-rule evidence may be insufficient; do not expand SNRs based on development results.')

def wilson(success,n):
    require(0<=success<=n and n>0,'Invalid binomial counts');z=1.959963984540054;p=success/n;den=1+z*z/n
    center=(p+z*z/(2*n))/den;half=z*((p*(1-p)/n+z*z/(4*n*n))**.5)/den
    return [0. if success==0 else max(0.,center-half),1. if success==n else min(1.,center+half)]

def load_calibration(path):
    with np.load(path,allow_pickle=False) as data:
        tokens=data['tokens'];ids=data['source_ids'];pre=data['preprocessing_ids']
    require(tokens.shape==(1000,680) and np.issubdtype(tokens.dtype,np.integer)
        and np.all((0<=tokens)&(tokens<4096)) and ids.shape==pre.shape==(1000,)
        and len(set(ids.tolist()))==1000,'Exact registered calibration payload population required')
    return tokens,ids,pre
