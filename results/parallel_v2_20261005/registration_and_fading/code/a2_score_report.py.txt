def write_csv(path, rows):
    path = Path(path); path.parent.mkdir(parents=True, exist_ok=True)
    opener = gzip.open if str(path).endswith(".gz") else open
    with opener(path, "wt", encoding="utf-8", newline="") as stream:
        keys = list(dict.fromkeys(k for row in rows for k in row))
        writer = csv.DictWriter(stream, keys, lineterminator="\n"); writer.writeheader()
        writer.writerows({k:u.clean(v) for k,v in row.items()} for row in rows)


def sealed_checkpoint(path, binding, index):
    cp = u.read(path)
    u.require(cp["binding"] == binding and cp["source_index"] == index and
              cp["payload_sha256"] == u.identity({k:v for k,v in cp.items() if k != "payload_sha256"}),
              "A2 source checkpoint identity differs")
    return cp


def row_key(row):
    return int(row["source_index"]), int(row["snr_db"]), int(row["noise_seed"]), row["method"]


def pixels(array):
    array = np.asarray(array)
    u.require(array.dtype == np.float32 and array.shape == (3,256,256) and np.isfinite(array).all()
              and array.min() >= 0 and array.max() <= 1, "Unchanged float32 CHW256 image required")
    return np.ascontiguousarray(array)


def source_images(cp, index, snrs):
    rows = cp["rows"]
    expected = {(index,s,n,m) for s in snrs for n in SEEDS for m in METHODS}
    u.require(len(rows) == len(expected) and {row_key(r) for r in rows} == expected, "A2 source is missing a four-arm frame")
    proof = cp["rgb"]; u.require(u.sha(proof["path"]) == proof["sha256"], "A2 float archive changed")
    with np.load(proof["path"], allow_pickle=False) as data:
        u.require(set(data.files) == {"images","source_rgb","row_ids","image_slots"}, "A2 float archive schema differs")
        images = data["images"].copy(); target = pixels(data["source_rgb"]).copy()
        slots = data["image_slots"].tolist(); ids = data["row_ids"].tolist()
    u.require(images.dtype == np.float32 and images.ndim == 4 and images.shape[1:] == (3,256,256), "Invalid A2 image array")
    u.require(len(slots) == len(rows) and ids == [r["replay_row_id"] for r in rows] and len(set(ids)) == len(rows), "A2 float row mapping differs")
    hashes = [u.rgb_sha(pixels(image)) for image in images]; reference = u.rgb_sha(target)
    identity = {(r["source_id"],r["preprocessing_id"],r["class_index"]) for r in rows}
    u.require(len(identity) == 1, "Source identity differs within one A2 checkpoint")
    for row, slot in zip(rows,slots):
        u.require(type(slot) is int and 0 <= slot < len(images) and row["image_sha256"] == hashes[slot]
                  and row["reference_sha256"] == reference, "A2 reference/reconstruction hash differs")
        u.require(row["N"] == 1024 and row["phy_family"] == "16QAM" and row["method"] == row["codec"]+"-"+row["rule"], "A2 registered arm differs")
    validate_pairing(rows, snrs, [index])
    return rows, images, target, slots


def validate_pairing(rows, snrs, indices=range(100)):
    indexed = {row_key(r):r for r in rows}
    expected = {(i,s,n,m) for i in indices for s in snrs for n in SEEDS for m in METHODS}
    u.require(len(indexed) == len(rows) and set(indexed) == expected, "A2 exact source/noise/SNR/arm grid missing or duplicated")
    for i in indices:
        for s in snrs:
            for n in SEEDS:
                group = [indexed[i,s,n,m] for m in METHODS]
                u.require(len({(r["source_id"],r["preprocessing_id"],r["reference_sha256"]) for r in group}) == 1,
                          "A2 paired arms use different targets")
                for codec in ("C","L"):
                    keep, drop = indexed[i,s,n,codec+"-KEEP"], indexed[i,s,n,codec+"-DROP"]
                    for field in ("shared_decoded_event_sha256","shared_observation_sha256","header_ok","body_crc_accept","E"):
                        u.require(keep[field] == drop[field], "KEEP/DROP do not share a single physical decode: "+field)
                    if not keep["header_ok"] or keep["body_crc_accept"]:
                        u.require(keep["image_sha256"] == drop["image_sha256"], "Equivalent accepted/erased receiver states produced different images")
                    if not keep["header_ok"]:
                        u.require(keep["body_crc_accept"] is None and keep["gray"] and drop["gray"], "Header rejection rule changed")
                    if keep["header_ok"] and keep["body_crc_accept"] is False:
                        u.require(not keep["gray"] and drop["gray"], "Registered body failure KEEP/DROP rule changed")
    return indexed


def interval(values, draws):
    values = np.asarray(values, dtype=np.float64)
    u.require(values.shape == (100,) and np.isfinite(values).all(), "100 finite paired source values required")
    low, high = np.quantile(values[draws].mean(1), [.025,.975])
    return dict(mean=float(values.mean()), ci_low=float(low), ci_high=float(high), sources=100, frames=300,
                bootstrap_unit="source_mean_over_three_registered_noise_seeds", bootstrap_replicates=10000,
                bootstrap_seed=20261002, interval_role="descriptive_reused_development_not_holdout")


def failure_subset(indexed, snr, codec, metric, draws):
    # Resample all sources as clusters; varying failure counts stay with source.
    numerator, denominator = np.zeros(100), np.zeros(100)
    for i in range(100):
        for seed in SEEDS:
            keep, drop = indexed[i,snr,seed,codec+"-KEEP"], indexed[i,snr,seed,codec+"-DROP"]
            if keep["header_ok"] and keep["body_crc_accept"] is False:
                numerator[i] += float(drop[metric])-float(keep[metric]); denominator[i] += 1
    frames = int(denominator.sum()); sources = int(np.count_nonzero(denominator))
    result = dict(snr_db=snr, codec=codec, metric=metric, contrast="DROP_minus_KEEP_given_body_CRC_rejected_and_header_accepted",
                  eligible_frames=frames, eligible_sources=sources, denominator_population_frames=300,
                  selected_using_reconstruction_quality=False, post_event_conditional_diagnostic=True,
                  general_population_effect=False, bootstrap_unit="source_cluster_ratio_of_paired_delta_sum_to_failure_count",
                  bootstrap_replicates=10000, bootstrap_seed=20261002)
    if not frames:
        return dict(result, status="NOT_ESTIMABLE_NO_BODY_REJECTIONS", mean=None, ci_low=None, ci_high=None)
    den = denominator[draws].sum(1); num = numerator[draws].sum(1)
    result.update(mean=float(numerator.sum()/frames), zero_denominator_replicates=int(np.count_nonzero(den==0)))
    if np.any(den==0):
        return dict(result, status="UNCERTAIN_SPARSE_FAILURES", ci_low=None, ci_high=None)
    low, high = np.quantile(num/den,[.025,.975])
    return dict(result,status="DESCRIPTIVE_CONDITIONAL_INTERVAL",ci_low=float(low),ci_high=float(high))


def analyze(rows, snrs):
    u.require(len(snrs)==2 and len(set(snrs))==2 and 13 in snrs, "A2 requires fixed13dB and one frozen transition point")
    indexed = validate_pairing(rows,snrs); draws=np.random.default_rng(20261002).integers(0,100,(10000,100))
    summary, effects, failures, means = [], [], [], {}
    for snr in snrs:
        for method in METHODS:
            group=[indexed[i,snr,n,method] for i in range(100) for n in SEEDS]
            for metric in METRICS:
                u.require(all(r.get(metric) is not None and math.isfinite(float(r[metric])) for r in group), "Missing mandatory A2 metric: "+metric)
                values=np.asarray([float(r[metric]) for r in group]).reshape(100,3).mean(1); means[snr,method,metric]=values
                summary.append(dict(N=1024,snr_db=snr,method=method,metric=metric,**interval(values,draws)))
            for name,values in (("header_accept_fraction",[float(r["header_ok"]) for r in group]),
                                ("body_CRC_reject_with_header_fraction",[float(r["header_ok"] and r["body_crc_accept"] is False) for r in group]),
                                ("gray_fraction",[float(r["gray"]) for r in group]),
                                ("undetected_fraction",[float(r["undetected"]) for r in group])):
                summary.append(dict(N=1024,snr_db=snr,method=method,metric=name,**interval(np.asarray(values).reshape(100,3).mean(1),draws)))
        for metric in METRICS:
            for name,weights in CONTRASTS.items():
                values=sum(weight*means[snr,method,metric] for method,weight in weights.items())
                effects.append(dict(N=1024,snr_db=snr,contrast=name,metric=metric,weights=json_weights(weights),
                                    direction="smaller_is_better" if metric in LOWER else "resource_no_quality_direction" if metric=="E" else "larger_is_better",
                                    **interval(values,draws)))
            for codec in ("C","L"):
                failures.append(failure_subset(indexed,snr,codec,metric,draws))
    return summary,effects,failures


def json_weights(value):
    import json
    return json.dumps(value,sort_keys=True,separators=(",",":"))

        rows,images,target,slots=source_images(cp,index,inputs.snrs); path=metrics/"source_checkpoints"/f"{index:04d}.json"

            reference=torch.from_numpy(target[None]); prepared=evaluator.prepare_reference(reference); source_prediction=int(prepared["resnet50"][0])
            original_convnext=classifier.predict(target); truth=int(rows[0]["class_index"])

                u.require(all(row.get(key) is not None and math.isfinite(float(row[key])) for key in METRICS), "Incomplete unified A2 metrics")
            saved=dict(binding=binding,source_index=index,rows=scored,unique_images=len(images),input_rgb=cp["rgb"])
            saved["payload_sha256"]=u.identity(saved); u.seal(path,saved)
        u.require([row_key(r) for r in saved["rows"]]==[row_key(r) for r in rows], "Scored A2 population differs")
        allrows.extend(saved["rows"]); outputs[str(path)]=u.sha(path)
        u.write(out/"score_status.json",dict(status="RUNNING",sources=index+1,method_rows=len(allrows),elapsed_seconds=time.monotonic()-started))
    summary,effects,failures=analyze(allrows,inputs.snrs)
    for name,rows in (("metrics_per_frame.csv.gz",allrows),("metrics_summary.csv",summary),("paired_effects.csv",effects),("failure_subset_diagnostics.csv",failures)):

def mdtable(headers,rows):
    def cell(value):return str(value).replace("|","\\|").replace("\n"," ")
    return "\n".join(["| "+" | ".join(map(cell,headers))+" |","| "+" | ".join("---" for _ in headers)+" |"]+
                     ["| "+" | ".join(map(cell,row))+" |" for row in rows])
