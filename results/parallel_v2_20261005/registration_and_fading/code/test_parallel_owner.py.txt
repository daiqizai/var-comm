class QueueTests(unittest.TestCase):
    def setUp(self):
        self.temp=tempfile.TemporaryDirectory();self.addCleanup(self.temp.cleanup)
        self.root=Path(self.temp.name).resolve();self.a1=self.root/'A1';self.a1.mkdir()
        self.expected=dict(pid=3823500,start_ticks='181766976',uid=1002)
        self.children=[dict(pid=4000+i,start_ticks=str(100+i),uid=1002,source_indices=list(range(i,1000,4))) for i in range(4)]
        self.binding='registered-science'
        report=self.root/'reports/result.md';report.parent.mkdir();report.write_text('actual result')
        owner.write(self.a1/'score_completion.json',dict(status='COMPLETE'))
        owner.write(self.a1/'report_completion.json',dict(status='COMPLETE'))
        owner.write(self.a1/'publication_inventory.json',dict(files={'reports/result.md':owner.sha(report)}))
        cohort=dict(status='READY',stage='production',schema='M1_GPU_COHORT_V1',owner=self.expected,workers=self.children,registration_sha256=self.binding)
        owner.write(self.a1/'production/cohort.json',cohort)
        owner.write(self.a1/'production/completion.json',dict(status='COHORT_PRODUCTION_COMPLETE',workers=4,registration_sha256=self.binding,
            cohort_manifest_sha256=owner.sha(self.a1/'production/cohort.json'),source_partition='exact nonoverlapping modulo workers over 0..999'))
        self.pub=dict(status='PUSHED_AND_STOPPED',registration_sha256=self.binding,commit='a'*40,
            score_completion_sha256=owner.sha(self.a1/'score_completion.json'),report_completion_sha256=owner.sha(self.a1/'report_completion.json'),
            publication_inventory_sha256=owner.sha(self.a1/'publication_inventory.json'))
        self.seal_pub()
        self.proc=patch.object(owner,'identity',side_effect=FileNotFoundError);self.proc.start();self.addCleanup(self.proc.stop)
        self.idle=patch.object(owner,'gpu_idle',return_value=True);self.idle.start();self.addCleanup(self.idle.stop)
        self.remote=patch.object(owner.subprocess,'check_output',return_value='a'*40+'\trefs/heads/main\n');self.remote.start();self.addCleanup(self.remote.stop)

    def seal_pub(self):
        owner.write(self.a1/'publication_completion.json',self.pub)
        owner.write(self.a1/'recovery_r1/completion.json',dict(status='RECOVERY_DELIVERED_AND_STOPPED',publication_completion_sha256=owner.sha(self.a1/'publication_completion.json')))

    def gate(self):return owner.validate_a1_delivery(self.root,self.a1,self.expected,self.binding)

    def test_verified_complete_and_absent_processes_release(self):
        result=self.gate();self.assertEqual(result['status'],'VERIFIED_A1_PUBLISHED_AND_EXITED');self.assertEqual(result['child_identities'],self.children)

    def test_identical_running_start_string_integer_is_not_exited(self):
        with patch.object(owner,'identity',return_value=dict(self.expected,start_ticks=int(self.expected['start_ticks']),state='S')):
            self.assertFalse(owner.exited(self.expected));self.assertIsNone(self.gate())

    def test_same_zombie_identity_only_after_gpu_release(self):
        with patch.object(owner,'identity',side_effect=lambda pid:dict(next(v for v in [self.expected]+self.children if v['pid']==pid),state='Z')):
            self.assertIsNotNone(self.gate())
            with patch.object(owner,'gpu_idle',return_value=False):self.assertIsNone(self.gate())

    def test_reused_pid_is_not_mistaken_for_original_process(self):
        with patch.object(owner,'identity',return_value=dict(self.expected,start_ticks='new-incarnation',state='S')):
            self.assertTrue(owner.exited(self.expected))

    def test_same_start_wrong_uid_is_an_error(self):
        with patch.object(owner,'identity',return_value=dict(self.expected,state='S',uid=2000)):
            with self.assertRaisesRegex(RuntimeError,'UID'):owner.exited(self.expected)

    def test_live_child_keeps_gate_closed(self):
        def identity(pid):
            if pid==self.children[2]['pid']:return dict(self.children[2],state='S')
            raise FileNotFoundError()
        with patch.object(owner,'identity',side_effect=identity):self.assertIsNone(self.gate())

    def test_missing_or_duplicate_child_manifest_is_rejected(self):
        path=self.a1/'production/cohort.json';original=owner.read(path)
        for workers in [self.children[:3],self.children[:3]+[self.children[0]]]:
            owner.write(path,dict(original,workers=workers))
            with self.assertRaises(RuntimeError):self.gate()

    def test_unsealed_cohort_manifest_is_rejected(self):
        path=self.a1/'production/cohort.json';owner.write(path,dict(owner.read(path),status='STARTING'))
        with self.assertRaisesRegex(RuntimeError,'cohort required'):self.gate()

    def test_missing_completion_waits(self):
        (self.a1/'recovery_r1/completion.json').unlink();self.assertIsNone(self.gate())

    def test_failed_predecessor_never_releases_or_signals(self):
        owner.write(self.a1/'owner_failure.json',dict(error='preserve'))
        with patch.object(owner.os,'kill') as kill,patch.object(owner.subprocess,'Popen') as launch:
            with self.assertRaisesRegex(RuntimeError,'new failure'):self.gate()
            kill.assert_not_called();launch.assert_not_called()

    def test_wrong_publication_registration_refuses(self):
        self.pub['registration_sha256']='other';self.seal_pub()
        with self.assertRaisesRegex(RuntimeError,'publication identity'):self.gate()

    def test_tampered_published_file_refuses(self):
        (self.root/'reports/result.md').write_text('tampered')
        with self.assertRaisesRegex(RuntimeError,'dependency changed'):self.gate()

    def test_changed_recovery_publication_hash_refuses(self):
        owner.write(self.a1/'recovery_r1/completion.json',dict(status='RECOVERY_DELIVERED_AND_STOPPED',publication_completion_sha256='wrong'))
        with self.assertRaisesRegex(RuntimeError,'recovery delivery'):self.gate()

    def test_advanced_remote_accepts_ancestor(self):
        with patch.object(owner.subprocess,'check_output',return_value='b'*40+'\trefs/heads/main\n'),patch.object(owner.subprocess,'run',return_value=SimpleNamespace(returncode=0)) as run:
            self.assertEqual(self.gate()['remote_at_release'],'b'*40)
            self.assertIn(['git','merge-base','--is-ancestor','a'*40,'b'*40],[c.args[0] for c in run.call_args_list])

    def test_divergent_remote_rejects(self):
        def run(command,**kwargs):
            if 'merge-base' in command:raise subprocess.CalledProcessError(1,command)
            return SimpleNamespace(returncode=0)
        with patch.object(owner.subprocess,'check_output',return_value='b'*40+'\trefs/heads/main\n'),patch.object(owner.subprocess,'run',side_effect=run):
            with self.assertRaises(subprocess.CalledProcessError):self.gate()

    def test_queue_failure_records_evidence_without_starting_a_stage(self):
        out=self.root/'PARALLEL-V2-20261005';out.mkdir();dependency=self.root/'code.py';dependency.write_text('frozen')
        cfg=dict(status='REGISTERED_A2_AFTER_A1',root=str(self.root),out=str(out),a1_out=str(self.a1),source_bindings={str(dependency):owner.sha(dependency)},
            input_bindings={},a1_owner_identity=self.expected,a1_registration_sha256=self.binding,stages=[dict(name='must_not_start')])
        config=self.root/'config.json';owner.write(config,cfg)
        with patch.object(sys,'argv',['parallel_owner','--config',str(config)]),patch.object(owner,'identity',return_value=dict(pid=77,start_ticks=55,uid=1002)),\
             patch.object(owner,'validate_a1_delivery',side_effect=RuntimeError('predecessor failed')),patch.object(owner.signal,'signal'),\
             patch.object(owner.subprocess,'Popen') as launch,patch.object(owner.os,'kill') as kill:
            with self.assertRaisesRegex(RuntimeError,'predecessor failed'):owner.main()
            launch.assert_not_called();kill.assert_not_called()
        failure=owner.read(out/'owner_failure.json');self.assertTrue(failure['A1_untouched']);self.assertFalse(failure['automatic_retry'])
        self.assertFalse((out/'A2/a1_release_gate.json').exists())

    def run_cleanup_case(self,bound):
        out=self.root/'PARALLEL-V2-20261005';out.mkdir();dependency=self.root/'code.py';dependency.write_text('frozen')
        stage=dict(name='own_child',gpu=False,command=['new-a2-only'],wall_seconds_max=-1,receipt=str(out/'missing.json'),expected_status=['COMPLETE'])
        cfg=dict(status='REGISTERED_A2_AFTER_A1',root=str(self.root),out=str(out),a1_out=str(self.a1),source_bindings={str(dependency):owner.sha(dependency)},
            input_bindings={},a1_owner_identity=self.expected,a1_registration_sha256=self.binding,stages=[stage])
        config=self.root/'config.json';owner.write(config,cfg)
        from unittest.mock import Mock
        child=Mock(pid=999,returncode=None);child.poll.side_effect=[None,None,0] if bound else [None]
        ident=Mock(side_effect=[dict(pid=77,start_ticks=55,uid=1002),dict(pid=999,start_ticks=66,uid=1002),FileNotFoundError()] if bound
                   else [dict(pid=77,start_ticks=55,uid=1002),FileNotFoundError()])
        with patch.object(sys,'argv',['parallel_owner','--config',str(config)]),patch.object(owner,'identity',ident),\
             patch.object(owner,'validate_a1_delivery',return_value={'status':'VERIFIED_A1_PUBLISHED_AND_EXITED'}),patch.object(owner.signal,'signal'),\
             patch.object(owner.subprocess,'Popen',return_value=child),patch.object(owner.os,'kill') as kill:
            with self.assertRaises(RuntimeError if bound else FileNotFoundError):owner.main()
            child.terminate.assert_not_called();kill.assert_not_called()
        return owner.read(out/'owner_failure.json')

    def test_fast_exit_before_identity_keeps_failure_evidence(self):
        failure=self.run_cleanup_case(False)
        self.assertIn('FileNotFoundError',failure['error']);self.assertIn('not captured',failure['cleanup_error'])
        self.assertEqual(failure['child_pid'],999);self.assertTrue(failure['A1_untouched'])

    def test_exit_during_cleanup_does_not_mask_original_failure(self):
        failure=self.run_cleanup_case(True)
        self.assertIn('wall limit',failure['error']);self.assertIsNone(failure['cleanup_error']);self.assertTrue(failure['A1_untouched'])


class TokenExportTests(unittest.TestCase):
    def setUp(self):
        self.temp=tempfile.TemporaryDirectory();self.addCleanup(self.temp.cleanup);self.base=Path(self.temp.name).resolve()
        self.ids=['image-%03d'%i for i in range(100)];self.pre=['prep-%03d'%i for i in range(100)]
        self.visual={'vae':'frozen-vae','var':'frozen-var'}
        self.tx=self.base/'tx_tokens';self.tx.mkdir();self.manifest=self.tx/'manifest.json'
        self.txreg=dict(schema='uep_tx_tokens_v1',population='development',source_count=100,source_ids=self.ids,
            source_identity=[dict(source_index=i,source_id=s,preprocessing_id=p) for i,(s,p) in enumerate(zip(self.ids,self.pre))],
            visual_identity=self.visual,training_updates=0,quality_selection_updates=0,labels_given_to_receiver=False,source_images_exported=False)
        u.write(self.tx/'registration.json',self.txreg);sources=[];outputs={}
        for i in range(100):
            archive=self.tx/'tokens'/('%04d.npz'%i);archive.parent.mkdir(exist_ok=True)
            np.savez(archive,tokens=np.full(680,i,dtype=np.int64))
            cp=dict(self.txreg['source_identity'][i],binding=u.identity(self.txreg),archive=str(archive),archive_sha256=u.sha(archive),source_seconds=.1)
            cp['payload_sha256']=u.identity(cp);cpath=self.tx/'source_checkpoints'/('%04d.json'%i);u.write(cpath,cp)
            sources.append({k:cp[k] for k in ('source_index','source_id','preprocessing_id','archive','archive_sha256')})
            outputs[str(archive)]=u.sha(archive);outputs[str(cpath)]=u.sha(cpath)
        self.m=dict(self.txreg,status='UEP_TX_TOKENS_COMPLETE',registration_sha256=u.sha(self.tx/'registration.json'),sources=sources,outputs=outputs)
        u.write(self.manifest,self.m)

    def arrays(self):return exporter.development_arrays(self.manifest,self.ids,self.pre,self.visual)

    def test_verified_token_export_preserves_exact_order(self):
        t,ids,bindings=self.arrays();self.assertEqual(t.shape,(100,680));self.assertEqual(ids,self.ids)
        np.testing.assert_array_equal(t[:,0],np.arange(100));self.assertEqual(len(bindings),202)

    def test_different_preprocessing_is_rejected_before_npz_read(self):
        wrong=self.pre.copy();wrong[1]='other'
        with patch.object(exporter.np,'load') as load:
            with self.assertRaisesRegex(RuntimeError,'preprocessing'):exporter.development_arrays(self.manifest,self.ids,wrong,self.visual)
            load.assert_not_called()

    def test_per_archive_source_reorder_is_rejected(self):
        self.m['sources'][0],self.m['sources'][1]=self.m['sources'][1],self.m['sources'][0];u.write(self.manifest,self.m)
        with self.assertRaisesRegex(RuntimeError,'source order'):self.arrays()

    def test_per_archive_preprocessing_mismatch_is_rejected(self):
        self.m['sources'][0]['preprocessing_id']='wrong';u.write(self.manifest,self.m)
        with self.assertRaisesRegex(RuntimeError,'preprocessing'):self.arrays()

    def test_changed_tx_visual_identity_rejected(self):
        with self.assertRaisesRegex(RuntimeError,'visual identity'):exporter.development_arrays(self.manifest,self.ids,self.pre,{'vae':'other'})

    def test_tampered_checkpoint_rejected(self):
        p=self.tx/'source_checkpoints/0000.json';cp=u.read(p);cp['source_id']='other';u.write(p,cp)
        with self.assertRaisesRegex(RuntimeError,'checkpoint changed'):self.arrays()

    def test_checkpoint_must_have_valid_self_digest_even_if_outer_hash_updates(self):
        p=self.tx/'source_checkpoints/0000.json';cp=u.read(p);cp['binding']='other';u.write(p,cp)
        self.m['outputs'][str(p)]=u.sha(p);u.write(self.manifest,self.m)
        with self.assertRaisesRegex(RuntimeError,'provenance'):self.arrays()

    def test_changed_archive_rejected(self):
        p=self.tx/'tokens/0000.npz';np.savez(p,tokens=np.ones(680,dtype=np.int64))
        with self.assertRaisesRegex(RuntimeError,'archive or source checkpoint changed'):self.arrays()


class FrozenSelectionTests(unittest.TestCase):
    def setUp(self):
        self.temp=tempfile.TemporaryDirectory();self.addCleanup(self.temp.cleanup);self.root=Path(self.temp.name).resolve();self.out=self.root/'A2';self.out.mkdir()
        dep=self.root/'code.py';dep.write_text('frozen');self.cal=self.out/'cal.npz';np.savez(self.cal,tokens=np.zeros((1000,680),dtype=np.int64))
        self.gate=self.out/'a1_release_gate.json';u.write(self.gate,dict(status='VERIFIED_A1_PUBLISHED_AND_EXITED'))
        self.registration=self.out/'registration.json';reg=dict(status='REGISTERED',protocol=u.CONTRACT,root=str(self.root),development_tokens=str(self.out/'a2_development_tokens.npz'),
            a1_release_gate=str(self.gate),calibration_payloads=str(self.cal),source_bindings={str(dep):u.sha(dep)},input_bindings={str(self.cal):u.sha(self.cal)})
        u.write(self.registration,reg);self.table=self.out/'calibration_counts.csv'
        points=[dict(codec=c,snr_db=s,n_blocks=100,n_reject=10 if s==7 else 70) for c in ['C','L'] for s in [1,4,7,10,19]]
        with self.table.open('w',newline='') as f:w=csv.DictWriter(f,points[0].keys());w.writeheader();w.writerows(points)
        self.policy=self.out/'frozen_workpoints.json';self.policy_value=dict(u.transition(points),registration_sha256=u.sha(self.registration),calibration_table_sha256=u.sha(self.table),
            payload=dict(N=1024,phy='16QAM',m=8,K=0,order='whole'),source_population_sha256=u.sha(self.cal),calibration_total_blocks=1000)
        self.seal_policy()

    def seal_policy(self):
        u.write(self.policy,self.policy_value)
        u.write(self.out/'calibrate_completion.json',dict(status='COMPLETE',stage='calibrate',registration_sha256=u.sha(self.registration),development_read=False,training_updates=0,physical_blocks=1000,
            outputs={str(self.policy):u.sha(self.policy),str(self.table):u.sha(self.table)}))

    def freeze(self):return exporter.development_freeze(self.registration,self.gate,self.policy,self.root,self.out)

    def test_valid_policy_is_independently_recomputed(self):
        with patch('a2_run.gate') as gate:
            result=self.freeze();gate.assert_called_once();self.assertEqual(result[str(self.policy)],u.sha(self.policy))

    def test_non_frozen_policy_never_reads_development(self):
        self.policy_value['status']='INCOMPLETE';self.seal_policy()
        with patch('a2_run.gate'),patch.object(exporter.np,'load') as load:
            with self.assertRaisesRegex(RuntimeError,'selection provenance'):self.freeze()
            load.assert_not_called()

    def test_wrong_calibration_winner_rejected_even_if_receipt_resealed(self):
        self.policy_value['snrs_db']=[13,19];self.seal_policy()
        with patch('a2_run.gate'):
            with self.assertRaisesRegex(RuntimeError,'transition differs'):self.freeze()

    def test_changed_calibration_csv_rejected(self):
        self.table.write_text(self.table.read_text().replace('C,7,100,10','C,7,100,70'))
        with patch('a2_run.gate'):
            with self.assertRaisesRegex(RuntimeError,'input changed'):self.freeze()

    def test_live_a1_gate_blocks_export(self):
        with patch('a2_run.gate',side_effect=RuntimeError('A1 still active')):
            with self.assertRaisesRegex(RuntimeError,'A1 still active'):self.freeze()


if __name__=='__main__':unittest.main()
