    def test_budget_stop_and_failure_no_retry(self):
        with tempfile.TemporaryDirectory() as td:
            root=Path(td);budget=root/'budget';budget.write_text('{}');before=dict(charged=164760,phase_charged={'development':10800},development_remaining=2400,failed=0,unresolved=0)
            fake=SimpleNamespace(budget_snapshot=lambda *a,**k:copy.deepcopy(before))
            ctx=dict(cfg=dict(ledger='db',budget_registration=str(budget),phase_limits={},stop_file=str(root/'STOP')),
                d=SimpleNamespace(DEADLINE=1e20),owner=fake,before=copy.deepcopy(before))
            op.unchanged(ctx);before['charged']+=1
            with self.assertRaisesRegex(RuntimeError,'budget changed'):op.unchanged(ctx)
            (root/'STOP').write_text('stop')
            with self.assertRaisesRegex(RuntimeError,'STOP'):op.unchanged(ctx)
            q=root/'q';q.mkdir();op.save(q/'failure.json',{'preserved':True})
            with self.assertRaisesRegex(RuntimeError,'failed qualification'):op.qualified({'loc':{'qualification':q}})
            e=root/'e';e.mkdir();out=root/'out';op.fresh_launch(e,out)
            op.save(e/'launch_attempt.json',{'once':True});raw=(e/'launch_attempt.json').read_bytes()
            with self.assertRaisesRegex(RuntimeError,'no retry'):op.fresh_launch(e,out)
            with self.assertRaises(FileExistsError):op.save(e/'launch_attempt.json',{})
            self.assertEqual(raw,(e/'launch_attempt.json').read_bytes())

    def test_one_P600_GPU_only_and_original_owner_overrides_hidden_CUDA(self):
        reg=dict(allowed_stage_ids=['development'],source_stage_scope='P600_EXISTING_RECEIVED_LATENT_RGB_AND_INDEPENDENT_CONVNEXT_ONLY',
            GPU_jobs=1,GPU_threads=6,new_packet_decodes=0,new_noise_draws=0,budget_writes=0,policy_selection=False,
            MAIN_started=False,C_started=False,future_stage_automatic=False)
        exp=dict(frame_count=600,source_count=100,noise_seeds=[2001,2002,2003],RGB_exact_parity=True,ConvNeXt_scored=True,
            new_packet_decodes=0,new_noise_draws=0,budget_writes=0,MAIN_complete=False,overall_development_complete=False)
        owner=dict(gpu_threads=6,stages=[dict(id='development',resource='gpu',jobs=[dict(id='p600_received_replay',argv=['UM'],receipt_expect=exp)])])
        op.launch_shape(owner,reg,'UM')
        for k,v in [('GPU_jobs',2),('new_noise_draws',1),('MAIN_started',True),('future_stage_automatic',True)]:
            with self.assertRaises(RuntimeError):op.launch_shape(owner,dict(reg,**{k:v}),'UM')
        with self.assertRaises(RuntimeError):op.launch_shape(owner,reg,'LDPC')
        path=Path(__file__).absolute().parent.parent/'runner/stage_owner.py'
        tree=ast.parse(path.read_text())
        expressions=[k.value for n in ast.walk(tree) if isinstance(n,ast.Call) and isinstance(n.func,ast.Attribute)
            and isinstance(n.func.value,ast.Name) and n.func.value.id=='env' and n.func.attr=='update'
            for k in n.keywords if k.arg=='CUDA_VISIBLE_DEVICES']
        self.assertEqual(len(expressions),1)
        value=eval(compile(ast.Expression(expressions[0]),'<original_owner>','eval'),
            {'resource':'gpu','self':SimpleNamespace(cfg={'gpu_device':0})})
        self.assertEqual(value,'0')

    def test_distinct_steps_fresh_paths_no_automatic_calls(self):
        tree=ast.parse(Path(op.__file__).read_text())
        for name in ('qualify','register','launch'):
            node=next(n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name==name)
            calls={n.func.id for n in ast.walk(node) if isinstance(n,ast.Call) and isinstance(n.func,ast.Name)}
            self.assertFalse(({'qualify','register','launch'}-{name}) & calls)
        loc=op.locations('/fixture')
        self.assertEqual(loc['execution'].name,'p600_replay_execution_v1');self.assertEqual(loc['out'].name,'p600_replay_v1')
        self.assertEqual(loc['qualification'].name,'p600_replay_cpu_qualification_v1')
        self.assertEqual(loc['operations'].name,'p600_replay_operations_v1')
