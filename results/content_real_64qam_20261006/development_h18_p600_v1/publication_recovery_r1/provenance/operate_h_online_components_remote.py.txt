def require(ok,message):
    if not ok:raise RuntimeError(message)
def read(p):return json.loads(Path(p).read_text(encoding='utf-8-sig'))
def sha(p):
    h=hashlib.sha256()
    with Path(p).open('rb') as f:
        for block in iter(lambda:f.read(8*1024**2),b''):h.update(block)
    return h.hexdigest()
def bind(paths):return {str(p):sha(p) for p in paths}
def pin(p):return dict(path=str(p),sha256=sha(p))
def merge(*maps):
    result={}
    for values in maps:
        for p,s in values.items():
            require(p not in result or result[p]==s,'Conflicting input SHA: '+p);result[p]=s
    return result
def verify(values):
    for p,s in values.items():require(Path(p).is_absolute() and sha(p)==s,'Changed frozen dependency: '+p)
def save(p,value):
    p=Path(p);p.parent.mkdir(parents=True,exist_ok=True)
    with p.open('x',encoding='utf-8',newline='\n') as f:
        json.dump(value,f,sort_keys=True,indent=2,allow_nan=False);f.write('\n');f.flush();os.fsync(f.fileno())
def module(p,name):
    spec=importlib.util.spec_from_file_location(name,p);m=importlib.util.module_from_spec(spec)
    sys.modules[name]=m;spec.loader.exec_module(m);return m


def locations(root):
    base=Path(root).absolute()/'outputs/CONTENT-REAL-64QAM-20261006';h=base/'H';prep=base/'prepared_v1'
    return dict(h=h,qualification=prep/'h_online_components_qualification_v1',

        out=h/'online_components_v1')


def code_locations(pcfg,preg):
    entry=pcfg['replay_driver_module'];bound=merge(preg['source_bindings'],preg['input_bindings'])
    require(bound.get(entry)==sha(entry)==P_HELPER_SHA,'Original normally executed P600 helper differs')
    code=Path(entry).parent;fresh={str(code/n):s for n,s in PINS.items()};verify(fresh)
    # Never infer runtime files from this helper's private parent.
    private=Path(__file__).absolute().parent
    require((private/'test_operate_h_online_components_remote.py').is_file(),'Private helper tests missing')
    return code,private,fresh


def original_python(cfg,owner,cp,root):
    require(len(owner['stages'])==1 and owner['stages'][0]['id']=='development'
        and owner['stages'][0]['resource']=='gpu' and len(owner['stages'][0]['jobs'])==1,'One completed P600 GPU job required')
    job=owner['stages'][0]['jobs'][0];argv=job['argv'];python=argv[0]
    require(job['id']=='p600_received_replay' and argv[1:]==['-B',cfg['replay_driver_module'],'--config',str(cp)]
        and owner['gpu_threads']==6 and Path(python).is_absolute() and Path(python).is_file()
        and '/UNIFIED-METRICS-' in python.replace('\\','/') and Path(root).absolute() in Path(python).parents,
        'Exact original UM venv path/command required')
    return python  # The venv symlink path is intentional; do not resolve it.


def environment(root,code,private):
    return dict(os.environ,CUDA_VISIBLE_DEVICES='',OMP_NUM_THREADS='2',MKL_NUM_THREADS='2',OPENBLAS_NUM_THREADS='2',
        NUMEXPR_NUM_THREADS='2',PYTHONDONTWRITEBYTECODE='1',
        PYTHONPATH=os.pathsep.join((str(code),str(private))),H64_REFERENCE_VAR_COMM=str(Path(root)/'src/var_comm'))


def unchanged(ctx):
    cfg=ctx['mc'];now=ctx['owner'].budget_snapshot(cfg['ledger'],sha(cfg['budget_registration']),cfg['phase_limits'],quiescent=True)
    require(now==ctx['before'] and now['charged']==164760 and now['phase_charged']['development']==10800
        and now['development_remaining']==2400 and now['failed']==now['unresolved']==0,'Original H budget changed')
    require(time.time()<ctx['d'].DEADLINE and not Path(cfg['stop_file']).exists(),'STOP/deadline blocks timing')


def context(root,p600_config,*,normal_closure=False):
    require(sys.platform.startswith('linux'),'Actual Linux operations/process evidence required')
    root=str(Path(root).absolute());loc=locations(root);cp=Path(p600_config).absolute();pcfg=read(cp)
    preg=read(pcfg['registration']);po=read(pcfg['visual_owner_config']);bound=merge(preg['source_bindings'],preg['input_bindings'])
    require(pcfg['root']==root and pcfg['H_out']==str(loc['h']) and pcfg['stop_file']==str(loc['h']/'STOP'),
        'Wrong original H/P workspace')
    for p in (str(cp),pcfg['visual_owner_config'],pcfg['owner_module']):require(bound.get(p)==sha(p),'Unbound original P config/owner')
    code,private,fresh=code_locations(pcfg,preg);sys.path.insert(0,str(code))
    tool=module(code/'register_h_online_components.py','private_h_timing_registrar');d=tool.d
    owner=module(pcfg['owner_module'],'private_timing_original_owner');python=original_python(pcfg,po,cp,root)
    spec=dict(config=str(cp),owner_config=pcfg['visual_owner_config'],registration=pcfg['registration'],
        launch=str(Path(po['owner_out'])/'launch.json'),completion=str(Path(pcfg['out'])/'completion.json'))
    ms=pcfg['metric_batch'];mc=read(ms['config']);mr=read(ms['registration']);visual=read(mc['render_batch']['config'])
    for p in ms.values():require(bound.get(p)==sha(p),'Original H metric predecessor unbound')
    require(mc['metric_driver_module']==pcfg['h_metric_driver_module']
        and mr['input_bindings'].get(mc['render_batch']['config'])==sha(mc['render_batch']['config']),
        'Original metric/render lineage differs')
    helper=module(pcfg['replay_driver_module'],'private_timing_P600_metadata_helper')
    prior=helper.original_metrics_closed(ms,pcfg['h_metric_driver_module']) if normal_closure else None
    before=prior['before'] if prior else read(spec['completion'])['budget_after']
    stub=dict(cfg=dict(mc,metric_batch=ms),prior=prior,owner=owner,before=before,
        source_ids=prior['context']['source_ids'] if prior else None)
    pclosed=tool.normal_p600_closed(spec,stub) if normal_closure else None
    ctx=dict(root=root,loc=loc,cp=cp,pcfg=pcfg,preg=preg,po=po,bound=bound,code=code,private=private,fresh=fresh,
        tool=tool,d=d,owner=owner,python=python,spec=spec,ms=ms,mc=mc,mr=mr,visual=visual,
        prior=prior,pclosed=pclosed,before=before)
    unchanged(ctx);return ctx


def qualification_sources(ctx):
    # Existing registered sources cover all unchanged sibling fixture imports.
    return merge(ctx['preg']['source_bindings'],ctx['fresh'],
        bind((Path(__file__).absolute(),ctx['private']/'test_operate_h_online_components_remote.py')))


def qualified(ctx):
    q=ctx['loc']['qualification'];require(not (q/'failure.json').exists(),'Prior timing qualification failed; no retry')
    value=read(q/'completion.json');argv=[ctx['python'],'-B','-m','unittest','-v',*TESTS]
    require(value['status']==ctx['tool'].QUALIFIED and value['python']==ctx['python']
        and value['results']==[dict(argv=argv,exit_code=0,test_count=TEST_COUNT)]
        and value['GPU_used'] is False and value['models_constructed'] is False
        and value['new_packet_decodes']==value['new_noise_draws']==0,'Exact CPU-only timing qualification required')
    for k in ('source_bindings','input_bindings','outputs'):verify(value[k])
    require(value['source_bindings']==qualification_sources(ctx)
        and value['budget_before']==value['budget_after']==ctx['before'],'Qualification sources/budget differ')
    text=(q/'tests.log').read_text(encoding='utf-8',errors='replace')
    require(f'Ran {TEST_COUNT} tests' in text and text.rstrip().endswith('OK'),'Exact timing test inventory required')
    return value


def qualify(root,p600_config):
    ctx=context(root,p600_config,normal_closure=True);q=ctx['loc']['qualification']
    require(not q.exists(),'Prior qualification preserved; no retry');sources=qualification_sources(ctx);verify(sources)
    inputs=merge(bind(ctx['spec'].values()),bind(ctx['ms'].values()))
    affinity=ctx['po']['cpu_affinities'][0]
    require(len(affinity)==len(set(affinity))==2 and set(affinity)<=os.sched_getaffinity(0),'Original two CPU cores unavailable')
    q.mkdir(parents=True);gate=q/'predecessor_closure.json'
    save(gate,dict(status='NORMAL_H_METRICS_AND_P600_CLOSED_BEFORE_TIMING',metric_bindings=ctx['prior']['bindings'],
        P600_bindings=ctx['pclosed']['bindings'],metric_batch=ctx['ms'],p600_batch=ctx['spec'],
        budget=ctx['before'],normal_owner_success=True,new_packet_decodes=0,GPU_used=False))
    inputs=merge(inputs,bind((gate,)));argv=[ctx['python'],'-B','-m','unittest','-v',*TESTS]
    save(q/'attempt.json',dict(argv=argv,source_bindings=sources,input_bindings=inputs,python=ctx['python'],
        cpu_affinity=affinity,nice=15,GPU_used=False,models_constructed=False,new_packet_decodes=0,new_noise_draws=0))
    def resources():os.sched_setaffinity(0,affinity);os.setpriority(os.PRIO_PROCESS,0,15)
    try:
        with (q/'tests.log').open('xb') as log:
            result=subprocess.run(argv,cwd=ctx['code'],env=environment(root,ctx['code'],ctx['private']),
                stdout=log,stderr=subprocess.STDOUT,check=False,timeout=1800,preexec_fn=resources)
        text=(q/'tests.log').read_text(encoding='utf-8',errors='replace')
        require(result.returncode==0 and f'Ran {TEST_COUNT} tests' in text and text.rstrip().endswith('OK'),
            'Timing CPU qualification failed or inventory differs')
        verify(sources);verify(inputs);unchanged(ctx)
        done=dict(status=ctx['tool'].QUALIFIED,python=ctx['python'],results=[dict(argv=argv,exit_code=0,test_count=TEST_COUNT)],
            source_bindings=sources,input_bindings=inputs,outputs=bind((q/'attempt.json',q/'tests.log',gate)),
            GPU_used=False,models_constructed=False,new_packet_decodes=0,new_noise_draws=0,
            budget_before=ctx['before'],budget_after=ctx['before'],workers_started=False)
        save(q/'completion.json',done);qualified(ctx);return dict(status=done['status'],completion=pin(q/'completion.json'))
    except BaseException:
        save(q/'failure.json',dict(status='FAILED_PRESERVE_NO_RETRY',traceback=traceback.format_exc()));raise


def build_request(ctx,q):
    loc,code,mc,visual=ctx['loc'],ctx['code'],ctx['mc'],ctx['visual']
    cfg={k:mc[k] for k in ('root','H_out','owner_module','wait_module','cpu_driver_module','protocol',
        'budget_registration','ledger','phase_limits','stop_file')}
    cfg.update({k:visual[k] for k in ('runtime_dir','native_runtime','uep_runtime','var_source','dino_source',
        'source_driver_module','receiver_module','numerical_reference','static_closure_module')})
    cfg.update(schema='H_ONLINE_COMPONENTS_CONFIG_V1',out=str(loc['out']),
        registration=str(loc['execution']/'execution_registration.json'),visual_owner_config=str(loc['execution']/'owner_config.json'),
        metadata_helper_module=ctx['pcfg']['replay_driver_module'],timing_core_module=str(code/'h_online_components.py'),
        timing_driver_module=str(code/'h_online_components_driver.py'),h_metric_driver_module=mc['metric_driver_module'],
        quality_driver_module=str(Path(visual['uep_runtime'])/'quality_driver.py'),
        codec_module=str(Path(visual['runtime_dir'])/'h64_source.py'),metric_batch=ctx['ms'],
        visual_source_closure=str(loc['operations']/'visual_source_closure.json'),
        max_seconds=14400,overall_deadline_unix=ctx['d'].DEADLINE)
    closure=module(cfg['static_closure_module'],'private_timing_current_visual_closure')
    actual=closure.collect_bindings(ctx['root'],cfg['native_runtime'],cfg['var_source'],cfg['dino_source'],cfg['uep_runtime'])
    oldgraphs=[visual['visual_source_closure'],ctx['pcfg']['visual_source_closure']]
    for path in oldgraphs:
        g=read(path);require(g['status']=='EXACT_SOURCE_CLOSURE_MATCH'
            and all(actual.get(p)==s for p,s in g['source_bindings'].items()),'Executed native source changed')
    save(cfg['visual_source_closure'],dict(status='EXACT_SOURCE_CLOSURE_MATCH',source_bindings=actual,
        prior_graphs=[pin(p) for p in oldgraphs],old_registered_files_changed=False,source_enumeration_only=True,GPU_used=False))
    gate=read(loc['qualification']/'predecessor_closure.json')
    sources=merge(ctx['preg']['source_bindings'],actual,q['source_bindings'])
    inputs=merge(ctx['preg']['input_bindings'],gate['metric_bindings'],gate['P600_bindings'],
        q['input_bindings'],q['outputs'],bind((loc['qualification']/'completion.json',cfg['visual_source_closure'])))
    return dict(schema='H_ONLINE_COMPONENTS_REGISTRATION_REQUEST_V1',config=cfg,execution_dir=str(loc['execution']),
        p600_batch=ctx['spec'],source_bindings=sources,input_bindings=inputs,
        prepared_qualification=str(loc['qualification']/'completion.json'))


def register(root,p600_config):
    ctx=context(root,p600_config);q=qualified(ctx);loc=ctx['loc']
    require(not any(loc[k].exists() for k in ('operations','execution','out')),'Prior timing request/output preserved')
    loc['operations'].mkdir(parents=True);save(loc['operations']/'attempt.json',dict(status='TIMING_REGISTRATION_ATTEMPT',helper=pin(__file__)))
    try:
        unchanged(ctx);request=build_request(ctx,q);rp=loc['operations']/'request.json';save(rp,request)
        result=ctx['tool'].register(rp);require(result['status']==ctx['tool'].REGISTERED,'Timing registration incomplete')
        return dict(status=result['status'],completion=pin(loc['execution']/'registration_completion.json'))
    except BaseException:
        save(loc['operations']/'registration_failure.json',dict(status='FAILED_PRESERVE_NO_LAUNCH',traceback=traceback.format_exc()));raise


def fresh_launch(e,out):
    require(not any((Path(e)/n).exists() for n in ('launch.json','launch_attempt.json','owner_identity.json','owner.log',
        'failure.json','registration_failure.json','launch_failure.json','completion.json','STOP')) and not Path(out).exists(),
        'Prior timing launch/output preserved; no retry')


def launch_shape(owner,reg,python):
    require(reg['allowed_stage_ids']==['development'] and reg['source_stage_scope']=='H_FIXED16_18_POINTS_FRESH_ONLINE_COMPONENTS_NO_PHY'
        and reg['GPU_jobs']==1 and reg['GPU_threads']==owner['gpu_threads']==6
        and reg['new_packet_decodes']==reg['new_noise_draws']==reg['budget_writes']==reg['new_quality_samples']==0
        and reg['future_stage_automatic'] is False and reg['MAIN_started'] is False and reg['C_started'] is False,
        'Only registered fixed timing may launch')
    require(len(owner['stages'])==1 and owner['stages'][0]['id']=='development' and owner['stages'][0]['resource']=='gpu'
        and len(owner['stages'][0]['jobs'])==1,'Only one timing GPU job allowed')
    job=owner['stages'][0]['jobs'][0];e=job['receipt_expect']
    require(job['id']=='h_online_components' and job['argv'][0]==python
        and e['source_count']==16 and e['component_case_count']==288 and e['noise_seed']==6201
        and e['warmup_repetitions']==1 and e['measured_repetitions']==3
        and e['exact_TX_parity'] is True and e['exact_RX_parity'] is True
        and e['new_packet_decodes']==e['new_noise_draws']==e['new_quality_samples']==0
        and e['end_to_end_latency_measured'] is False,'Wrong component timing GPU job')


def launch(root,p600_config):
    ctx=context(root,p600_config);qualified(ctx);loc=ctx['loc'];e=loc['execution'];fresh_launch(e,loc['out'])
    require(not (loc['operations']/'registration_failure.json').exists(),'Timing registration failed')
    cp,rp,op=e/'timing_config.json',e/'execution_registration.json',e/'owner_config.json'
    seal,reg,owner=read(e/'registration_completion.json'),read(rp),read(op)
    require(seal['status']==ctx['tool'].REGISTERED and seal['registration_sha256']==sha(rp)
        and seal['owner_config_sha256']==sha(op) and reg['p600_batch']==ctx['spec'],'Exact timing registration required')
    verify(seal['outputs']);ctx['owner'].validate_config(owner,reg,sha(op));launch_shape(owner,reg,ctx['python'])
    ready=ctx['d'].load_registered(cp);pclosed=ctx['tool'].normal_p600_closed(ctx['spec'],ready)
    require(ready['before']==ctx['before'] and all(ready['bound'].get(p)==s for p,s in pclosed['bindings'].items()),
        'Launch P600 lineage/budget no longer matches registration');unchanged(ctx)
    argv=[ctx['python'],'-B',ctx['mc']['owner_module'],'--config',str(op)]
    save(e/'launch_attempt.json',dict(status='LAUNCH_CLAIMED_NO_RETRY',argv=argv,registration_sha256=sha(rp),helper=pin(__file__)))
    try:
        with (e/'owner.log').open('xb') as log:
            process=subprocess.Popen(argv,cwd=ctx['root'],env=environment(root,ctx['code'],ctx['private']),stdin=subprocess.DEVNULL,
                stdout=log,stderr=subprocess.STDOUT,start_new_session=True)
            identity=ctx['owner'].identity(process.pid)
            require(identity['argv']==argv and int(identity['uid'])==os.getuid(),'Original owner process identity differs')
            value=dict(identity=identity,argv=argv,registration_sha256=sha(rp),owner_config_sha256=sha(op),launched_unix=time.time(),
                launcher_sha256=sha(__file__),launch_attempt_sha256=sha(e/'launch_attempt.json'),
                GPU_jobs=1,source_count=16,component_case_count=288,new_packet_decodes=0,new_noise_draws=0,future_stages_automatic=False)
            save(e/'launch.json',value)
        return dict(status='H_ONLINE_COMPONENTS_OWNER_LAUNCHED',launch=pin(e/'launch.json'),**value)
    except BaseException:
        save(e/'launch_failure.json',dict(status='FAILED_PRESERVE_NO_RESTART',pid=process.pid if 'process' in locals() else None,
            traceback=traceback.format_exc()));raise
