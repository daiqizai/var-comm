def require(ok,message):
    if not ok:raise RuntimeError(message)
def read(p):return json.loads(Path(p).read_text(encoding='utf-8-sig'))
def sha(p):
    h=hashlib.sha256()
    with Path(p).open('rb') as f:
        for block in iter(lambda:f.read(1048576),b''):h.update(block)
    return h.hexdigest()
def pin(p):return dict(path=str(p),sha256=sha(p))
def bind(paths):return {str(Path(p).absolute()):sha(p) for p in paths}
def verify(values):
    for p,s in values.items():require(Path(p).is_absolute() and len(s)==64 and sha(p)==s,'Changed bound file: '+p)
def merge(*maps):
    out={}
    for values in maps:
        for p,s in values.items():
            require(p not in out or out[p]==s,'Conflicting input: '+p);out[p]=s
    return out
def save(p,value):
    p=Path(p);p.parent.mkdir(parents=True,exist_ok=True)
    with p.open('x',encoding='utf-8',newline='\n') as f:
        json.dump(value,f,sort_keys=True,indent=2,allow_nan=False);f.write('\n');f.flush();os.fsync(f.fileno())
def module(p,name):
    sys.path.insert(0,str(Path(p).parent));spec=importlib.util.spec_from_file_location(name,p)
    m=importlib.util.module_from_spec(spec);sys.modules[name]=m;spec.loader.exec_module(m);return m

    require(hc['root']==pc['root']==root and pc['h_metric_driver_module']==hc['metric_driver_module'],
        'Same actual H/P workspace and metric predecessor required')
    code=Path(hc['metric_driver_module']).parent
    sources=merge(hr['source_bindings'],pr['source_bindings'],{str(code/n):s for n,s in PINS.items()},
        bind((Path(__file__).absolute(),Path(__file__).with_name('test_operate_h_p_statistics_remote.py'))))
    verify(sources);h,p=batch(hp,hc),batch(pp,pc)
    require(pc['metric_batch']==h,'P600 must use this exact original H metric batch')
    inputs=merge(hr['input_bindings'],pr['input_bindings'],bind([*h.values(),*p.values(),str(Path(template).absolute())]))
    hreq=dict(metric_driver=hc['metric_driver_module'],metric_batch=h,source_bindings=sources,input_bindings=inputs)
    return dict(schema='H_P_FIXED_STATISTICS_REQUEST_V1',root=root,metadata_template=str(Path(template).absolute()),

def qualified(x):
    q=x['loc']['qualification'];require(not(q/'failure.json').exists(),'Qualification failure preserved')
    value=read(q/'completion.json');argv=[x['python'],'-B','-m','unittest','-v',*TESTS]
    require(value['status']==x['d'].QUALIFIED and value['python']==x['python'] and value['GPU_used'] is False
        and value['new_packet_decodes']==0 and value['actual_quality_rows_read'] is False
        and value['results']==[dict(argv=argv,exit_code=0,test_count=TEST_COUNT)],'Exact synthetic13 CPU qualification required')
    for k in ('source_bindings','input_bindings','outputs'):verify(value[k])
    require(value['source_bindings']==x['sources'] and value['budget_before']==value['budget_after']==x['before'],
        'Qualified source/budget differs')
    for p,s in x['bindings'].items():require(value['input_bindings'].get(p)==s,'Qualifier predecessor differs: '+p)
    text=(q/'tests.log').read_text(encoding='utf-8',errors='replace')
    require(value['outputs'].get(str(q/'tests.log'))==sha(q/'tests.log') and f'Ran {TEST_COUNT} tests' in text
        and text.rstrip().endswith('OK'),'Unexpected actual qualification inventory')
    return value


def qualify(root,h_config,p_config,template):
    x=context(root,h_config,p_config,template);q=x['loc']['qualification'];require(not q.exists(),'Prior qualifier preserved; no retry')
    affinity=x['hclosed']['owner']['cpu_affinities'][0]
    require(len(affinity)==len(set(affinity))==2 and set(affinity)<=os.sched_getaffinity(0),'Two original CPU cores unavailable')
    x['d'].unchanged(x);q.mkdir(parents=True)
    argv=[x['python'],'-B','-m','unittest','-v',*TESTS]
    save(q/'attempt.json',dict(argv=argv,source_bindings=x['sources'],input_bindings=x['bindings'],python=x['python'],
        affinity=affinity,threads=2,nice=15,GPU_used=False,new_packet_decodes=0,actual_quality_rows_read=False))
    def resources():os.sched_setaffinity(0,affinity);os.setpriority(os.PRIO_PROCESS,0,15)
    try:
        with (q/'tests.log').open('xb') as log:
            result=subprocess.run(argv,cwd=x['code'],env=environment(root,x['code']),stdout=log,stderr=subprocess.STDOUT,
                timeout=1800,check=False,preexec_fn=resources)
        text=(q/'tests.log').read_text(encoding='utf-8',errors='replace')
        require(result.returncode==0 and f'Ran {TEST_COUNT} tests' in text and text.rstrip().endswith('OK'),
            'CPU comparison qualification failed; no retry')
        x['d'].unchanged(x)
        done=dict(status=x['d'].QUALIFIED,python=x['python'],results=[dict(argv=argv,exit_code=0,test_count=TEST_COUNT)],
            source_bindings=x['sources'],input_bindings=x['bindings'],outputs=bind((q/'attempt.json',q/'tests.log')),
            GPU_used=False,new_packet_decodes=0,actual_quality_rows_read=False,budget_before=x['before'],budget_after=x['before'])
        save(q/'completion.json',done);qualified(x);return dict(status=done['status'],completion=pin(q/'completion.json'))
    except BaseException:
        save(q/'failure.json',dict(status='FAILED_PRESERVE_NO_RETRY',traceback=traceback.format_exc()));raise

def fresh_launch(e,out):
    require(not any((e/n).exists() for n in ('launch.json','launch_attempt.json','owner_identity.json','owner.log',
        'completion.json','failure.json','registration_failure.json','launch_failure.json','STOP')) and not out.exists(),
        'Prior launch/output evidence preserved; no retry')


def launch(root,h_config,p_config,template):
    x=context(root,h_config,p_config,template);qualified(x);d=x['d'];e=x['loc']['execution'];fresh_launch(e,x['loc']['out'])
    cp=e/'statistics_config.json';seal=read(e/'registration_completion.json');cfg=read(cp)
    require(seal['status']==REGISTERED and seal['registration_sha256']==sha(cfg['registration'])
        and seal['owner_config_sha256']==sha(cfg['owner_config']),'Successful comparison registration required')
    verify(seal['outputs']);ready=d.registered_context(cp)
    require(ready['request']==x['request'] and ready['before']==x['before'],'Launch request/budget differs');d.unchanged(x)
    argv=[x['python'],'-B',x['hctx']['cfg']['owner_module'],'--config',cfg['owner_config']]
    save(e/'launch_attempt.json',dict(status='LAUNCH_CLAIMED_NO_RETRY',argv=argv,registration_sha256=sha(cfg['registration']),helper=pin(__file__)))
    try:
        with (e/'owner.log').open('xb') as log:
            p=subprocess.Popen(argv,cwd=x['root'],env=environment(root,x['code']),stdin=subprocess.DEVNULL,
                stdout=log,stderr=subprocess.STDOUT,start_new_session=True)
            ident=x['owner'].identity(p.pid)
            require(ident['argv']==argv and int(ident['uid'])==os.getuid(),'Actual CPU owner identity differs')
            value=dict(identity=ident,argv=argv,registration_sha256=sha(cfg['registration']),owner_config_sha256=sha(cfg['owner_config']),
                launched_unix=time.time(),launch_attempt_sha256=sha(e/'launch_attempt.json'),GPU_jobs=0,new_packet_decodes=0,future_stage_automatic=False)
            save(e/'launch.json',value)
        return dict(status='H_P_FIXED_STATISTICS_OWNER_LAUNCHED',launch=pin(e/'launch.json'))
    except BaseException:
        save(e/'launch_failure.json',dict(status='FAILED_PRESERVE_NO_RESTART',pid=p.pid if 'p' in locals() else None,
            traceback=traceback.format_exc()));raise
