def require(value,message):
    if not value:raise RuntimeError(message)
def read(path):return json.loads(Path(path).read_text(encoding='utf-8-sig'))
def sha(path):return hashlib.sha256(Path(path).read_bytes()).hexdigest()
def save(path,value):
    p=Path(path);p.parent.mkdir(parents=True,exist_ok=True)
    with p.open('x',encoding='utf-8',newline='\n') as f:
        json.dump(value,f,indent=2,ensure_ascii=False,allow_nan=False);f.write('\n');f.flush();os.fsync(f.fileno())
def bind(paths):return {str(p):sha(p) for p in paths}
def verify(values):
    for p,s in values.items():require(Path(p).is_absolute() and sha(p)==s,'Changed evidence: '+p)
def merge(*maps):
    out={}
    for values in maps:
        for p,s in values.items():require(p not in out or out[p]==s,'Conflicting evidence');out[p]=s
    return out
def pin(value):
    require(set(value)=={'path','sha256'} and Path(value['path']).is_absolute()
        and sha(value['path'])==value['sha256'],'Changed requested receipt')
    return Path(value['path'])

def git(root,*args):return subprocess.check_output(['git',*args],cwd=root,env=env(),text=True,timeout=120).strip()
def blob_sha(root,path):return hashlib.sha256(subprocess.check_output(['git','show',':'+path],cwd=root,env=env(),timeout=120)).hexdigest()

def archive_evidence(p):

    require(sha(ap)==ARCHIVE_SHA and sha(dp)==DIAGNOSIS_SHA,'Exact original publication failure required')
    a,d=read(ap),read(dp);expected={str(p['old']/n) for n in ('stage.log','private_export_receipt.json','diff_check.log',
        'publication_files.json','repository.log','failure.json','export_request.json','execution_registration.json')}
    expected.add(str(p['prep']/'publish_h18_descriptive_p600_v1.py'));seen=set();copies=set();evidence=bind((ap,dp))
    require(a['status']=='PUBLICATION_SCOPE_FAILURE_ARCHIVED_ORIGINALS_AND_INDEX_PRESERVED'
        and a['committed'] is a['pushed'] is False and a['new_scientific_calls']==0 and len(a['entries'])==9,'Exact9 archived inputs required')
    for item in a['entries']:
        src,dst=Path(item['original']),Path(item['archive'])
        require(str(src) not in seen and str(dst) not in copies and (p['rec']/'original_attempt') in dst.parents
            and sha(src)==sha(dst)==item['sha256'],'Original/archive content or placement changed')
        seen.add(str(src));copies.add(str(dst));evidence.update({str(src):item['sha256'],str(dst):item['sha256']})
    require(seen==expected and d['archive_receipt_sha256']==ARCHIVE_SHA and d['old_checker_unchanged']
        and d['old_results_unchanged'] and d['old_index_preserved'],'Failure evidence chain differs')
    original=read(p['old']/'publication_files.json')
    require(len(original)==53 and original==a['staged_files'] and not(p['old']/'completion.json').exists(),'Original53 files and failure must remain')
    require(sha(p['prep']/'publish_h18_descriptive_p600_v1.py')==ORIGINAL_HELPER_SHA,'Original publisher changed')
    log=(p['old']/'repository.log').read_text();require('untracked self-written source:' in log
        and all('experiments/content-real-64qam-20261006/'+n in log for n in TIMING),'A different failure needs separate recovery')
    oldreg=read(p['old']/'execution_registration.json');export=read(p['old']/'export_request.json')

def qualification(request_pin,kind,p):
    qpath=pin(request_pin);q=read(qpath);timing=kind=='timing';count=39 if timing else 13
    status='H_ONLINE_COMPONENTS_CPU_INTERFACE_QUALIFICATION_PASS' if timing else 'H_P_STATISTICS_CPU_QUALIFICATION_PASS'

    tests=TIMING_TESTS if timing else HP_TESTS
    require(q['status']==status and q['python']==py and q['GPU_used'] is False and q['new_packet_decodes']==0
        and q['results']==[dict(argv=[py,'-B','-m','unittest','-v',*tests],exit_code=0,test_count=count)],'Actual qualified runtime/test inventory differs')
    if timing:require(q['models_constructed'] is False and q['new_noise_draws']==0,'Timing qualification performed science')
    else:require(q['actual_quality_rows_read'] is False,'H/P qualification read actual score rows')
    require(not(qpath.parent/'failure.json').exists(),'Qualification failure cannot be exported as PASS')
    require(q['budget_before']==q['budget_after'] and q['budget_after']['charged']==164760
        and q['budget_after']['failed']==q['budget_after']['unresolved']==0,'Qualification changed H budget')

def check_index(p,expected,*,new_untracked):
    root=p['root'];require(git(root,'rev-parse','HEAD')==PARENT and git(root,'branch','--show-current')=='main','Parent/branch changed')
    require(git(root,'ls-remote','origin','refs/heads/main').split()[0]==PARENT,'Remote parent changed')
    require(not git(root,'diff','--name-only'),'Unstaged tracked work must be preserved')
    require(set(git(root,'diff','--cached','--name-only').splitlines())==set(expected),'Index not the exact owned set')
    for name,s in expected.items():require(sha(root/name)==s and blob_sha(root,name)==s,'Staged blob differs from immutable working byte')
    untracked=set(git(root,'ls-files','--others','--exclude-standard','--','experiments/content-real-64qam-20261006').splitlines())
    py={name for name in untracked if name.endswith('.py')}
    require(py==set(new_untracked),'Additional untracked engineering requires separate scope; nothing is removed')


def prestage(request_pin,p,original,new):
    cp=pin(request_pin);rp=cp.parent/'execution_registration.json';c,r=read(cp),read(rp)

        and c['new_PHY']==0 and c['committed'] is c['pushed'] is False,'Exact independently recorded64-file prestage required')
    require(r['status']=='REVIEWED_ELEVEN_ENGINEERING_SOURCES_STAGE_ONLY' and r['old53_files']==original
        and r['new_source_bindings']=={str(p['root']/n):s for n,s in new.items()}
        and r['commit_authorized_only_after_normal_publication_checks'] is True
        and r['source_bytes_changed'] is False and r['new_scientific_calls']==0,'Prestage scope differs')
    return expected,bind((cp,rp))

def alias_log(path,target):
    """Exact closed bytes as .txt, with provenance; no JSON-wrapped code or stripping."""
    raw=Path(path).read_bytes();raw.decode('utf-8');require(b'\0' not in raw,'Not a plain UTF8 log')
    target=Path(target);target.parent.mkdir(parents=True,exist_ok=True)
    with target.open('xb') as f:f.write(raw)
    require(sha(path)==sha(target),'Log alias changed bytes')
    return dict(original_path=str(path),preserved_path=str(target),sha256=sha(path),bytes=len(raw),byte_identical=True)

def publish(request_path):
    x=inspect(request_path);p=x['p'];w=p['work'];require(not(w/'failure.json').exists() and not(w/'publish_attempt.json').exists(),
        'Failed/prior publication attempt requires independent diagnosis')
    done=read(w/'prepare_completion.json');reg=read(w/'execution_registration.json');verify(done['outputs'])
    require(done['status']=='H18_P600_SCOPE_RECOVERY_PREPARED_NOT_PUBLISHED'
        and done['registration_sha256']==sha(w/'execution_registration.json') and reg['source_bindings']==x['sources']
        and reg['input_bindings']==x['inputs'] and reg['original_staged_files']==x['original']
        and reg['admitted_staged_files']==x['staged'] and reg['additional_engineering_files']==x['new'],'Sealed recovery changed')
    owned=read(w/'publication_files.json');require(owned==done['files'],'Publication inventory changed')
    for name,s in owned.items():require(sha(p['root']/name)==s,'Owned file changed before stage')
    check_index(p,x['staged'],new_untracked=[]);require(git(p['root'],'write-tree')==reg['admitted_index_tree'],'Admitted index tree changed')
    save(w/'publish_attempt.json',dict(status='SINGLE_NORMAL_COMMIT_PUSH_CLAIMED',registration_sha256=sha(w/'execution_registration.json')))
    commands=[]
    def run(argv,label,timeout=900):
        log=w/(label+'.log')
        with log.open('xb') as f:r=subprocess.run(argv,cwd=p['root'],env=env(),stdout=f,stderr=subprocess.STDOUT,timeout=timeout)
        commands.append(dict(argv=argv,exit_code=r.returncode,log=str(log),sha256=sha(log)))
        require(r.returncode==0,'Normal check/action failed: '+label);return log
    try:
        run(['git','add','--',*sorted(owned)],'stage')
        check_index(p,owned,new_untracked=[])
        run(['git','diff','--cached','--check'],'diff_check')
        python=str(p['root']/'outputs/UNIFIED-METRICS-20261002/environment/bin/python')
        run([python,'-B',str(p['root']/'tools/verify_repository.py')],'repository')
        log=run([python,'-B',str(p['root']/'tools/run_cpu_checks.py')],'CPU_checks')
        require('264 passed' in log.read_text(),'Expected original264 repository checks')
        verify(x['inputs']);verify(x['sources']);check_index(p,owned,new_untracked=[])
        run(['git','commit','-m','Publish H18 and P600 evidence with qualified engineering provenance'],'commit')
        commit=git(p['root'],'rev-parse','HEAD')
        require(git(p['root'],'rev-parse','HEAD^')==PARENT and set(git(p['root'],'diff-tree','--no-commit-id','--name-only','-r',commit).splitlines())==set(owned),
            'Normal commit scope differs')
        run(['git','push','origin','main'],'push')
        remote=git(p['root'],'ls-remote','origin','refs/heads/main').split()[0];require(remote==commit,'Pushed remote differs')
        result=dict(status='H18_DESCRIPTIVE_P600_SCOPE_RECOVERY_PUBLISHED',commit=commit,remote_verified_sha=remote,files=owned,
            commands=commands,original_failure_preserved=True,original_scientific_result_files_unchanged=True,
            repository_CPU_checks=264,actual_qualification_counts={'timing':39,'H_P':13},new_scientific_calls=0,
            cross_H_P_statistics_published=False,MAIN_included=False,H_success_evaluated=False)
        save(w/'completion.json',result);return dict(status=result['status'],commit=commit,remote_verified_sha=remote)
    except BaseException:
        save(w/'failure.json',dict(status='FAILED_PRESERVE_NO_RETRY',traceback=traceback.format_exc(),commands=commands));raise
