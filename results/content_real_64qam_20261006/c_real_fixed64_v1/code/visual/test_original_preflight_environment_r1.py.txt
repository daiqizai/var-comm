"""Pure real-source side effects, strict phase gates, and exact owner derivatives."""
import ast
from contextlib import contextmanager
import copy
import importlib.util
import os
from pathlib import Path
import sys
import tempfile
import types
import unittest
from unittest import mock
import original_preflight_environment_r1 as h

HERE=Path(__file__).absolute().parent
BASE=HERE.parent
PREFLIGHT=Path(os.environ.get('ORIGINAL_PREFLIGHT_TEST_SOURCE',BASE.parent/'step1_reference/run_preflight.py'))
C_OWNER=Path(os.environ.get('ORIGINAL_C_VISUAL_OWNER_TEST_SOURCE',BASE/'c_real_rx_prep/c_real_visual_execution_r1.py'))
MAIN_OWNER=Path(os.environ.get('ORIGINAL_MAIN_VISUAL_OWNER_TEST_SOURCE',BASE/'main1000_prep/main_actual_visual_owner_r1.py'))

@contextmanager
def fixture():
    with tempfile.TemporaryDirectory()as td:
        root=Path(td);path=root/h.PREFLIGHT;path.parent.mkdir(parents=True);path.write_bytes(PREFLIGHT.read_bytes())
        bound={str(path):h.PREFLIGHT_SHA,str(Path(h.__file__).absolute()):h.sha(h.__file__)}
        env={k:'6'for k in h.THREAD_KEYS}
        yield root,path,bound,env

def loaded(path):
    spec=importlib.util.spec_from_file_location('run_preflight',path)
    return importlib.util.module_from_spec(spec)

class Environment(unittest.TestCase):
    def test_frozen_actual_source_has_exact_two_thread_assignments(self):
        self.assertEqual(h.sha(PREFLIGHT),h.PREFLIGHT_SHA)
        tree=ast.parse(PREFLIGHT.read_bytes());assignments={}
        for n in tree.body:
            if isinstance(n,ast.Assign):
                for target in n.targets:
                    if isinstance(target,ast.Subscript) and ast.unparse(target.value)=='os.environ' and isinstance(target.slice,ast.Constant):
                        if target.slice.value in h.THREAD_KEYS:assignments[target.slice.value]=ast.literal_eval(n.value)
        self.assertEqual(assignments,{'OMP_NUM_THREADS':'6','OPENBLAS_NUM_THREADS':'2'})

    def test_original_preflight_controlled_import_changes_only_registered_thread_value(self):
        with fixture()as(root,path,bound,env):
            token=types.ModuleType('token_efficiency');chain=types.ModuleType('token_efficiency.delivery_chain');token.delivery_chain=chain
            latent=types.ModuleType('latent_enhancement');runtime=types.ModuleType('latent_enhancement.runtime')
            runtime.write_json=mock.Mock(side_effect=AssertionError('No job or file output may run'))
            latent.runtime=runtime;m=loaded(path);original_path=list(sys.path)
            injected={'run_preflight':m,'token_efficiency':token,'token_efficiency.delivery_chain':chain,
                'latent_enhancement':latent,'latent_enhancement.runtime':runtime}
            try:
                with mock.patch.dict(os.environ,env),mock.patch.dict(sys.modules,injected):
                    m.__spec__.loader.exec_module(m)
                    self.assertTrue(h.check(bound,root))
                    self.assertEqual(os.environ['OPENBLAS_NUM_THREADS'],'2')
                    self.assertTrue(all(os.environ[k]=='6'for k in h.THREAD_KEYS if k!='OPENBLAS_NUM_THREADS'))
                    self.assertEqual(chain.CHAIN,root/'outputs/RX-POSTERIOR-STEP1-20260929/supervisor')
                    runtime.write_json.assert_not_called()
            finally:sys.path[:]=original_path

    def test_before_and_after_import_are_distinct_exact_states(self):
        with fixture()as(root,path,bound,env):
            self.assertTrue(h.check(bound,root,environment=env,modules={}))
            with self.assertRaisesRegex(RuntimeError,'thread environment'):h.check(bound,root,environment=env,modules={'run_preflight':loaded(path)})
            env['OPENBLAS_NUM_THREADS']='2'
            self.assertTrue(h.check(bound,root,environment=env,modules={'run_preflight':loaded(path)}))
            with self.assertRaisesRegex(RuntimeError,'thread environment'):h.check(bound,root,environment=env,modules={})

    def test_every_other_thread_value_stays_strict_in_both_phases(self):
        for after in (False,True):
            with fixture()as(root,path,bound,env):
                modules={'run_preflight':loaded(path)}if after else {}
                if after:env['OPENBLAS_NUM_THREADS']='2'
                for key in h.THREAD_KEYS:
                    bad=dict(env);bad[key]='1'
                    with self.subTest(after=after,key=key),self.assertRaisesRegex(RuntimeError,'thread environment'):
                        h.check(bound,root,environment=bad,modules=modules)

    def test_unknown_module_path_spec_or_missing_module_refused(self):
        for case in ('type','path','spec','name','origin','none'):
            with fixture()as(root,path,bound,env):
                env['OPENBLAS_NUM_THREADS']='2';m=loaded(path)
                if case=='type':m=types.SimpleNamespace(**vars(m))
                if case=='path':m.__file__=str(root/'elsewhere.py')
                if case=='spec':m.__spec__=None
                if case=='name':m.__spec__.name='alias'
                if case=='origin':m.__spec__.origin=str(root/'elsewhere.py')
                if case=='none':m=None
                with self.subTest(case=case),self.assertRaisesRegex(RuntimeError,'module/path/spec'):
                    h.check(bound,root,environment=env,modules={'run_preflight':m})

    def test_changed_or_unbound_helper_and_preflight_refused(self):
        for case in ('helper','preflight','bytes'):
            with fixture()as(root,path,bound,env):
                if case=='helper':bound.pop(str(Path(h.__file__).absolute()))
                if case=='preflight':bound[str(path)]='0'*64
                if case=='bytes':path.write_bytes(path.read_bytes()+b'\n')
                with self.subTest(case=case),self.assertRaises(RuntimeError):h.check(bound,root,environment=env,modules={})

    def test_checker_does_not_mutate_environment_or_load_models(self):
        with fixture()as(root,path,bound,env):
            before=copy.deepcopy(env);modules={};h.check(bound,root,environment=env,modules=modules)
            self.assertEqual(env,before);self.assertEqual(modules,{})
        tree=ast.parse(Path(h.__file__).read_bytes())
        names={n.names[0].name for n in ast.walk(tree)if isinstance(n,ast.Import)}
        self.assertTrue({'torch','numpy','torchvision','subprocess','socket'}.isdisjoint(names))

    def test_both_owner_derivatives_change_only_the_one_guard_function(self):
        for kind,path,function in [('C',C_OWNER,'check_live'),('MAIN',MAIN_OWNER,'check_live_admission')]:
            original=path.read_bytes();new=h.derive_owner(original,kind)
            self.assertEqual(new.replace(h.IMPORT,b'',1).replace(h.NEW,h.OLD,1),original)
            funcs=lambda b:{n.name:ast.dump(n,include_attributes=False)for n in ast.parse(b).body if isinstance(n,(ast.FunctionDef,ast.ClassDef))}
            before,after=funcs(original),funcs(new)
            self.assertEqual(set(before),set(after));self.assertEqual([n for n in before if before[n]!=after[n]],[function])
            self.assertEqual(before['environment'],after['environment'])
            body=ast.unparse(next(n for n in ast.parse(new).body if isinstance(n,ast.FunctionDef)and n.name==function))
            self.assertIn("os.environ.get('CUDA_VISIBLE_DEVICES') == '0'",body)
            self.assertIn('os.sched_getaffinity(0)',body);self.assertIn('os.getpriority(os.PRIO_PROCESS, 0) == 15',body)

    def test_derived_owner_bytes_match_the_exact_recorded_derivation(self):
        expected={'C':'e73c4a5a67531d931209349ea65c70da96cae430a9775d98bf7767cf1c55f56e',
                  'MAIN':'b8582de7bf1fb1bfc7da7bfa6d5fb2e70078d9ad1bf7c3ac325af9b156e998df'}
        for kind,path in [('C',C_OWNER),('MAIN',MAIN_OWNER)]:
            self.assertEqual(h.hashlib.sha256(h.derive_owner(path.read_bytes(),kind)).hexdigest(),expected[kind])

    def test_unknown_owner_or_any_source_change_refused(self):
        with self.assertRaisesRegex(RuntimeError,'frozen owner'):h.derive_owner(C_OWNER.read_bytes()+b'\n','C')
        with self.assertRaisesRegex(RuntimeError,'frozen owner'):h.derive_owner(C_OWNER.read_bytes(),'unregistered')

if __name__=='__main__':unittest.main()
