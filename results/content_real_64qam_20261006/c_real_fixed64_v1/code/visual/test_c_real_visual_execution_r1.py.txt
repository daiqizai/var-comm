class Child:
    def __init__(self,code=0,polls=None):self.pid=1234;self.code=code;self.waited=0;self.polls=list(polls or [code])
    def poll(self):return self.polls.pop(0) if self.polls else self.code
    def wait(self):self.waited+=1;return self.code

class Execution(unittest.TestCase):

    def patches(self,child,*,fail_capture=False):
        stack=ExitStack();self.addCleanup(stack.close)
        self.held=[];self.events=[]
        @contextmanager
        def lock(p):
            self.held.append(p);self.events.append(('acquire',p))
            try:yield
            finally:self.events.append(('release',p));self.held.remove(p)
        oldwait=child.wait
        def wait():self.assertEqual(len(self.held),2);self.events.append(('wait',None));return oldwait()
        child.wait=wait
        stack.enter_context(patch.object(e,'load',return_value=self.ctx));stack.enter_context(patch.object(e,'lock',side_effect=lock))
        stack.enter_context(patch.object(e,'identity',return_value=self.me));stack.enter_context(patch.object(e,'absent',return_value=True))
        stack.enter_context(patch.object(e.os,'sched_getaffinity',return_value=set(range(32)),create=True))
        stack.enter_context(patch.object(e,'gpu_idle',return_value=dict(status='GPU_IDLE_CONFIRMED',device=0)))
        stack.enter_context(patch.object(e,'environment',return_value={'CUDA_VISIBLE_DEVICES':'0'}));stack.enter_context(patch.object(e.time,'sleep'))
        stack.enter_context(patch.object(e.subprocess,'Popen',return_value=child))
        stack.enter_context(patch.object(e,'capture',side_effect=RuntimeError('identity capture failure') if fail_capture else None,return_value=self.worker))
        return stack

    def test_exact_budget_no_new_reserve_or_failed_call(self):
        self.assertEqual(e.validate_budget(self.before),self.before)
        for k,v in [('charged',21496),('unresolved',1)]:
            x=deepcopy(self.before);x[k]=v
            with self.assertRaisesRegex(ValueError,'21495'):e.validate_budget(x)
        x=deepcopy(self.before);x['phase_charged']['diagnostic_reserve']=1
        with self.assertRaises(ValueError):e.validate_budget(x)

    def test_single_GPU_environment_preserves_venv_and_source_directory(self):
        e.validate_resources(self.cfg);self.assertEqual(e.command(self.cfg)[0],self.cfg['python'])
        env=e.environment(self.ctx);self.assertEqual(env['CUDA_VISIBLE_DEVICES'],'0');self.assertTrue(all(env[k]=='6' for k in e.THREAD_KEYS))
        self.assertIn('E',env['PYTHONPATH'].split(os.pathsep));self.assertIn('new-runtime',env['PYTHONPATH'])
        x=deepcopy(self.cfg);x['resources']['threads']=2
        with self.assertRaises(ValueError):e.validate_resources(x)

    def test_gpu_idle_rejects_foreign_and_multiple_devices(self):
        runner=Mock(side_effect=[SimpleNamespace(stdout='0, UUID\n'),SimpleNamespace(stdout='')]);self.assertEqual(e.gpu_idle(self.cfg['nvidia_smi'],runner)['device'],0)
        runner=Mock(side_effect=[SimpleNamespace(stdout='0, UUID\n'),SimpleNamespace(stdout='999\n')])
        with self.assertRaisesRegex(ValueError,'Foreign'):e.gpu_idle(self.cfg['nvidia_smi'],runner)
        with self.assertRaises(ValueError):e.gpu_idle(self.cfg['nvidia_smi'],Mock(return_value=SimpleNamespace(stdout='0, A\n1, B')))

    def test_finished_worker_waits_and_closes_log_before_receipt(self):
        child=Child();h=(self.ex/'worker.log').open('xb');h.write(b'complete\n');rec=dict(child=child,handle=h,identity=self.worker)
        x=e.finish(rec,self.ex);self.assertTrue(h.closed);self.assertEqual(child.waited,1);self.assertTrue(x['reaped'])
        self.assertEqual(x['log_sha256'],e.sha(self.ex/'worker.log'));self.assertEqual(e.finish(rec,self.ex),x);self.assertEqual(child.waited,1)

    def test_normal_poll_zombie_window_never_live_checks_unreaped_child(self):
        child=Child(polls=[None,0]);self.patches(child);self.science()
        # run owns output creation in production; science fixture is returned after the spawn.
        original_exists=Path.exists
        def exists(p):return False if p==Path(self.cfg['out']) else original_exists(p)
        with patch.object(Path,'exists',exists):result=e.run(self.cfg['_path'])
        self.assertEqual(result['status'],e.DONE);self.assertEqual(child.waited,1)
        self.assertEqual(self.events[-2:],[('release',e.lock_paths(self.ctx)[1]),('release',e.lock_paths(self.ctx)[0])])

    def test_identity_capture_failure_preserves_PID_and_drains_under_both_locks(self):
        child=Child();self.patches(child,fail_capture=True)
        with self.assertRaisesRegex(RuntimeError,'identity capture'):e.run(self.cfg['_path'])
        self.assertEqual(e.read(self.ex/'spawn_pending_identity.json')['pid'],1234)
        self.assertEqual(e.read(self.ex/'worker_exit.json')['exit_code'],0);self.assertEqual(child.waited,1)
        self.assertTrue((self.ex/'failure.json').exists());self.assertFalse((self.ex/'completion.json').exists());self.assertFalse(self.held)

    def test_nonzero_exit_retains_failure_no_normal_science_claim(self):
        child=Child(code=1);self.patches(child)
        with self.assertRaises((ValueError,FileNotFoundError)):e.run(self.cfg['_path'])
        self.assertEqual(child.waited,1);self.assertEqual(e.read(self.ex/'worker_exit.json')['exit_code'],1)
        self.assertTrue((self.ex/'failure.json').exists());self.assertFalse((self.ex/'completion.json').exists())

    def test_STOP_drains_without_kill_and_retains_lock(self):
        child=Child(polls=[None,0]);self.patches(child)
        calls=[0]
        def check(cfg,**kw):
            calls[0]+=1
            if calls[0]>=3:raise ValueError('registered STOP')
        with patch.object(e,'check_stop',side_effect=check):
            with self.assertRaises(ValueError):e.run(self.cfg['_path'])
        self.assertEqual(child.waited,1);self.assertTrue((Path(self.cfg['out'])/'STOP').exists());self.assertFalse((self.ex/'completion.json').exists())

    def test_live_worker_requires_exact_parent_resources_and_two_locks(self):
        proof=dict(worker_identity=self.worker,owner_identity=self.me)
        with patch.object(e,'identity',side_effect=[self.worker,self.me]),patch.object(e,'locked',return_value=True),\
            patch.object(e.os,'sched_getaffinity',return_value=set(range(4,10)),create=True),patch.object(e.os,'getpriority',return_value=15,create=True),\
            patch.object(e.os,'PRIO_PROCESS',0,create=True),patch.dict(os.environ,dict(CUDA_VISIBLE_DEVICES='0',**{k:'6' for k in e.THREAD_KEYS})):
            e.check_live(self.ctx,proof)
        with patch.object(e,'identity',side_effect=[self.worker,self.me]),patch.object(e,'locked',return_value=False):
            with self.assertRaisesRegex(ValueError,'ownership'):e.check_live(self.ctx,proof)

    def test_no_retry_and_history_does_not_invent_current_deadline(self):
        e.save(self.ex/'run_attempt.json',{'prior':True})
