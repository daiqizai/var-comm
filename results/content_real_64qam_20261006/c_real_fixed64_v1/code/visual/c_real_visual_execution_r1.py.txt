def require(ok,message):

def sha(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as f:
        for block in iter(lambda:f.read(1048576),b''):h.update(block)
    return h.hexdigest()

@contextmanager
def lock(path):
    import fcntl
    with Path(path).open('a+') as f:
        fcntl.flock(f,fcntl.LOCK_EX|fcntl.LOCK_NB)
        try:yield
        finally:fcntl.flock(f,fcntl.LOCK_UN)
def locked(path):
    import fcntl
    with Path(path).open('r+') as f:
        try:fcntl.flock(f,fcntl.LOCK_EX|fcntl.LOCK_NB)
        except BlockingIOError:return True
        fcntl.flock(f,fcntl.LOCK_UN);return False

        result=dict(status=DONE,registration_sha256=ctx['regsha'],config_sha256=sha(path),owner_identity=me,worker_identity=rec['identity'],

            outputs=combine(done['outputs'],bind([Path(cfg['out'])/'completion.json']+[e/n for n in
                ('owner_identity.json','gpu_admission.json','spawn_pending_identity.json','worker_launch.json','worker_admission.json','worker_exit.json','worker.log')])),
            images_scored=True,new_packet_decodes=0,new_noise=0,policy_selection=False,automatic_successor=False)
        require(result['budget_before']==result['budget_after'],'C ledger changed at normal closure');save(e/'completion.json',result);return result
    except BaseException:
        error=traceback.format_exc()
        if rec is not None:
            marker(cfg,'Owner failure; retain child evidence and drain safely')
            finish(rec,e)  # Both shared locks remain held until wait and closed-log sealing finish.
        save(e/'failure.json',dict(status='FAILED_PRESERVE_NO_RETRY',traceback=error,owner_identity=me,
            worker_pid=None if rec is None else rec['child'].pid,new_packet_decodes=0));raise
    finally:stack.close()

def closed_batch(spec):
    ctx=load(spec['config']);cfg=ctx['cfg'];e=Path(cfg['execution_dir']);check_stop(cfg,historical=True)
    require(spec['completion']==str(e/'completion.json') and not(e/'failure.json').exists() and not(Path(cfg['out'])/'failure.json').exists(),'Normal visual owner required')
    done=read(spec['completion']);owner=read(e/'owner_identity.json');launch=read(e/'worker_launch.json');x=read(e/'worker_exit.json')
    require(done['status']==DONE and done['registration_sha256']==ctx['regsha'] and done['config_sha256']==sha(spec['config'])
        and done['owner_identity']==owner['identity'] and done['worker_identity']==launch['identity']==x['identity']
        and done['worker_reaped'] is x['reaped'] is True and done['exit_code']==x['exit_code']==0
        and x['log_sha256']==sha(e/'worker.log') and absent(owner['identity']) and absent(x['identity']),'Actual normal owner/worker exit/log required')
    science=validate_done(ctx,dict(exit=x));verify(done['outputs'])
    require(done['budget_after']==done['budget_before']==ctx['before'] and done['source_bindings']==ctx['reg']['source_bindings']
        and done['input_bindings']==ctx['reg']['input_bindings'] and all(done['outputs'].get(p)==h for p,h in science['outputs'].items()),'Normal output map/budget differs')
    return dict(ctx=ctx,done=done,science=science,bindings=combine(ctx['bound'],done['outputs'],bind([spec['config'],spec['completion'],cfg['registration']])))
