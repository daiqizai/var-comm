class ClassifierAdmission(unittest.TestCase):
    def bindings(self):
        lib=SimpleNamespace(__file__=str(Path('/original_validation.py').absolute()),WEIGHTS_SHA256=v.WEIGHTS)
        pins={lib.__file__:'a'*64,'weights':v.WEIGHTS,'protocol':v.PROTOCOL,'policy':v.POLICY,'sources':v.SOURCES}
        return lib,pins
    def test_explicit_C64_population_not_fake_development(self):
        lib,pins=self.bindings()
        with patch.object(v,'sha',side_effect=lambda p:pins[str(p)]):
            got=v.admission(lib,'weights','protocol','policy','sources',pins)
        self.assertEqual(got['population'],'original_calibration_internal64');self.assertFalse(got['used_for_selection'])
    def test_missing_changed_weight_or_policy_or_population_refuses_before_model(self):
        lib,pins=self.bindings()
        for key in pins:
            changed=dict(pins);changed[key]='e'*64
            with patch.object(v,'sha',side_effect=lambda p:pins[str(p)]):
                with self.assertRaisesRegex(ValueError,'binding'):v.admission(lib,'weights','protocol','policy','sources',changed)
    def test_original_predict_and_score_methods_receive_same_object(self):
        seen=[];obj=object.__new__(v.C64ConvNeXt)
        obj.validation=SimpleNamespace(ConvNeXtValidation=SimpleNamespace(predict=lambda self,image:seen.append((self,image)) or 5,
            score=lambda self,image,**kwargs:seen.append((self,image,kwargs)) or {'original':True}))
        image=np.zeros((3,256,256),np.float32)
        self.assertEqual(obj.predict(image),5);self.assertEqual(obj.score(image,true_label=3,source_prediction=4),{'original':True})
        self.assertIs(seen[0][0],obj);self.assertIs(seen[0][1],image)
        self.assertEqual(seen[1][2],dict(true_label=3,source_prediction=4))
    def test_actual_original_score_uses_exact_self_fields_and_returns_false_selection(self):
        path=Path(os.environ.get('C_RX_ORIGINAL_VALIDATION_SOURCE',str(Path(__file__).resolve().parents[2]/'prior_aware_uep_20261004/runtime/validation_metrics.py')))
        spec=importlib.util.spec_from_file_location('actual_original_C_validation_fixture',path);old=importlib.util.module_from_spec(spec);spec.loader.exec_module(old)
        obj=object.__new__(v.C64ConvNeXt);obj.validation=old;obj.names=tuple(f'label{i}' for i in range(1000))
        obj.identity={'weights_sha256':v.WEIGHTS};obj.predict=lambda rgb:7

        self.assertIs(got['convnext_source_correct_to_wrong'],True);self.assertIs(got['convnext_used_for_selection'],False)
        self.assertEqual(got['convnext_weight_sha256'],v.WEIGHTS);self.assertEqual(got['convnext_prediction_name'],'label7')
