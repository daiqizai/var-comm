from contextlib import nullcontext
from copy import deepcopy
from pathlib import Path
from types import SimpleNamespace as NS
import tempfile
import unittest
from unittest.mock import patch
import numpy as np
import c_online_tx_driver_r1 as d
from test_c_online_tx_components_r1 import fixture,Calls,original_formula

def all_cases():
    result=[]
    for i in range(16):
        for s in d.core.SNRS:
            for m in d.core.METHODS:
                rows=[]
                for phase,n in (('warmup',1),('measured',3)):
                    for j in range(n):
                        row=dict(source_index=i,source_id='s'+str(i),average_snr_db=s,method=m,fingerprint=str((i,s,m)),
                            repetition_kind=phase,repetition=j,choice_parity=True,new_packet_decodes=0,new_noise=0,cross_frame_cache=False,
                            seconds={'source_encoding':1000. if phase=='warmup'else float(i+j),'TX_total':2000. if phase=='warmup'else float(i+j+2)})
                        if m=='QCSI-C8':row['seconds']['candidate_DINO']=4.
                        rows.append(row)
                result.append(dict(rows=rows))
    return result

class Driver(unittest.TestCase):
    def test_fixed_128_summary_excludes_warmup_and_keeps_inclusive_total(self):
        got=d.summarize(all_cases());self.assertEqual(got['case_count'],128)
        r=next(r for r in got['components']if r['average_snr_db']==4 and r['method']=='QCSI-U'and r['component']=='source_encoding')
        self.assertEqual(r['mean_seconds'],8.5);self.assertFalse(r['inclusive']);self.assertFalse(got['end_to_end_latency_measured'])

    def test_missing_or_duplicate_grid_refused(self):
        for rows in (all_cases()[:-1],all_cases()[:-1]+[all_cases()[0]]):
            with self.assertRaises(ValueError):d.summarize(rows)

    def test_fresh_repetition_mismatch_refused(self):
        rows=all_cases();rows[0]['rows'][-1]['fingerprint']='different'
        with self.assertRaisesRegex(ValueError,'Repeated'):d.summarize(rows)

    def test_U_has_no_fabricated_candidate_timing(self):
        got=d.summarize(all_cases())
        self.assertFalse(any(r['method']=='QCSI-U'and r['component']=='candidate_DINO' for r in got['components']))
        self.assertTrue(any(r['method']=='QCSI-C8'and r['component']=='candidate_DINO' for r in got['components']))

    def test_actual_model_table_npz_contract_four_original_snrs(self):
        f,_,meta,arrays=fixture()
        with tempfile.TemporaryDirectory()as td:
            pins={}
            for s in d.core.SNRS:
                folder=Path(td)/('snr'+str(s));folder.mkdir();d.save(folder/'selected_profiles.json',dict(meta,average_snr_db=s))
                np.savez(folder/'selected_coefficients.npz',**arrays);pins.update(d.bind(folder.iterdir()))
            models,inputs=d.model_tables(f,pins);self.assertEqual(set(models),set(d.core.SNRS));self.assertEqual(inputs,pins)

    def test_changed_coefficient_archive_refused(self):
        f,_,meta,arrays=fixture()
        with tempfile.TemporaryDirectory()as td:
            folder=Path(td)/'snr4';folder.mkdir();d.save(folder/'selected_profiles.json',meta);np.savez(folder/'selected_coefficients.npz',**arrays)
            pins=d.bind(folder.iterdir());pins[str(folder/'selected_coefficients.npz')]='0'*64
            with self.assertRaisesRegex(ValueError,'changed'):d.model_tables(f,pins)

    def test_true_core_measure_source_eight_cases_32_fresh_encoders(self):
        f,m,meta,arrays=fixture();calls=Calls();models_by_s={s:d.core.model_table(f,s,dict(meta,average_snr_db=s),arrays) for s in d.core.SNRS}
        torch=NS(inference_mode=nullcontext,autocast=lambda **kw:nullcontext(),cuda=NS(synchronize=lambda:None))
        native=NS(torch=torch);models=dict(native=native,metric=NS(no_network=nullcontext),quality=NS(q=object()),
            encoder=NS(native_encoder=lambda n,p:calls.encoder(p)),entropy=NS(entropy_orders=lambda n,q,t,m,g:calls.entropy(t,m)),
            feature=calls.feature,cosine=calls.cosine,flat=NS(score_block=original_formula))
        ctx=dict(frozen=f,tables=models_by_s,prior={'cpu_supervisor':NS(d=NS(c=NS(source_payloads=calls.pack)))})
        item=dict(pixels=np.zeros((3,256,256),np.uint8),source_index=0,tokens_sha256=d.core.token_sha(calls.tokens))
        with patch.object(d.core,'original_render',side_effect=lambda n,q,t,s:calls.render(t,s)):
            rows=d.measure_source(ctx,item,models,lambda:None)
        self.assertEqual(len(rows),8);self.assertEqual(calls.encoder_calls,32);self.assertEqual(calls.renders,32)
        self.assertEqual([r['rows'][0]['method']for r in rows],list(d.core.METHODS)*4)

    def test_current_budget_and_command_use_actual_prior_context(self):
        prior={'driver':NS(budget=lambda ctx:ctx['real_budget']),'real_budget':{'charged':21495}}
        self.assertEqual(d.budget({'prior':prior}),prior['real_budget'])
        cfg=dict(python='oldUM',driver_module='/new/driver.py',_path='/new/config.json')
        self.assertEqual(d.command(cfg),['oldUM','-B','/new/driver.py','--config','/new/config.json'])

    def test_metric_identity_fields_keep_L14_separate_from_legacyS14(self):
        # Actual loader API uses the original L14-only requested tuple, without a suite fallback.
        import ast
        tree=ast.parse(Path(d.__file__).read_text());fn=next(n for n in tree.body if isinstance(n,ast.FunctionDef)and n.name=='build_models')
        calls=[n for n in ast.walk(fn)if isinstance(n,ast.Call)and isinstance(n.func,ast.Attribute)and n.func.attr=='MetricEvaluator']
        self.assertEqual(len(calls),1);self.assertEqual(ast.literal_eval(next(k.value for k in calls[0].keywords if k.arg=='requested')),('dinov2_vitl14',))

if __name__=='__main__':unittest.main()
