def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def require(ok,msg):
    if not ok:raise ValueError(msg)
def read(p):return json.loads(Path(p).read_text(encoding='utf-8-sig'))

    value=read(p);require(value['metric_identity']==science['metric_evaluator_identity'],'Actual C metric identity differs')
    evaluator=value['metadata']['original_suite']['metric_evaluator']
    require(evaluator['numerical_backend']['device']=='cuda:0' and evaluator['metrics']['dinov2_vitl14']['status']=='READY',
        'Original admitted DINO-L on GPU0 required before any new model')
    return p

def qualify(ctx):
    e,old,loc=ctx['e'],ctx['old'],ctx['loc'];qdir=Path(loc['qualification']);op=Path(loc['operations'])
    require(not qdir.exists() and not op.exists(),'Independent fresh timing qualification/operations required')
    qdir.mkdir(parents=True);op.mkdir(parents=True);cmd=[ctx['visual']['python'],'-B','-m','unittest',*e.TEST_MODULES,'-v']
    env=dict(os.environ,PYTHONPATH=loc['runtime'],PYTHONDONTWRITEBYTECODE='1',CUDA_VISIBLE_DEVICES='',**{k:'2' for k in old.THREAD_KEYS})
    e.save(qdir/'attempt.json',dict(argv=cmd,source_bindings=ctx['sources'],new_GPU_calls=0,new_packet_decodes=0));rec=None
    try:
        with (qdir/'tests.log').open('xb')as log:
            child=subprocess.Popen(cmd,cwd=loc['root'],env=env,stdout=log,stderr=subprocess.STDOUT,stdin=subprocess.DEVNULL,start_new_session=True,
                preexec_fn=lambda:old.resources([0,1]));rec=child
            e.save(qdir/'pending_pid.json',dict(pid=child.pid,argv=cmd));identity=old.capture(child,cmd);e.save(qdir/'launch.json',dict(identity=identity,argv=cmd))
            code=child.wait();log.flush();os.fsync(log.fileno())
        e.save(qdir/'exit.json',dict(identity=identity,exit_code=code,process_waited=True,log_sha256=e.sha(qdir/'tests.log')))
        require(code==0 and old.absent(identity),'Timing pure tests failed; no retry')
        q=dict(status=e.QPASS,tests_run=e.TEST_COUNT,test_modules=e.TEST_MODULES,python=ctx['visual']['python'],exit_code=0,process_waited=True,
            new_GPU_calls=0,new_packet_decodes=0,source_bindings=ctx['sources'],input_bindings=ctx['inputs'],log=str(qdir/'tests.log'),
            outputs=e.bind(qdir/n for n in ('attempt.json','launch.json','exit.json','tests.log')))
        e.verify(ctx['sources']);e.verify(ctx['inputs']);e.save(qdir/'completion.json',q)
        request=make_request(ctx);qcfg=dict(request['config'],python=ctx['visual']['python'])
        e.qualification(str(qdir/'completion.json'),qcfg,e.combine(request['source_bindings'],request['input_bindings']))
        e.save(op/'request.json',request);done=dict(status='C_ONLINE_TX_QUALIFIED_REQUEST_NOT_REGISTERED_R1',outputs=e.bind([qdir/'completion.json',op/'request.json']),new_GPU_calls=0)
        e.save(op/'qualification_completion.json',done);return done
    except BaseException:
        if rec is not None and rec.poll()is None:rec.wait()
        e.save(qdir/'failure.json',dict(traceback=traceback.format_exc(),GPU_started=False,automatic_retry=False));raise

def operation(stage,loc):
    ctx=context(loc);e=ctx['e'];op=Path(loc['operations'])
    if stage=='qualify':return qualify(ctx)
    done=read(op/'qualification_completion.json');require(done['status']=='C_ONLINE_TX_QUALIFIED_REQUEST_NOT_REGISTERED_R1'
        and not(Path(loc['qualification'])/'failure.json').exists(),'Successful actual timing pure qualification required');e.verify(done['outputs'])
    e.save(op/(stage+'_attempt.json'),dict(automatic_retry=False))
    try:
        if stage=='run':signal.signal(signal.SIGTERM,e.stop);signal.signal(signal.SIGINT,e.stop)
        result=e.register(str(op/'request.json'))if stage=='register'else e.run(str(Path(loc['execution'])/'config.json'))
        e.save(op/(stage+'_completion.json'),dict(status=result['status'],outputs=e.bind([Path(loc['execution'])/('registration_completion.json'if stage=='register'else'completion.json')])));return result
    except BaseException:e.save(op/(stage+'_failure.json'),dict(traceback=traceback.format_exc(),GPU_started=None if stage=='run'else False,inspect_preserved_PID=stage=='run'));raise
