def sha(p):
    h=hashlib.sha256()
    with Path(p).open('rb')as f:
        for b in iter(lambda:f.read(1048576),b''):h.update(b)
    return h.hexdigest()
def read(p):return json.loads(Path(p).read_text(encoding='utf-8-sig'))
def save(p,value):
    p=Path(p);p.parent.mkdir(parents=True,exist_ok=True)
    with p.open('x',encoding='utf-8',newline='\n')as f:json.dump(value,f,indent=2,allow_nan=False);f.write('\n');f.flush();os.fsync(f.fileno())
def bind(paths):return {str(Path(p).absolute()):sha(p) for p in paths}
def combine(*maps):

def summarize(cases):

        rows=case['rows'];require(len(rows)==4 and [(r['repetition_kind'],r['repetition']) for r in rows]==[('warmup',0),('measured',0),('measured',1),('measured',2)],'One warmup and three measured rows required')
        base=rows[0];key=base['source_index'],base['average_snr_db'],base['method'];require(key not in seen,'Duplicate case');seen.add(key)
        require(all((r['source_index'],r['average_snr_db'],r['method'])==key and r['fingerprint']==base['fingerprint']
            and r['choice_parity'] is True and r['new_packet_decodes']==r['new_noise']==0 and r['cross_frame_cache'] is False for r in rows),'Repeated fresh computation changed')
        measured=rows[1:];names={k for r in measured for k in r['seconds']}
        for name in names:groups.setdefault((key[1],key[2],name),{})[key[0]]=statistics.mean(r['seconds'].get(name,0.) for r in measured)
    require(seen=={(i,s,m) for i in range(16) for s in core.SNRS for m in core.METHODS},'Fixed128 grid incomplete')
    out=[]

            median_source_seconds=statistics.median(source_means),source_means=source_means,source_count=16,
            measured_repetitions_per_source=3,warmup_excluded=True,inclusive=name=='TX_total'))
    return dict(status='C_FIRST16_ONLINE_TX_COMPONENT_SUMMARY',case_count=128,source_count=16,total_fresh_encoder_calls=512,

def run(path):

    out.mkdir();save(out/'attempt.json',dict(registration_sha256=ctx['regsha'],new_packet_decodes=0));start=time.monotonic()
    try:
        owner=module(cfg['owner_module'],'c_timing_live_owner',ctx['bound']);proof=owner.admit_worker(ctx);models=None
        def guard():
            require(not STOP and time.monotonic()-start<cfg['resources']['max_seconds'],'Timing signal/wall limit');owner.check_live(ctx,proof)
            if models is not None:models['native'].common.check()
        guard();models=build_models(ctx);guard();cases=[];outputs={}
        for index in range(16):
            guard();require(not models['quality'].boundary(models['native']),'Original thermal/source boundary');item=timing_source(ctx,index)
            rows=measure_source(ctx,item,models,guard);rp=out/'sources'/f'{index:04d}.json';cp=out/'source_checkpoints'/f'{index:04d}.json';save(rp,rows)
            save(cp,dict(status='C_FIRST16_ONLINE_TX_SOURCE_COMPLETE_R1',source_index=index,source_id=item['source_id'],bin=index%8,
                registration_sha256=ctx['regsha'],case_count=8,fresh_encoder_calls=32,outputs=bind([rp]),input_bindings=item['input_bindings'],
                new_packet_decodes=0,new_noise=0,policy_selection=False,new_quality_samples=0));outputs.update(bind([rp,cp]));cases.extend(rows)
        guard();models['native'].frozen();require(models['evaluator'].identity()==models['metric_identity'],'DINO-L identity changed')
        summary=summarize(cases);rp=out/'component_cases.json';sp=out/'component_summary.json';save(rp,cases);save(sp,summary);outputs.update(bind([rp,sp]))
        after=budget(ctx);require(after==ctx['before'],'C budget changed during zero-PHY timing');verify(ctx['bound']);verify(outputs)
        done=dict(status=DONE,registration_sha256=ctx['regsha'],config_sha256=sha(path),worker_identity=proof['worker_identity'],
            source_count=16,source_indices=list(range(16)),source_ids=ctx['frozen']['source_ids'][:16],case_count=128,
            fresh_encoder_calls=512,warmup_repetitions=1,measured_repetitions=3,timing_design=ctx['scope'],outputs=outputs,
            source_bindings=ctx['reg']['source_bindings'],input_bindings=ctx['reg']['input_bindings'],

            DINO_L_metadata=models['quality'].q.jsonable(models['evaluator'].metadata),budget_before=ctx['before'],budget_after=after,
            new_packet_decodes=0,new_noise=0,new_quality_samples=0,policy_selection=False,holdout_used=False,
            source_population='original_calibration_internal64_first16',automatic_successor=False,
            end_to_end_latency_measured=False,overall_C_delivery_complete=False,elapsed_seconds=time.monotonic()-start)

    except BaseException:save(out/'failure.json',dict(status='FAILED_PRESERVE_NO_RETRY',traceback=traceback.format_exc(),new_packet_decodes=0));raise
