class Child:
    def __init__(self,pid,code=0,polls=None):
        self.pid=pid;self.code=code;self.returncode=None;self.waits=0;self.polls=list(polls or [code])
    def poll(self):
        value=self.polls.pop(0) if self.polls else self.code
        if value is not None:self.returncode=value
        return value
    def wait(self):self.waits+=1;self.returncode=self.code;return self.code

    def test_exception_drains_live_child_before_releasing_owner_lock(self):
        held=[];events=[];original_finish=s.finish
        @contextmanager
        def lock(cfg):
            held.append(True);events.append('acquire')
            try:yield
            finally:held.pop();events.append('release')
        def launch(ctx,i,me):
            if i==1:raise RuntimeError('second launch failed')
            return self.fake_launch(ctx,i,me)
        def finish(ctx,r):
            self.assertTrue(held);events.append('wait');return original_finish(ctx,r)
        with ExitStack() as stack:
            stack.enter_context(patch.object(s,'load',return_value=self.ctx));stack.enter_context(patch.object(s.sys,'platform','linux'))
            stack.enter_context(patch.object(s.os,'sched_getaffinity',return_value={0,1,2,3},create=True));stack.enter_context(patch.object(s,'identity',return_value=self.me))
            stack.enter_context(patch.object(s,'owner_lock',side_effect=lock));stack.enter_context(patch.object(s.d,'ledger_snapshot',return_value=self.before))
            stack.enter_context(patch.object(s,'launch',side_effect=launch));stack.enter_context(patch.object(s,'finish',side_effect=finish))
            with self.assertRaisesRegex(RuntimeError,'second launch'):s.run(self.cfg['owner_config'])
        self.assertEqual(events,['acquire','wait','release']);self.assertTrue((self.e/'owner_failure.json').exists());self.assertFalse((self.e/'completion.json').exists())
