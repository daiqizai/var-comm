def packed(bits):return dict(bits=len(bits),hex=np.packbits(bits,bitorder='big').tobytes().hex())

class FakePHY:
    def __init__(self):self.ledger=FakeLedger();self.actual=[];self.wrong=None;self.reject=False
    def header_wave(self,pid):self.pid=pid;self.actual=[];return np.ones((68,2))
    def body_wave(self,payload,g,counter,j,session):self.actual.append(payload.copy());return np.ones((g['symbols'],2))
    def header_receive(self,y,h,v,book,event,phase):
        pid=self.pid if self.wrong is None else self.wrong
        return self.ledger.record(event,'header',dict(profile_id=pid,header_crc_accept=not self.reject,header_fields_legal=str(pid) in book,header_ok=not self.reject and str(pid) in book,decoded=packed(np.zeros(34,dtype=np.uint8))))
    def body_receive(self,y,h,v,g,counter,j,session,event,phase):
        bits=self.actual[j] if j<len(self.actual) and len(self.actual[j])==g['source_bits'] else np.zeros(g['source_bits'],dtype=np.uint8)
        return self.ledger.record(event,'body',dict(payload=packed(bits),crc_accept=True,decoded_sha256=c.array_sha(bits)))
