def pack(bits):return dict(bits=len(bits),hex=np.packbits(bits,bitorder='big').tobytes().hex())

class FakePHY:
    def __init__(self,ledger):
        self.ledger=ledger;self.last_header=0;self.tx={};self.wrong_header=None;self.reject_header=False;self.reject_body=None;self.exception=False
        ledger.register_configuration('header',{})
    def admit(self,g,new):self.ledger.register_configuration(g['phy_key'],{'layout':g['layout']})
    def header_wave(self,i):self.last_header=i;return np.ones((68,2))
    def body_wave(self,bits,g,counter,j,session):
        self.tx[g['phy_key'],counter,j,session]=bits.copy();return np.ones((g['symbols'],2))
    def header_receive(self,y,h,v,book,event,phase):
        def decode():
            pid=self.last_header if self.wrong_header is None else self.wrong_header
            return dict(profile_id=pid,header_crc_accept=not self.reject_header,header_fields_legal=str(pid) in book,
                header_ok=not self.reject_header and str(pid) in book,decoded=pack(np.zeros(34,dtype=np.uint8)))
        return self.ledger.decode_once(phase,event,'header','header',dict(y=c.array_sha(y),book=c.digest(book)),decode)
    def body_receive(self,y,h,v,g,counter,j,session,event,phase):
        def decode():
            if self.exception:raise RuntimeError('fake actual decoder exception')
            bits=self.tx.get((g['phy_key'],counter,j,session),np.ones(g['source_bits'],dtype=np.uint8))
            return dict(payload=pack(bits),crc_accept=j!=self.reject_body,decoded_sha256=c.array_sha(bits))
        return self.ledger.decode_once(phase,event,'body',g['phy_key'],dict(y=c.array_sha(y),counter=counter,j=j,session=session),decode)

    def test_actual_frozen4096_choices26layouts16profiles_no_reselection(self):

        copy=self.path/'protocol.json';copy.write_bytes((ASSETS/'C_REAL_PROTOCOL.json').read_bytes()+b' ')
        with self.assertRaisesRegex(ValueError,'changed'):c.load_frozen(copy,*[ASSETS/n for n in require_files[1:]])

    def test_seed_exact_draw_order_and_shared_full992_noise(self):
        sid=self.ctx['source_ids'][0];h,z,v=c.standard_channel(sid,7,3)
        key=json.dumps(['CONTENT-REAL-64QAM-20261006/C/channel',2026100604,sid,7,3],ensure_ascii=False,separators=(',',':')).encode()
        g=np.random.Generator(np.random.PCG64(int.from_bytes(hashlib.sha256(key).digest()[:16],'little')))
        self.assertEqual(h,complex(*g.standard_normal(2))/np.sqrt(2));np.testing.assert_array_equal(z,g.standard_normal((992,2)))
        self.assertEqual(v,10**(-.7));self.assertEqual(len({c.public_counter(i,s,r) for i in range(64) for s in c.SNRS for r in range(16)}),4096)
        for snr,trial in ((7.,3),(7,16),(1,0)):
            with self.assertRaises(ValueError):c.standard_channel(sid,snr,trial)
