class Execution(unittest.TestCase):
    @classmethod
    def setUpClass(cls):f.Entropy.setUpClass();cls.plan=f.Entropy.plan
    def test_original_visual_normal_chain_and_missing_self_registration(self):
        with tempfile.TemporaryDirectory() as td,mock.patch.object(x,'exited',return_value=True):
            spec=original_fixture(Path(td));ctx=x.original_visual_closed(spec)
            self.assertIn(str(Path(td)/'source_execution/registration.json'),ctx['bindings']);self.assertEqual(ctx['python'],'um')
            p=Path(td)/'source_execution/stages/source/workers/full1000_source/worker.log';p.write_text('changed')
            with self.assertRaisesRegex(ValueError,'normal closure'):x.original_visual_closed(spec)
    def test_original_visual_live_or_failure_rejected(self):
        with tempfile.TemporaryDirectory() as td:
            spec=original_fixture(Path(td))
            with mock.patch.object(x,'exited',return_value=False),self.assertRaises(ValueError):x.original_visual_closed(spec)
            with mock.patch.object(x,'exited',return_value=True):
                x.save(Path(td)/'source/failure.json',{'failure':True})
                with self.assertRaisesRegex(ValueError,'failure retained'):x.original_visual_closed(spec)
    def test_gpu_idle_rejects_foreign_and_multiple_devices(self):
        with tempfile.TemporaryDirectory() as td:
            p=Path(td)/'nvidia-smi';p.write_text('bound executable')
            run=mock.Mock(side_effect=[SimpleNamespace(stdout='0, GPU-exact\n'),SimpleNamespace(stdout='')])
            self.assertEqual(x.gpu_idle(str(p),run)['status'],'GPU_IDLE_CONFIRMED')
            run=mock.Mock(side_effect=[SimpleNamespace(stdout='0, GPU-exact\n'),SimpleNamespace(stdout='999')])
            with self.assertRaisesRegex(ValueError,'compute process'):x.gpu_idle(str(p),run)
            run=mock.Mock(return_value=SimpleNamespace(stdout='0, GPU-a\n1, GPU-b'))
            with self.assertRaisesRegex(ValueError,'Exactly'):x.gpu_idle(str(p),run)
    def test_write_source_exact_arrays_and_reject_overwrite_or_wrong_order(self):

    def test_no_phy_constructor_metrics_or_automatic_successor(self):
        tree=ast.parse(Path(x.__file__).read_text(encoding='utf-8-sig'));calls={ast.unparse(n.func) for n in ast.walk(tree) if isinstance(n,ast.Call)}
        for name in ('BudgetLedger','PacketPHY','MetricScorer','load_suite','encode_prefixes'):self.assertNotIn(name,calls)
        self.assertIn('quality.build_native',calls);self.assertIn('core.entropy_orders',calls);self.assertIn('core.parity_source',calls)
        supervise=next(n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name=='supervise')
        text=ast.unparse(supervise);self.assertIn('child.wait()',text);self.assertNotIn('child.kill()',text);self.assertNotIn('child.terminate()',text)
