def identity(pid):

def exited(i):
    try:now=identity(i['pid'])
    except FileNotFoundError:return True
    require(now['start_ticks']!=i['start_ticks'] or now['uid']!=i['uid'],'Original process remains present, including unreaped zombie')
    return True

@contextmanager
def lock(path):
    import fcntl
    with Path(path).open('a+') as f:
        fcntl.flock(f,fcntl.LOCK_EX|fcntl.LOCK_NB)
        try:yield
        finally:fcntl.flock(f,fcntl.LOCK_UN)
def locked(path):
    import fcntl
    with Path(path).open('r+') as f:
        try:fcntl.flock(f,fcntl.LOCK_EX|fcntl.LOCK_NB)
        except BlockingIOError:return True
        fcntl.flock(f,fcntl.LOCK_UN);return False

        require(native.loaded['identity']==ctx['calibration']['identity'] and native.flags==ctx['flags'] and native.driver_bindings==ctx['static'],'Original complete native identity/FP32/source graph differs')
        rows=[];records=[];outputs={}
        for record in ctx['plan']['sources']:
            check(True);flat=core.load_raw_tokens(record,bound=ctx['bound']);orders,stats=core.entropy_orders(native,quality.q,flat,record['required_ms'],check)
            parity=core.parity_source(native,quality.q,ctx['c848']['sup'].c,ctx['plan'],record['source_index'],flat,orders,check) if record['source_index']<8 else []
            item,pins=write_source(ctx,record,flat,orders,stats,parity);records.append(item);outputs.update(pins);rows.extend(parity)
            status=out/'status.json';tmp=status.with_suffix('.pending');tmp.write_text(json.dumps(dict(completed_sources=len(records),entropy_pairs=sum(len(x['required_ms']) for x in records),parity_cases=len(rows)))+'\n');os.replace(tmp,status)
        check(True);native.frozen();require(budget(ctx)==ctx['before'],'Entropy stage changed C charged calls');verify(ctx['sources']);verify(ctx['inputs'])
        parity=core.finish_parity(ctx['plan'],rows);parity['registration_sha256']=ctx['regsha'];pp=out/'parity_completion.json';save(pp,parity);outputs.update(bind([pp]))
        result=dict(status=core.COMPLETE,registration_sha256=ctx['regsha'],config_sha256=sha(config_path),source_ids=ctx['plan']['source_ids'],source_count=64,
            entropy_pair_count=206,sources=records,parity_completion=dict(path=str(pp),sha256=sha(pp)),outputs=outputs,source_bindings=ctx['reg']['source_bindings'],input_bindings=ctx['reg']['input_bindings'],
            numerical_runtime=native.flags,frozen_visual_identity=native.loaded['identity'],visual_source_bindings=native.driver_bindings,budget_before=ctx['before'],budget_after=budget(ctx),worker_identity=me,
            new_packet_decodes=0,new_source_encoding=0,policy_selection=False,quality_metrics_run=False,development_results_created=False,automatic_successor=False,elapsed_seconds=time.monotonic()-start)
        save(out/'completion.json',result);return result
    except BaseException:save(out/'failure.json',dict(status='FAILED_PRESERVE_NO_RETRY',traceback=traceback.format_exc(),new_packet_decodes=0));raise

def supervise(request_path):
    r=read(request_path);e=Path(r['execution_dir']);seal=read(e/'registration_completion.json');require(seal['status']==SEALED and not(e/'registration_failure.json').exists(),'Successful new registration required');verify(seal['outputs'])
    ctx=load_registered(e/'config.json');require(not(e/'run_attempt.json').exists() and not Path(r['out']).exists(),'No retry of prior C entropy run')
    save(e/'run_attempt.json',dict(registration_sha256=ctx['regsha']));child=None;me=identity(os.getpid());log=e/'worker.log'
    try:
        require(set(r['resources']['affinity'])<=os.sched_getaffinity(0),'Six original CPU cores unavailable')
        with ExitStack() as stack:
            stack.enter_context(lock(ctx['visual']['owner']['visual_lock_path']));stack.enter_context(lock(str(Path(ctx['c848']['ctx']['cfg']['ledger']).with_suffix('.owner.lock'))))
            check_stop(ctx);require(budget(ctx)==ctx['before'],'C ledger changed before GPU');gpu=gpu_idle(ctx['visual']['owner']['nvidia_smi'])
            save(e/'owner_identity.json',dict(identity=me,registration_sha256=ctx['regsha']));save(e/'gpu_admission.json',gpu)
            command=[ctx['visual']['python'],'-B',str(Path(__file__).absolute()),'--stage','worker','--config',str(e/'config.json')]
            with log.open('xb') as h:
                child=subprocess.Popen(command,cwd=r['root'],env=env(ctx,True),stdin=subprocess.DEVNULL,stdout=h,stderr=subprocess.STDOUT,
                    preexec_fn=lambda:resources(r['resources']['affinity']),start_new_session=True)
                save(e/'pending_pid.json',dict(pid=child.pid,argv=command));ident=capture(child,command)
                save(e/'worker_launch.json',dict(identity=ident,registration_sha256=ctx['regsha']));save(e/'worker_admission.json',dict(owner_identity=me,worker_identity=ident,registration_sha256=ctx['regsha'],gpu=gpu))
                start=time.monotonic()
                while child.poll() is None:
                    if STOP or time.monotonic()-start>r['resources']['max_seconds'] or time.time()>DEADLINE:
                        p=Path(r['out'])/'STOP';p.parent.mkdir(parents=True,exist_ok=True)
                        if not p.exists():save(p,dict(reason='Registered C entropy STOP; preserve evidence'))
                    time.sleep(.25)
                code=child.wait();h.flush();os.fsync(h.fileno())
            save(e/'worker_exit.json',dict(identity=ident,exit_code=code,reaped=True,log_sha256=sha(log)))
            require(code==0 and exited(ident),'Worker did not complete normally');check_stop(ctx)
            done=read(Path(r['out'])/'completion.json');require(done['status']==core.COMPLETE and done['worker_identity']==ident and done['registration_sha256']==ctx['regsha']
                and done['source_count']==64 and done['entropy_pair_count']==206 and done['budget_before']==done['budget_after']==ctx['before'],'Exact entropy worker completion required')
            verify(done['outputs']);verify(done['source_bindings']);verify(done['input_bindings']);require(budget(ctx)==ctx['before'],'C budget changed')
            p=read(done['parity_completion']['path']);require(p['status']==core.PARITY_STATUS and p['cases']==91 and p['choice_check_count']==512,'Exact first8 independent parity required')
            result=dict(status=DONE,registration_sha256=ctx['regsha'],worker_identity=ident,owner_identity=me,worker_reaped=True,exit_code=0,budget_after=ctx['before'],
                source_count=64,entropy_pair_count=206,parity_cases=91,source_bindings=ctx['reg']['source_bindings'],input_bindings=ctx['reg']['input_bindings'],
                outputs=combine(done['outputs'],bind([Path(r['out'])/'completion.json',log]+[e/n for n in ('owner_identity.json','worker_launch.json','worker_exit.json','gpu_admission.json','worker_admission.json')])),
                new_packet_decodes=0,automatic_successor=False)
            save(e/'completion.json',result);return result
    except BaseException:
        if child is not None and child.poll() is None:
            p=Path(r['out'])/'STOP';p.parent.mkdir(parents=True,exist_ok=True)
            if not p.exists():save(p,dict(reason='Supervisor failure; stop safely at boundary'))
            code=child.wait()
            if not(e/'worker_exit.json').exists():save(e/'worker_exit.json',dict(pid=child.pid,exit_code=code,reaped=True,log_sha256=sha(log)))
        save(e/'failure.json',dict(status='FAILED_PRESERVE_NO_RETRY',traceback=traceback.format_exc(),worker_pid=None if child is None else child.pid));raise

def main():
    p=argparse.ArgumentParser();p.add_argument('--stage',choices=('register','run','worker','imports'),required=True);p.add_argument('--request');p.add_argument('--config');a=p.parse_args()
    signal.signal(signal.SIGTERM,stop);signal.signal(signal.SIGINT,stop)
    if a.stage=='imports':
        ctx=inspect(a.request);done=targeted_imports(ctx);save(Path(ctx['request']['execution_dir'])/'import_completion.json',done)
    else:done=worker(a.config) if a.stage=='worker' else (register if a.stage=='register' else supervise)(a.request)
    print(done['status'],flush=True)
if __name__=='__main__':main()
