def read(p):return json.loads(Path(p).read_text(encoding='utf-8-sig'))
def sha(p):
    h=hashlib.sha256()
    with Path(p).open('rb') as f:
        for part in iter(lambda:f.read(1048576),b''):h.update(part)
    return h.hexdigest()
def canonical(v):return json.dumps(v,sort_keys=True,separators=(',',':'),allow_nan=False)
def save(p,v):
    p=Path(p);p.parent.mkdir(parents=True,exist_ok=True)
    with p.open('x',encoding='utf-8',newline='\n') as f:
        f.write(json.dumps(v,sort_keys=True,indent=2,allow_nan=False)+'\n');f.flush();os.fsync(f.fileno())
def verify(pins):
    for p,s in pins.items():require(Path(p).is_absolute() and core.raw.sha(s) and sha(p)==s,'Unbound/changed file: '+p)
def bind(paths):return {str(Path(p).absolute()):sha(p) for p in paths}
def combine(*maps):
    result={}
    for values in maps:
        for p,s in values.items():require(p not in result or result[p]==s,'Conflicting SHA: '+p);result[p]=s
    return result
def module(p,name):

def ledger_view(path,budget_sha,*,expected=None,worker=None):
    """Read-only original H schema, historical fingerprint and bounded new rows."""
    require(Path(path).is_file(),'Original H ledger must already exist')
    history=hashlib.sha256();old_counts={k:0 for k in PHASES};new={};observed={k:0 for k in PHASES};unresolved=0
    with closing(sqlite3.connect(Path(path).absolute().as_uri()+'?mode=ro',uri=True)) as db:
        meta=dict(db.execute('SELECT key,value FROM meta'))
        require(meta.get('schema')=='PACKET_PRECHARGE_V1' and meta.get('branch')=='H'
            and meta.get('registration_sha256')==budget_sha and meta.get('total_cap')=='200000'
            and json.loads(meta['limits'])==PHASES,'Original H ledger identity/full budget differs')
        counts=dict(db.execute('SELECT phase,charged FROM counters'))
        for row in db.execute('SELECT event_id,phase,kind,phy_key,request_sha,status,worker,reserved_at,completed_at,result_sha,error FROM events ORDER BY event_id'):
            eid,phase,kind,key,request,status,who,_,_,result,_=row
            require(phase in PHASES,'Foreign phase');observed[phase]+=1;match=PATTERN.fullmatch(eid)
            if match:
                require(phase=='development' and kind==match[4] and int(match[2])<100,'Foreign MAIN4 event')
                require(status in ('COMPLETE','RESERVED'),'Failed event remains charged; stop')
                new[eid]=dict(kind=kind,phy_key=key,request_sha256=request,result_sha256=result,status=status,identity=json.loads(who))
                unresolved+=int(status!='COMPLETE')
            else:
                require(status=='COMPLETE','Historical failed/reserved event blocks MAIN4')
                old_counts[phase]+=1;history.update((canonical(row)+'\n').encode())
        require(counts==observed and old_counts==BEFORE and len(new)<=2400
            and counts['development']==10800+len(new) and sum(counts.values())==164760+len(new),
            'Original counters changed or MAIN4 exceeded reserved2400')
        if expected is not None:
            actual={e:v for e,v in new.items() if worker is None or int(PATTERN.fullmatch(e)[2])%2==worker}
            require(set(actual)==set(expected),'Missing/foreign MAIN4 paid events')
            for eid,wanted in expected.items():
                got=actual[eid];require(got['status']=='COMPLETE' and all(got[k]==wanted[k] for k in ('kind','phy_key','request_sha256','result_sha256')),
                    'Charged event request/result does not match actual trace')
                stored=db.execute('SELECT result FROM events WHERE event_id=?',(eid,)).fetchone()[0]
                require(digest(json.loads(stored))==got['result_sha256'] and json.loads(stored)==wanted['result'],
                    'Stored paid result differs from the actual header/body trace')
    return dict(schema='MAIN4_H_LEDGER_READ_ONLY_VIEW_R1',branch='H',registration_sha256=budget_sha,
        phase_charged=counts,charged=sum(counts.values()),historic_events_sha256=history.hexdigest(),
        MAIN4_calls=len(new),unresolved=unresolved,events=new)


def stable_budget(ctx,*,expected=None,worker=None):

        and before['historic_events_sha256']==v['historic_events_sha256'],'Historical H events changed since registration')
    return v


def audit_inventory(ctx):
    """Re-audit the complete CPU output and paid results without any backend."""
    cfg=ctx['cfg'];paths=[];outputs={};identities={}
    for w in (0,1):
        folder=Path(cfg['out'])/f'worker_{w}';completion=folder/'completion.json';done=read(completion)
        require(done['status']==WORKER_DONE and done['worker_index']==w and done['source_indices']==indices(w)
            and done['source_ids']==ctx['source_ids'] and done['registration_sha256']==ctx['regsha']
            and done['config_sha256']==sha(cfg['_path']) and not(folder/'failure.json').exists(),
            'Exact complete worker receipt required for inventory audit')
        verify(done['outputs']);identities[w]=done['worker_identity'];outputs.update(bind([completion]))
        for i in indices(w):
            tp=folder/'traces'/f'{i:04d}.json';cp=folder/'source_checkpoints'/f'{i:04d}.json';checkpoint=read(cp)
            require(done['outputs'].get(str(tp))==sha(tp) and done['outputs'].get(str(cp))==sha(cp)
                and checkpoint['status']=='MAIN4_H_DEVELOPMENT_CPU_SOURCE_COMPLETE_R1'
                and checkpoint['source_index']==i and checkpoint['source_id']==ctx['source_ids'][i]
                and checkpoint['registration_sha256']==ctx['regsha'] and checkpoint['frame_count']==12
                and checkpoint['outputs']==bind([tp]),'Exact source checkpoint mapping required')
            paths.append(tp);outputs.update(bind([tp,cp]))
        journal=folder/'traces.journal.jsonl'
        require(done['outputs'].get(str(journal))==sha(journal),'Worker journal omitted or changed')
        outputs.update(bind([journal]))
    expected,summary=trace_inventory(ctx,paths);view=stable_budget(ctx,expected=expected)
    require(summary['source_count']==100 and summary['frame_count']==1200 and view['unresolved']==0
        and view['MAIN4_calls']==len(expected)<=2400,'Complete MAIN4 inventory/ledger required')
    for eid,event in view['events'].items():
        require(event['identity']==identities[int(PATTERN.fullmatch(eid)[2])%2],'Paid event belongs to a different worker')
    return dict(summary=summary,outputs=outputs,budget={k:v for k,v in view.items() if k!='events'},
        trace_paths=[str(p) for p in paths],events=expected)

def load_tokens(record,bound):
    paths=[record['checkpoint'],record['codec_checkpoint'],record['archive']]
    for p in paths:require(bound.get(p)==sha(p),'Changed bound raw-token source')
    with np.load(record['archive'],allow_pickle=False) as z:
        require(set(z.files)=={'tokens','pixels'},'Original development NPZ schema differs')
        tokens=z['tokens']  # Never access the co-stored pixels or arithmetic archive.
    require(tokens.dtype==np.dtype('int64') and tokens.shape==(680,) and ((tokens>=0)&(tokens<4096)).all(),'Exact680 int64 tokens required')
    checksum=hashlib.sha256(b'int64:680\0'+tokens.astype('<i8',copy=False).tobytes()).hexdigest()
    require(checksum==record['tokens_sha256'],'Frozen Encoder-verified tokens differ')
    return [tokens[core.raw.PREFIX[i]:core.raw.PREFIX[i+1]] for i in range(10)],bind(paths)

def trace_inventory(ctx,paths,worker=None):
    wanted=set(range(100) if worker is None else indices(worker));seen=set();events={};gray=0;frame_count=0
    points=ctx['context']['by_slot']
    for path in paths:
        doc=read(path);i=doc['source_index'];require(i in wanted and i not in seen,'Duplicate/foreign source trace');seen.add(i)
        require(doc['status']=='MAIN4_H_DEVELOPMENT_SOURCE_TRACES_R1' and doc['source_id']==ctx['source_ids'][i]
            and doc['registration_sha256']==ctx['regsha'],'Wrong source/registration identity')
        record=ctx['records'][i]
        require(doc['source_bindings']==bind([record['checkpoint'],record['codec_checkpoint'],record['archive']]),
            'Trace must bind the exact original development source, not unrelated inputs')
        for p,s in doc['source_bindings'].items():require(ctx['bound'].get(p)==s==sha(p),'Trace source not registered')
        keys=set()
        for f in doc['frames']:
            slot,seed=f['development_slot'],f['noise_seed'];require(slot in points and seed in core.h.SEEDS and (slot,seed) not in keys,'Unknown/duplicate frame');keys.add((slot,seed))
            point=points[slot];event=core.frame_event(slot,i,seed)
            require(all(f[k]==v for k,v in point.items()) and f['source_id']==ctx['source_ids'][i] and f['source_index']==i
                and f['population_role']=='development' and f['execution_registration_sha256']==ctx['regsha']
                and f['engineering_choices_sha256']==digest(core.ENGINEERING) and f['phase']=='development'
                and f['physical_ledger_branch']=='H' and f['public_frame_counter']==core.h.frame_counter(slot,i,seed)
                and f['noise_sha256']==core.raw.array_sha(core.h.standard_noise(f['source_id'],f['snr_db'],seed))
                and f['scrambling_session']=='actual-body' and f['N']==1024 and f['source_decode_complete'] is True
                and f['image_reconstruction_complete'] is False and f['GPU_used'] is False and f['neural_metrics_run'] is False
                and f['policy_selection'] is False,'Actual MAIN4 frame scope differs')
            parsed=core.raw.present_actual(f['header'],f['body'],ctx['catalogue'])
            require(all(f[k]==v for k,v in parsed.items()),'RX-only state must match actual accepted header/hard payload')
            kinds=['header']+(['body'] if f['body'] is not None else [])
            require(f['packet_event_ids']==[event+'.'+k for k in kinds] and f['logical_packet_calls']==len(kinds)
                and len(f['packet_events'])==len(kinds),'Actual metering event count differs')
            for k,e in zip(kinds,f['packet_events']):
                key=core.raw.HEADER_KEY if k=='header' else ctx['catalogue'].entry(f['rx_profile_id'])['groups'][0]['phy_key']
                require(e['event_id']==event+'.'+k and e['kind']==k and e['phy_key']==key and e['result_sha256']==digest(f[k])
                    and core.raw.sha(e['request_sha256']) and e['event_id'] not in events,'Missing/colliding charged event evidence')
                events[e['event_id']]={x:e[x] for x in ('kind','phy_key','request_sha256','result_sha256')}
                events[e['event_id']]['result']=f[k]
            gray+=int(f['gray']);frame_count+=1
        require(keys=={(s,n) for s in points for n in core.h.SEEDS},'Missing one of twelve source frames')
    require(seen==wanted,'Missing source traces')
    return events,dict(source_count=len(seen),frame_count=frame_count,logical_packet_events=len(events),header_gray_frames=gray,
        raw_KEEP_source_decoded_frames=frame_count-gray)

def worker_body(ctx,worker,proof,ledger,backend,phy,transmitter,started):
    out=Path(ctx['cfg']['out'])/f'worker_{worker}';claim(out,ctx,worker);outputs={};paths=[];receiver=None
    with (out/'traces.journal.jsonl').open('x',encoding='utf-8',newline='\n') as journal:
        for i in indices(worker):
            check_boundary(ctx,proof,started,out);scales,source_pins=load_tokens(ctx['records'][i],ctx['bound']);rows=[]
            for point in ctx['context']['schedule']:
                for seed in core.h.SEEDS:
                    check_boundary(ctx,proof,started,out)
                    if receiver is None:
                        boundary=core.MainFrameLedger(ledger,point['development_slot'],i,seed,budget_sha=sha(ctx['cfg']['budget_registration']),phase_limits=PHASES)
                        receiver=core.HReservedReceiver(ctx['catalogue'],header=transmitter.header,backend=backend,phy=phy,ledger=boundary)
                    row=core.run_frame(ctx=ctx['context'],development_slot=point['development_slot'],source_id=ctx['source_ids'][i],
                        source_index=i,noise_seed=seed,scales=scales,transmitter=transmitter,backend=backend,phy=phy,ledger=ledger,receiver=receiver)
                    journal.write(canonical(row)+'\n');journal.flush();os.fsync(journal.fileno());rows.append(row)
            path=out/'traces'/f'{i:04d}.json';save(path,dict(status='MAIN4_H_DEVELOPMENT_SOURCE_TRACES_R1',registration_sha256=ctx['regsha'],
                source_index=i,source_id=ctx['source_ids'][i],frames=rows,source_bindings=source_pins))
            cp=out/'source_checkpoints'/f'{i:04d}.json';save(cp,dict(status='MAIN4_H_DEVELOPMENT_CPU_SOURCE_COMPLETE_R1',
                registration_sha256=ctx['regsha'],source_index=i,source_id=ctx['source_ids'][i],frame_count=12,outputs=bind([path])))
            outputs.update(bind([path,cp]));paths.append(path)
    outputs.update(bind([out/'traces.journal.jsonl']))
    expected,summary=trace_inventory(ctx,paths,worker);view=stable_budget(ctx,expected=expected,worker=worker)
    require(all(view['events'][e]['identity']==ledger.worker for e in expected),'Paid events belong to another worker')
    return receipt(ctx,worker,outputs,summary,view,proof,ledger.worker)

def merge_body(ctx,proof,started):
    out=Path(ctx['cfg']['out'])/'merge'
    check_boundary(ctx,proof,started,out);claim(out,ctx,None);closed=ctx['owner'].closed_workers(ctx)
    require(len(closed)==2 and {x['worker_index'] for x in closed}=={0,1},'Two normally reaped workers required')
    outputs={};paths=[];owners={}
    for closed_worker in closed:
        i=closed_worker['worker_index'];folder=Path(ctx['cfg']['out'])/f'worker_{i}';p=folder/'completion.json';done=read(p)
        require(closed_worker['completion']==str(p) and closed_worker['completion_sha256']==sha(p)
            and closed_worker['exit_code']==0 and closed_worker['wait_completed'] is True and closed_worker['exited'] is True
            and closed_worker['identity']==done['worker_identity'] and not(folder/'failure.json').exists()
            and done['status']==WORKER_DONE and done['worker_index']==i and done['source_indices']==indices(i)
            and done['registration_sha256']==ctx['regsha'] and done['config_sha256']==sha(ctx['cfg']['_path']),
            'Worker completion/normal exit differs')
        verify(done['outputs']);verify(closed_worker['bindings']);outputs=combine(outputs,done['outputs'],closed_worker['bindings'],bind([p]));owners[i]=done['worker_identity']
        for index in indices(i):
            path=folder/'traces'/f'{index:04d}.json';cp=folder/'source_checkpoints'/f'{index:04d}.json'
            require(done['outputs'].get(str(path))==sha(path) and done['outputs'].get(str(cp))==sha(cp)
                and read(cp)['outputs']==bind([path]),'Unsealed source checkpoint/trace');paths.append(path)
    events,summary=trace_inventory(ctx,paths);view=stable_budget(ctx,expected=events)
    require(summary['frame_count']==1200 and view['unresolved']==0 and view['MAIN4_calls']==len(events)<=2400,'Whole MAIN4 grid/ledger must close')
    for eid,value in view['events'].items():require(value['identity']==owners[int(PATTERN.fullmatch(eid)[2])%2],'Wrong physical event owner')
    check_boundary(ctx,proof,started,out)
    return receipt(ctx,None,outputs,summary,view,proof,proof['worker_identity'])


def run(config_path,worker=None):
    ctx=load_registered(config_path);cfg=ctx['cfg'];out=Path(cfg['out'])/(f'worker_{worker}' if worker is not None else 'merge')
    require(sys.platform.startswith('linux'),'Actual process/resource admission requires Linux')
    started=time.monotonic()
    try:
        proof=ctx['owner'].admit_worker(ctx,worker)
        require(proof['status']=='MAIN4_CPU_ADMISSION_VERIFIED_R2' and proof['registration_sha256']==ctx['regsha']
            and proof['config_sha256']==sha(cfg['_path']) and proof['worker_index']==worker
            and proof['phase']=='development' and proof['owner_locks_held']==['shared_visual','H_ledger_owner'],
            'Explicit live owner/locks admission required')
        verify(proof['bindings']);require(proof['worker_identity'] is not None,'Current worker or merge identity required')
        cores=cfg['cpu_affinities'][0 if worker is None else worker]
        require(os.sched_getaffinity(0)==set(cores) and os.getpriority(os.PRIO_PROCESS,0)==15
            and os.environ.get('CUDA_VISIBLE_DEVICES')=='' and all(os.environ.get(k)=='2' for k in
            ('OMP_NUM_THREADS','MKL_NUM_THREADS','OPENBLAS_NUM_THREADS','NUMEXPR_NUM_THREADS')),'Bound CPU resources differ')
        check_boundary(ctx,proof,started,out)
        if worker is None:done=merge_body(ctx,proof,started)
        else:
            indices(worker);m=module(cfg['ledger_module'],'main4_original_H_ledger')
            ledger=m.BudgetLedger(cfg['ledger'],sha(cfg['budget_registration']),'H',PHASES)
            require(m.same_identity(ledger.worker,proof['worker_identity']),'Admission differs from actual ledger worker')
            backend,phy,tx=original_runtime(ctx)
            done=worker_body(ctx,worker,proof,ledger,backend,phy,tx,started)
        verify(ctx['reg']['source_bindings']);verify(ctx['reg']['input_bindings']);check_boundary(ctx,proof,started,out)
        save(out/'completion.json',done);return done
    except BaseException as error:
        if not(out/'failure.json').exists():save(out/'failure.json',dict(status='FAILED_PRESERVE_NO_RETRY',
            registration_sha256=ctx['regsha'],error=repr(error),traceback=traceback.format_exc(),ledger_never_refunded=True))
        raise


def main():
    p=argparse.ArgumentParser();p.add_argument('--config',required=True);p.add_argument('--stage',choices=['worker','merge'],required=True)
    p.add_argument('--worker-index',type=int);a=p.parse_args()
    require((a.stage=='worker' and a.worker_index in (0,1)) or (a.stage=='merge' and a.worker_index is None),'Explicit worker/merge only')
    def stop(*_):raise RuntimeError('Signal: preserve attempt and paid events')
    signal.signal(signal.SIGTERM,stop);signal.signal(signal.SIGINT,stop)
    print(run(a.config,a.worker_index)['status'],flush=True)


if __name__=='__main__':main()
