class Child:
    def __init__(self,pid,code=0,polls=None):
        self.pid=pid;self.code=code;self.returncode=None;self.waits=0;self.polls=list(polls or [code])
    def poll(self):
        result=self.polls.pop(0) if self.polls else self.code
        if result is not None:self.returncode=result
        return result
    def wait(self,timeout=None):self.waits+=1;self.returncode=self.code;return self.code

    def test_owner_locks_stay_held_during_failed_launch_drain(self):
        held=[];events=[];original=s.drain
        @contextmanager
        def locks(cfg):
            held.append(True);events.append('lock')
            try:yield
            finally:held.pop();events.append('unlock')
        def launch(ctx,index,owner):
            if index==1:raise RuntimeError('second spawn failure')
            return self.fake_launch(ctx,index,owner)
        def drain(ctx):self.assertTrue(held);events.append('wait');return original(ctx)
        with ExitStack() as stack:
            self.run_patches(stack);stack.enter_context(patch.object(s,'owner_locks',side_effect=locks));stack.enter_context(patch.object(s,'launch',side_effect=launch));stack.enter_context(patch.object(s,'drain',side_effect=drain))
            stack.enter_context(patch.object(s,'marker',side_effect=OSError('STOP file write failed')))
            with self.assertRaisesRegex(RuntimeError,'spawn'):s.run(self.cfg['owner_config'])
        self.assertEqual(events,['lock','wait','unlock']);self.assertTrue(self.ctx['_active'][0]['handle'].closed)
        self.assertIn('STOP file write failed',s.read(self.e/'owner_failure.json')['stop_marker_error'])
