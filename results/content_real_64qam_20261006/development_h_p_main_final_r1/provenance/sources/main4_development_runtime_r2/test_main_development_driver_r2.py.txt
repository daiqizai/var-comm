def write(path,value):
    path=Path(path);path.parent.mkdir(parents=True,exist_ok=True);path.write_text(json.dumps(value),encoding='utf-8')


def assets(folder,args):
    pop=copy.deepcopy(args['development_registration']);pop['identity']={'frozen':'fixture'}
    records=[];source_records=[];bound={};asset_outputs={};source_outputs={}
    for i,sid in enumerate(pop['source_ids']):
        a=Path(folder)/'assets'/f'{i:04d}.json';z=a.with_suffix('.npz');s=Path(folder)/'source'/f'{i:04d}.json';a.parent.mkdir(exist_ok=True)
        tokens=np.arange(680,dtype=np.int64);np.savez(z,tokens=tokens,pixels=np.zeros((3,256,256),np.uint8))
        th=hashlib.sha256(b'int64:680\0'+tokens.astype('<i8').tobytes()).hexdigest()
        acp=dict(source_index=i,source_id=sid,preprocessing_id=pop['preprocessing_ids'][i],tokens_sha256=th,
            original_development_data_binding=pop['data_bindings'][i],archive=str(z),outputs=d.bind([z]))
        write(a,acp);asset=dict(source_index=i,source_id=sid,preprocessing_id=acp['preprocessing_id'],
            tokens_sha256=th,checkpoint=str(a),checkpoint_sha256=d.sha(a),archive=str(z))
        write(s,dict(status='H_DEVELOPMENT100_SOURCE_CODEC_SOURCE_COMPLETE',source_index=i,source_id=sid,
            registration_sha256='b'*64,encoder_tokens_verified=True,independent_roundtrip=True,tokens_sha256=th,
            source_assets_checkpoint=dict(path=str(a),sha256=d.sha(a))))
        records.append(asset);source_records.append(dict(source_index=i,source_id=sid,checkpoint=str(s),sha256=d.sha(s)))
        bound.update(d.bind([a,z,s]));asset_outputs.update(d.bind([a,z]));source_outputs.update(d.bind([s]))
    manifest=dict(status='H_DEVELOPMENT100_CPU_ASSETS_READY_SOURCE_ENCODING_INCOMPLETE',source_ids=pop['source_ids'],
        preprocessing_ids=pop['preprocessing_ids'],original_data_bindings=pop['data_bindings'],visual_identity=pop['identity'],records=records)

        encoder_tokens_verified=True,independent_roundtrip=True,new_packet_decodes=0,new_metric_calls=0,holdout_used=False,
