def require(v,m):
    if not v:raise RuntimeError(m)
def read(p):return json.loads(Path(p).read_text(encoding='utf-8-sig'))
def sha(p):
    h=hashlib.sha256()
    with Path(p).open('rb') as f:
        for b in iter(lambda:f.read(1048576),b''):h.update(b)
    return h.hexdigest()
def save(p,v):
    p=Path(p);p.parent.mkdir(parents=True,exist_ok=True)
    with p.open('x',encoding='utf-8',newline='\n') as f:
        json.dump(v,f,sort_keys=True,indent=2,allow_nan=False);f.write('\n');f.flush();os.fsync(f.fileno())
def combine(*maps):
    out={}
    for m in maps:
        for p,s in m.items():require(p not in out or out[p]==s,'Conflicting binding '+p);out[p]=s
    return out
def bind(paths):return {str(Path(p).absolute()):sha(p) for p in paths}
def verify(m):
    for p,s in m.items():require(Path(p).is_absolute() and sha(p)==s,'Unbound/changed file '+p)
def module(p,name,bound):
    p=str(Path(p).absolute());require(bound.get(p)==sha(p),'Must bind module before import '+p)
    spec=importlib.util.spec_from_file_location(name,p);m=importlib.util.module_from_spec(spec)
    sys.modules[name]=m;spec.loader.exec_module(m);return m
def identity(i):return {k:i[k] for k in ('pid','start_ticks','uid','argv')}


def checked_shape(r,*,fresh=True):
    require(r['schema']==REQUEST and r['deadline_unix']==DEADLINE,'Independent MAIN4 request/original deadline required')
    root=Path(r['root']);base=root/'outputs/CONTENT-REAL-64QAM-20261006'
    e,out=Path(r['execution_dir']),Path(r['out'])
    require(root.is_absolute() and e.parent==out.parent==base/'MAIN' and e!=out,'Fresh disjoint MAIN4 evidence/output paths')
    require(not fresh or (not e.exists() and not out.exists()),'Old execution/output preserved; no retry')
    require(r['resources']['cpu_affinities']==[[0,1],[2,3]] and r['resources']['threads']==2
        and r['resources']['nice']==15 and r['resources']['GPU_jobs']==0
        and 0<r['resources']['max_worker_seconds']<=7200,'Original H two-worker CPU resource contract required')
    require(r['automatic_successor'] is False and r['development_policy_selection'] is False,'No successor or development selection')
    require(Path(r['python']).is_absolute() and r['python']==str(root/'outputs/PRIOR-AWARE-UEP-20261004-V1/ldpc_environment/bin/python'),
        'Original qualified LDPC interpreter, without resolving the venv symlink')
    require(len(r['pythonpath'])==len(set(r['pythonpath'])) and all(Path(p).is_absolute() for p in r['pythonpath'])
        and r['pythonpath'][0]==str(Path(__file__).absolute().parent),'Explicit new/private and original runtime paths required')
    require(set(r['H_zero_PHY_batches'])==set(ZERO_STATUSES),'Every historical164760 zero-PHY branch must close')
    require(r['visual_lock']==str(base/'shared_visual.lock'),'Original shared visual lock required')
    return e,out


def normal_h(spec,expected,ctx):
    """Use the original actual receipt/exit verifier, without source re-encoding."""
    require(set(spec)=={'config','registration','launch','completion'},'Exact original stage_owner spec required')
    for p in spec.values():require(ctx['inputs'].get(p)==sha(p),'Explicit prerequisite control pin missing '+p)
    reg=read(spec['registration']);verify(reg['source_bindings'])
    ctx['sources']=combine(ctx['sources'],reg['source_bindings'])
    old=ctx['h_cpu'].closed_batch(ctx['h_owner'],ctx['h_wait'],spec,ctx['request']['H_owner_module'],expected,ctx['h_owner'].raw_process_state)
    ctx['inputs']=combine(ctx['inputs'],old['bindings'],bind(spec.values()))
    # The old verifier checked every original input. Preserve the immutable
    # registration containing those pins; do not repeatedly hash unused historic
    # RGB/model assets while the new CPU stage consumes only explicitly bound
    # token archives, checkpoints, codebook and physical dependencies.
    ctx.setdefault('historical_registration_bindings',{}).update(bind([spec['registration']]))
    ctx['identities'] += [identity(old['owner_identity'])]+[identity(i) for i in old['children']]
    return old

        historical_registration_bindings=ctx.get('historical_registration_bindings',{}),
        historical_input_scope='Original normal APIs verified every historical input. Their exact registration bytes preserve all pins; unused ancestor model/RGB files are not newly consumed. All actual MAIN4 inputs are direct registration pins.')

def environment(r):
    env=dict(os.environ,CUDA_VISIBLE_DEVICES='',PYTHONDONTWRITEBYTECODE='1',PYTHONUNBUFFERED='1',
        PYTHONPATH=os.pathsep.join(r['pythonpath']),MAIN4_TEST_SUPPORT_PATHS=json.dumps(r['test_support_paths']),
        MAIN_TEST_PLAN_SUMMARY=r['test_plan_summary'],MAIN_TEST_UEP_RUNTIME=r['test_uep_runtime'],

    env.update({k:'2' for k in ('OMP_NUM_THREADS','MKL_NUM_THREADS','OPENBLAS_NUM_THREADS','NUMEXPR_NUM_THREADS')})

def cpu_resources(cores):
    require(set(cores)<=os.sched_getaffinity(0),'CPU cores unavailable')
    os.sched_setaffinity(0,cores);os.setpriority(os.PRIO_PROCESS,0,15)


def run_child(ctx,folder,argv,cores):
    """One explicit child, normal wait, closed log; never relaunch an attempt."""
    folder=Path(folder);require(not folder.exists(),'Prior operation preserved; no retry');folder.mkdir(parents=True)
    save(folder/'attempt.json',dict(argv=argv,request_sha256=sha(ctx['rp']),automatic_retry=False))
    child=None;handle=None;ident=None
    try:
        handle=(folder/'process.log').open('xb')
        child=subprocess.Popen(argv,cwd=str(Path(__file__).parent),env=environment(ctx['request']),stdin=subprocess.DEVNULL,
            stdout=handle,stderr=subprocess.STDOUT,start_new_session=True,preexec_fn=lambda:cpu_resources(cores))
        save(folder/'spawn_pending_identity.json',dict(pid=child.pid,argv=argv,identity_verified=False))
        ident=ctx['owner'].identity(child.pid);require(ident['argv']==argv and ident['uid']==os.getuid(),'Actual child argv/UID differ')
        save(folder/'launch.json',dict(identity=ident,argv=argv));rc=child.wait()
        handle.flush();os.fsync(handle.fileno());handle.close();handle=None
        save(folder/'exit.json',dict(identity=ident,exit_code=rc,process_waited=True,launch_sha256=sha(folder/'launch.json'),log_sha256=sha(folder/'process.log')))
        require(rc==0 and ctx['owner'].absent(ident),'Actual child failed or was not reaped')
        return bind([folder/'attempt.json',folder/'spawn_pending_identity.json',folder/'launch.json',folder/'exit.json',folder/'process.log'])
    except BaseException:
        error=traceback.format_exc();marker_error=None;rc=None
        if child is not None:
            # A launch child can already own two physical workers. Never kill
            # that owner or release evidence while those workers remain live.
            # Its bound STOP path makes its own registered drain hold both locks.
            if '--config' in argv:
                try:
                    stop=Path(ctx['hcfg']['stop_file'])
                    if not stop.exists():save(stop,dict(status='MAIN4_LAUNCHER_FAILURE_STOP',request_sha256=sha(ctx['rp']),pid=child.pid))
                except BaseException:marker_error=traceback.format_exc()
            while True:
                try:rc=child.wait();break
                except KeyboardInterrupt:continue
            if handle is not None:
                handle.flush();os.fsync(handle.fileno());handle.close();handle=None
            xp=folder/'exit.json'
            if not xp.exists():save(xp,dict(identity=ident,pid=child.pid,exit_code=rc,process_waited=True,
                normal_launch_identity_verified=ident is not None,log_sha256=sha(folder/'process.log')))
        save(folder/'failure.json',dict(status='FAILED_PRESERVE_INSPECT_NO_RETRY',pid=None if child is None else child.pid,
            traceback=error,exit_code=rc,child_wait_completed=child is not None,stop_marker_error=marker_error))
        raise
    finally:
        if handle is not None:handle.close()

    require(sys.platform.startswith('linux'),'Original Linux process identity required')
    ctx=inspect_inputs(request_path);r=ctx['request'];q=Path(r['prepared_qualification'])

    log=q.parent/'process.log';text=log.read_text(encoding='utf-8',errors='replace')
    require(re.search(r'Ran '+str(r['qualification_test_count'])+r' tests in .*\n\s*OK\s*$',text)
        and not re.search(r'\bskipped\b',text,re.I),'Exact full pure qualification, no skips')
    check_stops(ctx);now=ctx['driver'].ledger_view(ctx['hcfg']['ledger'],sha(ctx['hcfg']['budget_registration']))
    require({k:v for k,v in now.items() if k!='events'}==ctx['before'],'Pure qualification changed H accounting')
    value=dict(status=QUALIFIED,python=r['python'],test_modules=TEST_MODULES,tests_run=r['qualification_test_count'],
        process_waited=True,exit_code=0,source_bindings=ctx['sources'],input_bindings=ctx['inputs'],outputs=outputs,
        log=str(log),pythonpath=r['pythonpath'],budget_before=ctx['before'],budget_after=ctx['before'],
        new_packet_decodes=0,GPU_used=False,actual_backend_called=False,actual_development_observations_read=False)
    save(q,value);return value


def launch(request_path):
