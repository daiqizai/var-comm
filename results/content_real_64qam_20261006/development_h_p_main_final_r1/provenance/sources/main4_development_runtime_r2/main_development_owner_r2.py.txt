def identity(pid):
    p=Path('/proc')/str(pid);fields=(p/'stat').read_text().rsplit(')',1)[1].split()
    require(fields[0] not in ('Z','X'),'Live process identity required')
    return dict(pid=pid,start_ticks=int(fields[19]),uid=p.stat().st_uid,
        argv=(p/'cmdline').read_bytes().decode().rstrip('\0').split('\0'))


def absent(value):
    """A matching zombie is not a reaped predecessor; a reused PID is distinct."""
    p=Path('/proc')/str(value['pid'])
    try:
        fields=(p/'stat').read_text().rsplit(')',1)[1].split()
        return int(fields[19])!=value['start_ticks'] or p.stat().st_uid!=value['uid']
    except FileNotFoundError:return True


def atomic_save(path,value):
    path=Path(path);tmp=path.with_name(path.name+'.pending-'+str(os.getpid()))
    save(tmp,value)
    try:os.link(tmp,path)
    finally:tmp.unlink()


def main_budget(cfg,bound):
    for key in ('main_budget_module','main_budget_registration'):
        require(bound.get(cfg[key])==sha(cfg[key]),'Explicit MAIN readonly dependency pin required: '+key)
    require(Path(cfg['main_ledger']).is_file() and cfg['main_ledger'] not in bound,
        'Existing mutable MAIN ledger is readonly, never a file-SHA input')
    m=d.module(cfg['main_budget_module'],'main4_readonly_MAIN_budget')
    value=m.readonly_budget(cfg['main_ledger'],sha(cfg['main_budget_registration']))
    require(value['branch']=='MAIN' and value['charged']==105176 and value['unresolved']==0
        and value['phase_charged']==dict(qualification=2168,proxy_phy=43008,actual_calibration=60000)
        and all(x['status']=='COMPLETE' for x in value['counts']), 'Normal MAIN105176 accounting required')
    return value

    tail=Path(log).read_text(encoding='utf-8').rstrip()

def validate_contract(cfg,ctx):
    p=cfg['owner_config'];oc=read(p)
    require(ctx['bound'].get(p)==sha(p) and ctx['bound'].get(cfg['owner_module'])==sha(__file__),
        'Explicit actual owner/config bindings required')
    require(oc['schema']==SCHEMA and oc['execution_dir']==str(Path(cfg['registration']).parent)
        and oc['worker_config']==cfg['_path'] and oc['driver_module']==str(Path(d.__file__).absolute())
        and oc['automatic_successor'] is False,'Exact standalone MAIN4 owner contract required')
    for k in ('root','registration','python','cpu_affinities','nice','max_worker_seconds','deadline_unix',
        'visual_lock','ledger','stop_file','prepared_qualification','pythonpath'):
        require(oc[k]==cfg[k],'Owner/worker contract mismatch: '+k)
    cpus=cfg['cpu_affinities']
    require(len(cpus)==2 and all(len(x)==len(set(x))==2 and all(type(c)is int and c>=0 for c in x) for x in cpus)
        and len(set(sum(cpus,[])))==4 and cfg['nice']==15 and 0<cfg['max_worker_seconds']<=21600
        and cfg['deadline_unix']==d.DEADLINE,'Two disjoint CPU2 workers under original deadline required')
    require(cfg['pythonpath'] and cfg['pythonpath'][0]==str(Path(__file__).parent)
        and len(set(cfg['pythonpath']))==len(cfg['pythonpath']) and all(Path(p).is_absolute() for p in cfg['pythonpath']),
        'Explicit qualified runtime order required')
    require(cfg['ledger'] not in ctx['bound'] and cfg['main_ledger'] not in ctx['bound']
        and Path(cfg['ledger']).absolute()!=Path(cfg['main_ledger']).absolute(),'Separate existing ledgers, no mutable SHA pins')
    qualification(cfg,ctx);return oc


def load(path,*,require_seal=True):
    oc=read(path);ctx=d.load_registered(oc['worker_config']);cfg=ctx['cfg']
    require(cfg['owner_config']==str(Path(path).absolute()),'Wrong owner config')
    ctx['owner_config']=validate_contract(cfg,ctx)
    e=Path(oc['execution_dir'])
    require(not(e/'registration_failure.json').exists(),'Failed registration cannot be used')
    if require_seal:
        seal=read(e/'registration_completion.json')
        require(seal['status']==REGISTERED and seal['registration_sha256']==ctx['regsha'],'Normal new registration seal required')
        verify(seal['outputs'])
    return ctx


def check_stop(cfg,*,deadline=True):
    base=Path(cfg['root'])/'outputs/CONTENT-REAL-64QAM-20261006'
    require(not STOP and (not deadline or time.time()<cfg['deadline_unix']) and not any(p.exists() for p in
        (base/'STOP',base/'MAIN/STOP',Path(cfg['stop_file']),Path(cfg['out'])/'STOP',Path(cfg['registration']).parent/'STOP')),
        'MAIN4 STOP/deadline requires preserving paid evidence')

    value=read(p)

        and value['config_sha256']==sha(cfg['_path']) and value['worker_index']==index
        and value['phase']=='development' and value['owner_locks_held']==['shared_visual','H_ledger_owner']
        and value['worker_identity']['argv']==command(cfg,index)
        and value['source_indices']==([] if index is None else d.indices(index))

        'Exact current CPU worker/merge admission required')
    value['bindings']=bind([p,Path(cfg['registration']).parent/'owner_identity.json',folder(cfg,index)/'launch.json'])
    check_live_admission(ctx,value)
    if index is None:closed_workers(ctx)

def check_live_admission(ctx,value):
    cfg=ctx['cfg'];check_stop(cfg)
    require(value['registration_sha256']==ctx['regsha'] and identity(os.getpid())==value['worker_identity']
        and identity(os.getppid())==value['owner_identity'],'Actual original live owner/child identity differs')
    require(os.environ.get('CUDA_VISIBLE_DEVICES')=='' and all(os.environ.get(k)=='2' for k in THREAD_KEYS)
        and os.sched_getaffinity(0)==set(value['affinity']) and os.getpriority(os.PRIO_PROCESS,0)==15,
        'Actual CPU-only affinity/thread/nice differs')
    require(lock_held(cfg['visual_lock']) and lock_held(str(Path(cfg['ledger']).with_suffix('.owner.lock'))),
        'Registered shared visual and H owner locks must remain held')

@contextmanager
def owner_locks(cfg):
    import fcntl
    with ExitStack() as stack:
        for p in (Path(cfg['visual_lock']),Path(cfg['ledger']).with_suffix('.owner.lock')):
            require(p.parent.is_dir(),'Existing registered lock directory required')
            f=stack.enter_context(p.open('a+'));fcntl.flock(f,fcntl.LOCK_EX|fcntl.LOCK_NB)
        yield


def lock_held(path):
    import fcntl
    with Path(path).open('r+') as f:
        try:fcntl.flock(f,fcntl.LOCK_EX|fcntl.LOCK_NB)
        except BlockingIOError:return True
        fcntl.flock(f,fcntl.LOCK_UN);return False


def environment(cfg):

        PYTHONDONTWRITEBYTECODE='1',PYTHONUNBUFFERED='1',**{k:'2' for k in THREAD_KEYS})
def resources(affinity):os.sched_setaffinity(0,set(affinity));os.setpriority(os.PRIO_PROCESS,0,15)


def capture(child,argv):
    for _ in range(100):
        now=identity(child.pid)
        if now['argv']==argv:return now
        require(child.poll() is None,'Child exited before identity admission');time.sleep(.02)
    raise RuntimeError('Actual child argv never matched')


def launch(ctx,index,owner):
    cfg=ctx['cfg'];f=folder(cfg,index);require(not f.exists(),'No retry of old worker/merge');f.mkdir(parents=True)
    argv=command(cfg,index);aff=cfg['cpu_affinities'][0 if index is None else index];handle=(f/'worker.log').open('xb')
    try:
        child=subprocess.Popen(argv,cwd=cfg['root'],env=environment(cfg),stdin=subprocess.DEVNULL,stdout=handle,
            stderr=subprocess.STDOUT,start_new_session=True,preexec_fn=lambda:resources(aff))
    except BaseException:handle.close();raise
    record=dict(child=child,handle=handle,folder=f,index=index,identity=None,closed=False);ctx['_active'].append(record)
    save(f/'pending_pid.json',dict(pid=child.pid,argv=argv,identity_verified=False))
    me=capture(child,argv);record['identity']=me
    save(f/'launch.json',dict(identity=me,argv=argv,registration_sha256=ctx['regsha']))
    atomic_save(f/'admission.json',dict(status='MAIN4_CPU_ADMISSION_VERIFIED_R2',registration_sha256=ctx['regsha'],
        config_sha256=sha(cfg['_path']),owner_identity=owner,worker_identity=me,worker_index=index,phase='development',
        owner_locks_held=['shared_visual','H_ledger_owner'],affinity=aff,source_indices=[] if index is None else d.indices(index)))
    return record


def finish(ctx,record):
    if record['closed']:return record['exit']
    child=record['child'];code=child.wait();h=record['handle'];h.flush();os.fsync(h.fileno());h.close();record['closed']=True
    cfg=ctx['cfg'];i=record['index'];cp=Path(cfg['out'])/('merge' if i is None else f'worker_{i}')/'completion.json'
    end=dict(identity=record['identity'],pid=child.pid,exit_code=code,wait_completed=True,registration_sha256=ctx['regsha'],
        closed_log_sha256=sha(record['folder']/'worker.log'),completion=str(cp),completion_sha256=sha(cp) if cp.is_file() else None)
    save(record['folder']/'exit.json',end);record['exit']=end;return end


def wait_group(ctx,records,started):
    while any(not r['closed'] for r in records):
        check_stop(ctx['cfg']);require(time.monotonic()-started<ctx['cfg']['max_worker_seconds'],'Owner wall cap')
        for r in records:
            # Do not query a process identity between poll and wait: it may be Z.
            if not r['closed'] and r['child'].poll() is not None:
                require(finish(ctx,r)['exit_code']==0,'Registered child failed; preserve paid H events')
        if any(not r['closed'] for r in records):time.sleep(.25)


def drain(ctx):
    for r in ctx['_active']:
        if r['closed']:continue
        child=r['child']
        if child.poll() is None:
            # A direct unreaped Popen child cannot have its PID reused here.
            try:os.killpg(child.pid,signal.SIGTERM)
            except ProcessLookupError:pass
        try:child.wait(timeout=30)
        except subprocess.TimeoutExpired:
            try:os.killpg(child.pid,signal.SIGKILL)
            except ProcessLookupError:pass
            child.wait()
        finish(ctx,r)

        and done['maximum_new_packet_calls']==2400 and done['source_decode_complete'] is True
        and done['development_used'] is True and all(done[k] is False for k in
        ('image_reconstruction_complete','images_scored','neural_metrics_run','GPU_used','holdout_used','policy_selection','MAIN_final_delivery','automatic_successor')),
        'Exact MAIN4 raw KEEP CPU merged receipt required')
    require(done['source_bindings']==ctx['reg']['source_bindings'] and done['input_bindings']==ctx['reg']['input_bindings'],
        'Merged source/input contract differs')
    verify(done['outputs'])
    inventory=d.audit_inventory(ctx);summary=inventory['summary'];events=inventory['events'];budget=inventory['budget']

def run(path):
    ctx=load(path);cfg=ctx['cfg'];e=Path(cfg['registration']).parent;ctx['_active']=[]
    require(sys.platform.startswith('linux') and not(e/'owner_attempt.json').exists()
        and not(e/'owner_failure.json').exists() and not Path(cfg['out']).exists(),'Fresh standalone MAIN4 Linux owner required')
    require(set(sum(cfg['cpu_affinities'],[]))<=os.sched_getaffinity(0),'Both registered CPU pairs must be available to owner')
    save(e/'owner_attempt.json',dict(registration_sha256=ctx['regsha'],automatic_retry=False))
    me=identity(os.getpid());started=time.monotonic();locks=ExitStack()
    try:
        locks.enter_context(owner_locks(cfg));check_stop(cfg)
        before=d.stable_budget(ctx);before={k:v for k,v in before.items() if k!='events'}
        require(before==ctx['reg']['budget_before'] and before['MAIN4_calls']==before['unresolved']==0,
            'Fresh original H164760 plus untouched MAIN4 reserve required')
        original_main=main_budget(cfg,ctx['bound']);require(original_main==ctx['predecessor']['main_budget_before'],'MAIN predecessor budget drift')
        save(e/'owner_identity.json',dict(identity=me,registration_sha256=ctx['regsha'],config_sha256=sha(path)))
        workers=[launch(ctx,i,me) for i in (0,1)];wait_group(ctx,workers,started);closed=closed_workers(ctx)
        require(d.stable_budget(ctx)['unresolved']==0,'Both workers must close all charged events before merge')
        merge=launch(ctx,None,me);wait_group(ctx,[merge],started);science,after=audit_merge(ctx)
        require(science['worker_identity']==merge['identity'],'Merged scientific receipt must name actual admitted merge process')
        require(main_budget(cfg,ctx['bound'])==original_main,'MAIN ledger must remain unchanged')
        check_stop(cfg);verify(ctx['reg']['source_bindings']);verify(ctx['reg']['input_bindings'])
        outputs=combine(science['outputs'],*(x['bindings'] for x in closed),bind([Path(cfg['out'])/'merge/completion.json',
            path,cfg['_path'],cfg['registration'],e/'registration_completion.json']+[folder(cfg,None)/n for n in ('launch.json','admission.json','exit.json','worker.log')]))
        done=dict(status=COMPLETE,registration_sha256=ctx['regsha'],config_sha256=sha(cfg['_path']),owner_identity=me,
            worker_identities=[r['identity'] for r in workers],merge_identity=merge['identity'],all_waited=True,
            worker_exit_codes=[0,0],merge_exit_code=0,source_count=100,frame_count=1200,source_ids=ctx['source_ids'],
            new_packet_decodes=science['new_packet_decodes'],maximum_new_packet_calls=2400,budget_before=before,budget_after=after,
            MAIN_budget_before=original_main,MAIN_budget_after=original_main,ledger_charge_authority='ORIGINAL_H_ONLY',
            outputs=outputs,source_bindings=ctx['reg']['source_bindings'],input_bindings=ctx['reg']['input_bindings'],
            GPU_used=False,images_scored=False,development_used=True,holdout_used=False,policy_selection=False,
            automatic_successor=False,MAIN_final_delivery=False)
        save(e/'completion.json',done);return done
    except BaseException:
        error=traceback.format_exc();marker_error=None
        try:marker(cfg,'MAIN4 owner failed; preserve charged H events')
        except BaseException:marker_error=traceback.format_exc()
        try:drain(ctx)
        finally:
            save(e/'owner_failure.json',dict(status='FAILED_PRESERVE_NO_RETRY',traceback=error,owner_identity=me,
                children=[dict(pid=r['child'].pid,identity=r['identity'],exit_code=r['child'].returncode) for r in ctx['_active']],
                stop_marker_error=marker_error,refund=False))
        raise
    finally:locks.close()

    verify(done['outputs']);return dict(ctx=ctx,done=done,science=science,workers=workers,
        bindings=combine(ctx['bound'],done['outputs'],bind([spec['completion'],cfg['registration'],cfg['owner_config']])) )
