def put(p,v):
    p=Path(p);p.parent.mkdir(parents=True,exist_ok=True);p.write_text(json.dumps(v),encoding='utf8');return str(p)

class FakeTorch:
    float32=np.float32
    @staticmethod
    def as_tensor(a,**kwargs):return np.asarray(a,dtype=kwargs['dtype'])


class FakeTX:
    def __init__(self):self.calls=[]
    def transmit(self,pid,payloads,counter):
        self.calls.append((pid,payloads,counter));wave=np.ones((1024,2),dtype=np.float64)*(pid+1)
        return wave,dict(N=1024,header_uses=68,body_uses=100,idle_uses=856,E=float((wave*wave).sum()),
            waveform_sha256=d.rx.array_sha(wave),profile_id=pid,frame_counter=counter)


def setup(tmp,*,pid=1,accepted=False,fail=False):
    cat=fake_cat();ident=dict(pid=123,start_ticks=456,uid=1002,argv=['fake-worker'])
    ledger=d.budget.BudgetLedger(Path(tmp)/'main.sqlite','a'*64,worker_identity=ident,identity_reader=lambda p:ident)
    backend=types.SimpleNamespace(identity=cat.backend_identity,device='cpu',torch=FakeTorch)
    def prepaid(kind):
        with closing(sqlite3.connect(ledger.path)) as c:
            assert c.execute('SELECT count(*) FROM events WHERE kind=? AND status=?',(kind,'RESERVED')).fetchone()[0]==1
    def receive_header(y,snr,codebook):
        prepaid('header');return dict(profile_id=pid,header_crc_ok=pid is not None,
            header_fields_legal=pid is not None,header_ok=pid is not None,score=1.)
    def receive_body(b,y,k,n,q,snr,counters,group,session):
        prepaid('body');assert session=='actual-body' and group==0
        if fail:raise RuntimeError('fake backend failure')
        hard=np.ones((1,k-16),dtype=np.uint8);decoded=np.concatenate([hard,np.zeros((1,16),dtype=np.uint8)],axis=1)
        decoded[0,-1]=int(accepted);return hard,np.array([accepted],dtype=bool),decoded
    phy=types.SimpleNamespace(receive_packet=receive_body,crc_accept_batch=lambda b:np.asarray(b)[...,-1].astype(bool))
    receiver=d.rx.Receiver(cat,header=types.SimpleNamespace(receive=receive_header),backend=backend,phy=phy,ledger=ledger)
    record=dict(source_index=0,source_id='synthetic-cal-source',tokens_sha256='c'*64)
    point=dict(slot=0,snr_db=13,candidate_id=cat.entry(0)['candidate_id'],profile_id=0,
        wire_key=cat.entry(0)['wire_key'],family_memberships=['WHOLE','PARTIAL'])
    scales=[np.zeros(s*s,dtype=np.int64) for s in d.rx.SIZES]
    return dict(cat=cat,ledger=ledger,identity=ident,receiver=receiver,record=record,point=point,scales=scales,tx=FakeTX())


def actual_events(ledger):
    fields=('event_id','phase','kind','phy_key','request_sha','status','worker','result','result_sha')
    with closing(sqlite3.connect(ledger.path)) as c:return {r[0]:dict(zip(fields,r)) for r in c.execute('SELECT '+','.join(fields)+' FROM events')}


class Tests(unittest.TestCase):
    def test_complete_frozen_schedule_shared_K0_once(self):
        p,a,ids,spec,proxy,short=fixture();out=d.schedule(p,a,short,spec)
        self.assertEqual(len(out),10);self.assertEqual([x['slot'] for x in out],list(range(10)))
        self.assertEqual(sum(len(x['family_memberships'])==2 for x in out),2)
        bad=copy.deepcopy(short);bad['cells'][1]['candidate_ids'].append(p[-1]['candidate_id'])
        with self.assertRaises(ValueError):d.schedule(p,a,bad,spec)

    def test_original_namespace_no_candidate_or_snr_seed_fullsix_counter(self):
        z=d.noise('source',4101);raw=json.dumps(['MAIN_CALIBRATION_V1',1024,'source',4101],sort_keys=True,separators=(',',':')).encode()
        n=int.from_bytes(hashlib.sha256(raw).digest()[:16],'little')
        np.testing.assert_array_equal(z,np.random.Generator(np.random.PCG64(n)).standard_normal((1024,2)))
        self.assertEqual(z.dtype,np.float64);self.assertEqual(d.frame_counter(0,13,4101),12000)
        self.assertEqual(d.frame_counter(999,19,4103),17999)
        self.assertFalse(np.array_equal(z,d.noise('source',4102)))
        with self.assertRaises(ValueError):d.frame_counter(0,4,4101)

    def test_actual_wrong_header_rejected_body_is_kept_and_prepaid(self):
        with tempfile.TemporaryDirectory() as td:
            f=setup(td);ctx={'catalogue':f['cat']}
            row=d.execute_frame(ctx,f['record'],f['scales'],f['point'],4101,f['tx'],f['receiver'])
            self.assertEqual(row['profile_id'],0);self.assertEqual(row['rx_profile_id'],1)
            self.assertFalse(row['body_crc_accept']);self.assertFalse(row['gray']);self.assertEqual(row['receiver_state']['m'],2)
            self.assertEqual(row['receiver_state']['prefix'][0],[4095]);self.assertEqual(row['logical_packet_calls'],2)
            events=d.audit_frame(row,ctx,f['record'],f['point'],4101,actual_events(f['ledger']),f['identity'])
            self.assertEqual(len(events),2)

    def test_unknown_header_only_one_charge_no_body_or_truth_repair(self):
        with tempfile.TemporaryDirectory() as td:
            f=setup(td,pid=None);ctx={'catalogue':f['cat']}
            row=d.execute_frame(ctx,f['record'],f['scales'],f['point'],4101,f['tx'],f['receiver'])
            self.assertTrue(row['gray']);self.assertEqual(row['source_status'],'HEADER_REJECT_GRAY')
            self.assertIsNone(row['body']);self.assertEqual(row['logical_packet_calls'],1)
            self.assertEqual(len(d.audit_frame(row,ctx,f['record'],f['point'],4101,actual_events(f['ledger']),f['identity'])),1)

    def test_ledger_actual_bits_request_and_source_grid_tampering_refused(self):
        with tempfile.TemporaryDirectory() as td:
            f=setup(td);ctx={'catalogue':f['cat']};row=d.execute_frame(ctx,f['record'],f['scales'],f['point'],4101,f['tx'],f['receiver'])
            rows=actual_events(f['ledger'])
            for mutation in (lambda x:x.update(source_id='wrong'),lambda x:x.update(received_sha256='0'*64),
                lambda x:x['body']['hard_payload'].__setitem__(0,0),lambda x:x.update(noise_sha256='0'*64),
                lambda x:x['packet_events'][0].update(request_sha256='0'*64)):
                bad=copy.deepcopy(row);mutation(bad)
                with self.assertRaises(ValueError):d.audit_frame(bad,ctx,f['record'],f['point'],4101,rows,f['identity'])

    def test_same_K0_logical_roles_reuse_exact_paid_event_only(self):
        with tempfile.TemporaryDirectory() as td:
            f=setup(td);ctx={'catalogue':f['cat']}
            first=d.execute_frame(ctx,f['record'],f['scales'],f['point'],4101,f['tx'],f['receiver'])
            second=d.execute_frame(ctx,f['record'],f['scales'],f['point'],4101,f['tx'],f['receiver'])
            self.assertEqual(first,second);self.assertEqual(f['ledger'].snapshot()['phase_charged'][d.PHASE],2)
            f['scales'][0][0]=5
            # FakeTX waveform deliberately does not encode content. Production
            # execute uses original FrameLink, whose waveform changes; ledger
            # request identity itself rejects any altered received samples.
            with self.assertRaises(d.budget.BudgetError):
                f['receiver'].receive(np.zeros((1024,2)),13,12000,phase=d.PHASE,event_prefix=first['packet_event_ids'][0][:-7])

    def test_only_tokens_loaded_from_original_two_array_archive(self):
        with tempfile.TemporaryDirectory() as td:
            td=Path(td);a=td/'source.npz';tokens=np.arange(680,dtype=np.int64)
            np.savez(a,tokens=tokens,pixels=np.zeros((3,2,2),dtype=np.uint8));cp=td/'cp.json'
            row=dict(source_index=0,source_id='s',tokens_sha256=d.token_sha(tokens),archive=str(a))
            put(cp,dict(row,outputs={str(a):d.sha(a)}));record=dict(row,checkpoint=str(cp),checkpoint_sha256=d.sha(cp))
            result=d.load_tokens(record,{str(cp):d.sha(cp),str(a):d.sha(a)})
            np.testing.assert_array_equal(np.concatenate(result),tokens)
            record['tokens_sha256']='0'*64
            with self.assertRaises(ValueError):d.load_tokens(record,{str(cp):d.sha(cp),str(a):d.sha(a)})

    def test_phase_reserve_dynamic_calls_and_no_H_borrow(self):
