class Tests(unittest.TestCase):
    def setUp(self):
        t=tempfile.TemporaryDirectory();self.addCleanup(t.cleanup);self.base=Path(t.name);self.ctx=ctx_fixture(self.base)

    def test_request_preserves_exact_predecessors_and_independent_budget(self):
        r=m.make_request(self.ctx)
        self.assertEqual(r['schema'],'MAIN_ACTUAL_CALIBRATION_REGISTRATION_REQUEST_R1')
        self.assertEqual(r['proxy_predecessor'],self.ctx['spec']);self.assertEqual(r['prescreen_predecessor'],self.ctx['prescreen_spec'])
        self.assertEqual(r['asset_predecessor'],self.ctx['asset_spec']);self.assertEqual(r['resources']['cpu_affinity'],[[24,25],[26,27]])
        self.assertEqual(r['engineering'],{'counter':'frozen'});self.assertEqual(r['python'],self.ctx['cfg']['python'])
        self.assertEqual(r['qualification_test_count'],m.TEST_COUNT);self.assertFalse(r['H_C_ledger_opened'])
        self.assertIn(str(self.ctx['loc']['q']/'completion.json'),r['input_bindings'])

    def test_runtime_order_preserves_physical_action_module_origin(self):
        e=m.environment(self.ctx);parts=e['PYTHONPATH'].split(os.pathsep)
        self.assertEqual(parts,[str(self.ctx['loc'][k])for k in ('code','oldcode','prescreen_code')])
        self.assertEqual(e['CUDA_VISIBLE_DEVICES'],'')
        self.assertTrue(all(e[k]=='2'for k in self.ctx['old'].THREAD_KEYS))
        tree=ast.parse(Path(m.__file__).read_text(encoding='utf-8'))
        self.assertFalse(any(isinstance(n,ast.Call)and isinstance(n.func,ast.Attribute)and n.func.attr=='resolve'for n in ast.walk(tree)))

    def test_changed_budget_deadline_stop_rejected_without_action(self):
        with mock.patch.object(m.time,'time',return_value=m.DEADLINE-1):m.unchanged(self.ctx)
        with mock.patch.object(m.time,'time',return_value=m.DEADLINE+1),self.assertRaises(RuntimeError):m.unchanged(self.ctx)
        self.ctx['registrar'].readonly_budget=lambda *a:{'charged':45177}
        with mock.patch.object(m.time,'time',return_value=m.DEADLINE-1),self.assertRaises(RuntimeError):m.unchanged(self.ctx)
        self.ctx['loc']['stop'].parent.mkdir(parents=True,exist_ok=True);m.save(self.ctx['loc']['stop'],{'reason':'preserve'})
        with mock.patch.object(m.time,'time',return_value=m.DEADLINE-1),self.assertRaises(RuntimeError):m.unchanged(self.ctx)

    def test_one_shot_attempt_and_save_refuse_overwrite(self):
        f=m.attempt(self.ctx,'launch')
        with self.assertRaises(RuntimeError):m.attempt(self.ctx,'launch')
        with self.assertRaises(FileExistsError):m.save(f/'attempt.json',{})

    def test_spawn_capture_failure_keeps_real_pending_pid(self):
        f=m.attempt(self.ctx,'qualify');child=SimpleNamespace(pid=123,wait=lambda:0)
        self.ctx['old'].capture=lambda *a:(_ for _ in ()).throw(RuntimeError('capture failed'))
        with mock.patch.object(m.subprocess,'Popen',return_value=child),self.assertRaises(RuntimeError):
            m.invoke(self.ctx,f,['/exact/python','-B','entry'])
        p=m.read(f/'spawn_pending_identity.json');self.assertEqual(p['pid'],123);self.assertFalse(p['identity_verified'])

    def test_launcher_parent_uses_four_core_union_only_actual_phase(self):

        pools=[n for n in ast.walk(fn)if isinstance(n,ast.Call)and isinstance(n.func,ast.Name)and n.func.id=='resources']

        self.assertIn('actual_calibration',ast.unparse(fn));self.assertNotIn('build_native',ast.unparse(tree))
        self.assertEqual(len(m.NEW_SOURCES),8);self.assertEqual(len(set(m.NEW_SOURCES)),8)
