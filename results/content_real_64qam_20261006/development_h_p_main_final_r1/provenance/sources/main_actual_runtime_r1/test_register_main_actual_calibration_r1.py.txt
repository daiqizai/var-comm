def put(path,value):
    p=Path(path);p.parent.mkdir(parents=True,exist_ok=True);p.write_text(json.dumps(value),encoding='utf8');return str(p)

def config_context(base):
    e=Path(base)/'e';request=dict(root='/repo',out=str(Path(base)/'out'),ledger='/repo/MAIN.sqlite',budget_registration='/repo/budget.json',
        profiles='/repo/profiles.json',aliases='/repo/aliases.json',legacy_runtime='/repo/old',legacy_qualification='/repo/qualification.json',
        python='/repo/ldpc_environment/bin/python',stop_file='/repo/STOP',deadline_unix=r.w.DEADLINE,
        resources=dict(cpu_affinity=[[24,25],[26,27]],max_seconds=14400))
    return dict(request=request,e=e,out=Path(request['out']),short='/repo/frozen_shortlist.json',spec={'frozen':'spec'},schedule=[{'slot':0}],
        assets=dict(manifest='/repo/assets/manifest.json',completion='/repo/assets/completion.json',calibration_registration='/repo/calibration.json'),
        sources={},inputs={},before=snapshot(0),catalogue=type('Cat',(),{'digest':'a'*64})())


class Tests(unittest.TestCase):
    def test_real_owner_config_interface_and_scope(self):
        with tempfile.TemporaryDirectory() as td:
            ctx=config_context(td);cfg,owner=r.make_configs(ctx)
            r.o.validate_shape(owner,dict(cfg,_path=str(ctx['e']/'worker_config.json')))
            self.assertEqual(cfg['engineering'],r.w.ENGINEERING);self.assertEqual(owner['cpu_affinity'],[24,25,26,27])
            self.assertFalse(owner['development_allowed']);self.assertFalse(owner['quality_selection_allowed'])
            self.assertEqual(cfg['shortlist'],'/repo/frozen_shortlist.json')

    def test_normal_prescreen_actual_parent_and_worker_required(self):
        with tempfile.TemporaryDirectory() as td:
            spec=prescreen_fixture(td)
            with mock.patch.object(r.w,'exited',return_value=True):c=r.prescreen_closed(spec,{})
            self.assertEqual(c['science']['proxy_rows'],540)
            with mock.patch.object(r.w,'exited',return_value=False),self.assertRaisesRegex(ValueError,'both exit'):r.prescreen_closed(spec,{})

    def test_prescreen_changed_output_or_failed_exit_refused(self):
        with tempfile.TemporaryDirectory() as td:
            spec=prescreen_fixture(td);cp=Path(td)/'science/shortlist.json';cp.write_text('changed')
            with mock.patch.object(r.w,'exited',return_value=True),self.assertRaises(ValueError):r.prescreen_closed(spec,{})
        with tempfile.TemporaryDirectory() as td:
            spec=prescreen_fixture(td);(Path(td)/'ops/launch/failure.json').write_text('{}')
            with mock.patch.object(r.w,'exited',return_value=True),self.assertRaisesRegex(ValueError,'failure'):r.prescreen_closed(spec,{})

    def test_real1000_asset_metadata_only_closure(self):
        with tempfile.TemporaryDirectory() as td:
            spec=assets_fixture(td)
            with mock.patch.object(r.w,'exited',return_value=True):ctx=r.assets_closed(spec,{})
            self.assertEqual(len(ctx['records']),1000);self.assertTrue(ctx['records'][-1]['archive'] in ctx['bindings'])

    def test_asset_incomplete_or_changed_original_source_refused(self):
        with tempfile.TemporaryDirectory() as td:
            spec=assets_fixture(td);(Path(td)/'assets/source_checkpoints/0999.json').write_text('{}')
            with mock.patch.object(r.w,'exited',return_value=True),self.assertRaisesRegex(ValueError,'checkpoint'):r.assets_closed(spec,{})

    def test_registration_seal_only_after_new_owner_preflight(self):
        with tempfile.TemporaryDirectory() as td:
            ctx=config_context(td)
            original=r.sha
            with mock.patch.object(r,'context',return_value=ctx),mock.patch.object(r,'readonly_budget',return_value=ctx['before']),\
                 mock.patch.object(r,'verify'),mock.patch.object(r,'sha',wraps=r.sha) as digest,\
                 mock.patch.object(r.o,'load') as load:
                # Only the external frozen budget file is synthetic here.
                digest.side_effect=lambda p:'b'*64 if str(p)=='/repo/budget.json' else original(p)
                result=r.register('not-read-in-this-isolated-test')
            self.assertEqual(result['status'],r.REGISTERED);load.assert_called_once()
            reg=r.read(ctx['e']/'execution_registration.json');self.assertFalse(reg['H_development_reserve_touched'])
            self.assertEqual(reg['phase_packet_cap'],60000);self.assertFalse(result['workers_started'])

    def test_failed_registration_preserves_failure_without_launch(self):
        with tempfile.TemporaryDirectory() as td:
            ctx=config_context(td)
            with mock.patch.object(r,'context',return_value=ctx),mock.patch.object(r,'make_configs',side_effect=ValueError('bad shape')):
                with self.assertRaises(ValueError):r.register('unused')
            self.assertTrue((ctx['e']/'registration_failure.json').exists());self.assertFalse((ctx['e']/'registration_completion.json').exists())

    def test_wrong_scheme_or_nonlinux_cannot_create_registration(self):
        with tempfile.TemporaryDirectory() as td:
            request=put(Path(td)/'request.json',{'schema':'wrong'})
            with self.assertRaisesRegex(ValueError,'Linux calibration'):r.context(request)


if __name__=='__main__':unittest.main()
