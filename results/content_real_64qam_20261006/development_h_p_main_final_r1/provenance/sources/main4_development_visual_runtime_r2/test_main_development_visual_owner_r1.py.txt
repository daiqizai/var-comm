            fcntl=SimpleNamespace(LOCK_EX=2,LOCK_NB=4,flock=lambda *a:None)
            with mock.patch.object(o.sys,'platform','linux'),mock.patch.object(o,'load',return_value=c),\
                 mock.patch.object(o,'identity',return_value=parent),mock.patch.object(o,'visual_lock',lock),\
