def put(p,v):
    p=Path(p);p.parent.mkdir(parents=True,exist_ok=True);p.write_text(json.dumps(v),encoding='utf-8');return str(p)
def ident(v):return hashlib.sha256(json.dumps(v,sort_keys=True,separators=(',',':'),allow_nan=False).encode()).hexdigest()

class DriverTests(unittest.TestCase):
    def setUp(self):
        self.tmp=tempfile.TemporaryDirectory();self.addCleanup(self.tmp.cleanup);self.root=Path(self.tmp.name)

    def history(self):
        ids=[f'dev{i}' for i in range(100)];spec={k:str(self.root/(k+'.json')) for k in ('config','registration','completion','owner_completion')}
        cfg=dict(schema='H_DEVELOPMENT_METRICS_CONFIG_V1',registration=spec['registration'])
        put(spec['config'],cfg);put(spec['registration'],{'input_bindings':d.bind([spec['config']])})
        metadata={'original_B1_metric':'sealed','precision':'float32'}
        done=dict(status='H_DEVELOPMENT_METRICS_COMPLETE',registration_sha256=d.sha(spec['registration']),config_sha256=d.sha(spec['config']),
            source_count=100,frame_count=5400,source_ids=ids,noise_seeds=[6201,6202,6203],metric_batch_size=1,new_packet_decodes=0,
            policy_selection=False,metric_metadata=metadata,metric_evaluator_identity=ident(metadata))
        put(spec['completion'],done);put(spec['owner_completion'],dict(status='REGISTERED_H_STAGE_BATCH_COMPLETE',
            registration_sha256=d.sha(spec['registration']),budget=dict(charged=164760,unresolved=0,failed=0)))
        pins=d.bind(spec.values());return spec,dict(source_ids=ids,predecessor=dict(bindings=pins)),pins

    def test_static_history_does_not_require_current_H164760(self):
        spec,cc,pins=self.history();cc['current_H_charged']=167160
        got=d.historical_metrics(spec,cc,pins)
        self.assertEqual(got['done']['frame_count'],5400)
        self.assertEqual(cc['current_H_charged'],167160)
        # No callable old H closed_batch/budget module is supplied or imported.
        self.assertEqual(set(got),{'config','registration','done'})

    def test_history_must_be_predecessor_sealed_and_normal(self):
        spec,cc,pins=self.history();wrong=deepcopy(cc);wrong['predecessor']['bindings'].pop(spec['completion'])
        with self.assertRaisesRegex(RuntimeError,'predecessor seal'):d.historical_metrics(spec,wrong,pins)
        v=d.read(spec['owner_completion']);v['status']='FAILED';put(spec['owner_completion'],v)
        pins=d.bind(spec.values());cc['predecessor']['bindings']=pins
        with self.assertRaisesRegex(RuntimeError,'normally complete'):d.historical_metrics(spec,cc,pins)
        v['status']='REGISTERED_H_STAGE_BATCH_COMPLETE';put(spec['owner_completion'],v)
        doc=d.read(spec['completion']);doc['metric_metadata']['precision']='changed';put(spec['completion'],doc)
        pins=d.bind(spec.values());cc['predecessor']['bindings']=pins
        with self.assertRaisesRegex(RuntimeError,'metadata identity'):d.historical_metrics(spec,cc,pins)

    def source(self):
        pixels=np.arange(3*256*256,dtype=np.uint8).reshape(3,256,256);pre=hashlib.sha256(pixels.tobytes()).hexdigest()
        archive=self.root/'assets/0001.npz';archive.parent.mkdir();np.savez(archive,tokens=np.zeros(680,np.int64),pixels=pixels)
        ap=self.root/'assets/0001.json';asset=dict(source_index=1,source_id='dev1',preprocessing_id=pre,evaluation_class_index=17,
            archive=str(archive),outputs=d.bind([archive]));put(ap,asset)
        rec=dict(asset,checkpoint=str(ap));cp=self.root/'cpu/worker_1/source_checkpoints/0001.json';tp=self.root/'cpu/worker_1/traces/0001.json'
        rows=[{'synthetic_frame':i}for i in range(12)];put(tp,dict(source_index=1,source_id='dev1',registration_sha256='a'*64,frames=rows))
        put(cp,dict(source_index=1,source_id='dev1',outputs=d.bind([tp])))
        calls=[];core=SimpleNamespace(validate_frames=lambda *args:calls.append(args))
        cc=dict(records=[None,rec],cfg={'out':str(self.root/'cpu')},regsha='a'*64,context={'schedule':['sealed']})
        ctx=dict(cpu_ctx=cc,bound=d.bind([cp,tp,ap,archive]),core=core)
        return ctx,rec,pixels,calls

    def test_original_development_pixels_exact_source_label_no_tokens(self):
        ctx,rec,pixels,calls=self.source();original=np.load;access=[]
        class Guard:
            def __init__(self,*a,**kw):self.z=original(*a,**kw);self.files=self.z.files
            def __enter__(self):return self
            def __exit__(self,*_):self.z.close()
            def __getitem__(self,k):
                access.append(k)
                if k!='pixels':raise AssertionError('TX tokens must not enter image worker')
                return self.z[k]
        with mock.patch.object(d.np,'load',Guard):record,frames,target,pins=d.source_inputs(ctx,1)
        self.assertEqual(access,['pixels']);self.assertEqual(record['class_index'],17);self.assertEqual(len(frames),12)
        self.assertTrue(np.array_equal(target,pixels));self.assertEqual(len(calls),1);self.assertEqual(len(pins),4)
        wrong=d.read(rec['checkpoint']);wrong['evaluation_class_index']=18;put(rec['checkpoint'],wrong)
        ctx['bound'].update(d.bind([rec['checkpoint']]))
        with self.assertRaisesRegex(RuntimeError,'label provenance'):d.source_inputs(ctx,1)

    def test_write_source_preserves_float32_and_all_twelve_logical_rows(self):
        images=np.stack([np.full((3,256,256),x,np.float32) for x in (.5,.125)])
        rows=[dict(source_index=0,source_id='dev0',image_slot=i%2,psnr_db=17.1,legacy_quality_psnr_db=17.099999)for i in range(12)]
        rec=dict(source_index=0,source_id='dev0',preprocessing_id='b'*64);out=self.root/'visual'
        saved,outputs,size=d.write_source(out,rec,rows,images,{'original':'a'*64},{'unique_images':2},'c'*64)
        self.assertEqual(len(saved),12);self.assertEqual(len(outputs),3);self.assertGreater(size,images.nbytes)
        with np.load(saved[0]['image_archive'],allow_pickle=False) as z:
            self.assertEqual(z.files,['images']);self.assertEqual(z['images'].dtype,np.dtype('float32'));self.assertTrue(np.array_equal(z['images'],images))
        cp=d.read(out/'source_checkpoints/0000.json');self.assertEqual(cp['frame_count'],12);self.assertFalse(cp['policy_selection'])
        self.assertEqual(rows[0].keys(),{'source_index','source_id','image_slot','psnr_db','legacy_quality_psnr_db'})
        with self.assertRaisesRegex(RuntimeError,'Preserve prior'):d.write_source(out,rec,rows,images,{}, {},'c'*64)

    def test_model_build_uses_original_B1_constructor_and_canonical_identity(self):
        files={k:put(self.root/(k+'.json'),{})for k in ('suite_module','replay_module','validation_module','convnext_weights')}
        policy=put(self.root/'policy.json',{'frozen_H_policy_for_classifier_provenance_only':True})
        cfg=dict(runtime_dir=str(self.root),uep_runtime=str(self.root),root=str(self.root),native_runtime=str(self.root),
            quality_driver_module='quality',source_driver_module='source',legacy_phy_module='phy',metric_policy=policy)
        native=SimpleNamespace(loaded={'identity':{'visual':'original'}},flags={'original':6},driver_bindings={'source':'pin'},torch='original-native-torch')
        metadata={'preprocess':(1,2,3),'batch':1};backend=SimpleNamespace(identity=ident(metadata),metadata=metadata);seen=[]
        interop=[]
        @contextmanager
        def repeated(torch):
            interop.append(torch);interop.append('enter');yield;interop.append('exit')
        def load_suite(*a):
            self.assertEqual(interop,['original-native-torch','enter'])
            return (1,2,3,4,5,{}, {'metric':6})
        def classifier(*a,**kw):seen.append((a,kw));return 'original_classifier'
        metric=SimpleNamespace(CONVNEXT_SHA='w'*64,SuiteBackend=lambda *a:backend,frozen_mismatch=lambda *a:{'original':True})
        modules={'quality':SimpleNamespace(build_native=lambda *a:native,verify_repeated_interop=repeated),'source':'source','phy':'phy',
            files['suite_module']:SimpleNamespace(load_suite=load_suite,numeric_flags='numeric'),
            files['validation_module']:SimpleNamespace(ConvNeXtValidation=classifier),files['replay_module']:'original_replay'}
        ctx=dict(cfg=cfg,bound={},metric_assets={'paths':files},native_identity=native.loaded['identity'],expected_flags=native.flags,
            static_bindings=native.driver_bindings,metric_flags={'metric':6},metric=metric,cpu_ctx={'source_ids':['fixture']},
            historical_metrics={'done':{'metric_evaluator_identity':backend.identity,'metric_metadata':d.jsonable(metadata)}})
        with mock.patch.object(d,'module',side_effect=lambda p,*a:modules[p]),mock.patch.dict('sys.modules',{'torch':SimpleNamespace()}):
            got=d.build_models(ctx)
        self.assertIs(got['backend'],backend);self.assertEqual(seen[0][1]['stage'],'development')
        self.assertEqual(seen[0][1]['policy_path'],policy);self.assertEqual(seen[0][1]['device'],'cuda:0')
        self.assertEqual(d.jsonable(metadata)['preprocess'],[1,2,3])
        self.assertEqual(interop,['original-native-torch','enter','exit'])
