def synthetic_rows():
    scope=dict(source_indices=list(fixture.t.FIXED),source_ids=['s'+str(i) for i in fixture.t.FIXED],slots=[18,19,20,21])
    rows=[]
    for i in scope['source_indices']:
        for slot in scope['slots']:
            reps=[]
            for phase,n in [('warmup',0),('measured',0),('measured',1),('measured',2)]:
                reps.append(dict(phase=phase,repetition=n,fingerprint='same',new_packet_decodes=0,
                    seconds={'RX_source_and_image_total':.01},component_calls={'RX_source_and_image_total':1},receiver_summary={'image_sha256':'rgb'}))
            rx=dict(status='MEASURED_UNCACHED_COMPONENTS',warmups=reps[:1],measured=reps[1:],warmup_repetitions=1,measured_repetitions=3)
            tx=deepcopy(rx)
            for r in tx['warmups']+tx['measured']:r['seconds']={'TX_visual_encoding':.004,'TX_source_total':.001};r['component_calls']={'TX_visual_encoding':1,'TX_source_total':1}
            rows.append(dict(source_index=i,source_id='s'+str(i),development_slot=slot,noise_seed=6201,image_sha256='rgb',gray=False,TX=tx,RX=rx))
    return scope,rows
