def require(ok,message):
    if not ok:raise RuntimeError(message)


def read(path):return json.loads(Path(path).read_text(encoding='utf-8-sig'))
def sha(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as f:
        for block in iter(lambda:f.read(1048576),b''):h.update(block)
    return h.hexdigest()


def save(path,value):
    path=Path(path);path.parent.mkdir(parents=True,exist_ok=True)
    with path.open('x',encoding='utf-8',newline='\n') as f:
        json.dump(value,f,ensure_ascii=False,sort_keys=True,indent=2,allow_nan=False)
        f.write('\n');f.flush();os.fsync(f.fileno())


def verify(mapping):
    for p,s in mapping.items():require(Path(p).is_absolute() and sha(p)==s,'Bound file changed: '+p)


def combine(*maps):
    result={}
    for values in maps:
        for p,s in values.items():
            require(p not in result or result[p]==s,'Conflicting binding: '+p);result[p]=s
    return result


def bind(paths):return {str(Path(p).absolute()):sha(p) for p in paths}
def module(path,name):
    spec=importlib.util.spec_from_file_location(name,path);m=importlib.util.module_from_spec(spec)
    sys.modules[name]=m;spec.loader.exec_module(m);return m

def invoke(ctx,folder,argv,admit=None):
    """Never inspect a child as live after poll reports exit; wait then bind log."""
    owner=ctx['owner'];p=None;identity=None;failed=False;began=time.monotonic();error=None;handlers={};rc=None
    log=(folder/'process.log').open('xb')
    def request_stop(*_):
        nonlocal failed
        failed=True;stop_path=Path(ctx['request']['execution_dir'])/'STOP'
        if not stop_path.exists():save(stop_path,dict(status='SELECTION_STOP_PRESERVE',pid=p.pid if p is not None else None))
    for n in (signal.SIGTERM,signal.SIGINT):handlers[n]=signal.signal(n,request_stop)
    try:
        p=subprocess.Popen(argv,cwd=ctx['request']['runtime_dirs'][0],env=environment(ctx),stdin=subprocess.DEVNULL,
            stdout=log,stderr=subprocess.STDOUT,preexec_fn=resources)
        save(folder/'spawn_pending_identity.json',dict(pid=p.pid,argv=argv,identity_verified=False))
        for _ in range(100):
            require(p.poll() is None,'Child exited before verified identity capture')
            try:
                identity=owner.ids.process_identity(p.pid)
                if identity['argv']==argv:break
            except (FileNotFoundError,ProcessLookupError):pass
            time.sleep(.01)
        require(identity is not None and identity['argv']==argv,'Spawned child argv differs')
        save(folder/'launch.json',dict(identity=identity,argv=argv,threads=2,cpu_affinity=CPU,nice=15))
        if admit is not None:admit(identity)
        while p.poll() is None:
            try:
                boundary(ctx,budget=False)
                require(time.monotonic()-began<ctx['request']['resources']['max_seconds'],'CPU wall cap reached')
            except BaseException:
                request_stop()
            time.sleep(.2)
        rc=p.wait()
    except BaseException:
        error=sys.exc_info()
        if p is not None:
            request_stop();rc=p.wait()
    finally:
        log.close()
        for n,h in handlers.items():signal.signal(n,h)
    if p is not None:save(folder/'exit.json',dict(identity=identity,exit_code=rc,wait_completed=True,
        launch_sha256=sha(folder/'launch.json') if (folder/'launch.json').exists() else None,log_sha256=sha(folder/'process.log')))
    if error is not None:raise error[1].with_traceback(error[2])
    require(not failed and rc==0 and owner.exited(identity),'Child must wait/exit0; failure evidence retained')
    return bind([folder/n for n in ('spawn_pending_identity.json','launch.json','exit.json','process.log')])

        and not re.search(r'\bskipped\b',log,re.I),'Exact pure test inventory with no skips required')
    boundary(ctx)
    value=dict(status=QUALIFIED,python=ctx['python'],test_modules=TESTS,test_count=TEST_COUNT,
        source_bindings=ctx['sources'],input_bindings=ctx['inputs'],outputs=outputs,budget_before=ctx['before'],
        budget_after=ctx['before'],actual_quality_rows_read=False,new_packet_decodes=0,new_metric_calls=0,GPU_used=False)
    save(q/'completion.json',value);return value

def streamed_rows(ctx,checkpoint=lambda:None):
    """Per-source30 rows; reproduce the original full-inventory JSON byte hash."""
    vc=ctx['visual'];done=ctx['closed']['done'];out=Path(vc['cfg']['out']);count=0;digest=hashlib.sha256(b'[\n')
    expected=done['outputs'][str(out/'frame_metrics.json')]
    for index,rec in enumerate(ctx['cpu']['records']):
        checkpoint();cp=out/'source_checkpoints'/f'{index:04d}.json';rp=out/'sources'/f'{index:04d}.json'
        archive=out/'images'/f'{index:04d}.npz'
        for p in (cp,rp):require(done['outputs'].get(str(p))==sha(p),'Source checkpoint/rows differ from completed visual')
        c=read(cp)
        require(c['status']==ctx['owner'].w.SOURCE_DONE and c['registration_sha256']==vc['regsha']
            and c['source_index']==index and c['source_id']==rec['source_id'] and c['frame_count']==30
            and c['noise_seeds']==[4101,4102,4103] and c['new_packet_decodes']==0
            and c['source_decode_complete'] is True and c['image_reconstruction_complete'] is True
            and c['images_scored'] is True and c['primary_metric']=='dinov2_vitl14_cosine','Actual visual source receipt differs')
        require(set(c['outputs'])=={str(rp),str(archive)} and all(done['outputs'].get(p)==s for p,s in c['outputs'].items())
            and all(ctx['closed']['bindings'].get(p)==s for p,s in c['input_bindings'].items()), 'Original per-source evidence chain differs')
        rows=read(rp);ctx['owner'].w.validate_grid(rows,index,rec['source_id'],ctx['cpu']['schedule'])
        for row in rows:
            require(row['image_archive']==str(archive) and type(row['image_slot']) is int and row['image_slot']>=0,
                'Actual image archive/slot differs')
            if count:digest.update(b',\n')
            digest.update(json.dumps(row,ensure_ascii=False,allow_nan=False).encode('utf-8'));count+=1
            yield row
    digest.update(b'\n]\n')
    require(count==30000 and digest.hexdigest()==expected,'Full row inventory not identical to the sealed per-source rows')

        require(ctx['owner'].ids.process_identity(os.getppid())==parent,'Parent owner changed')
    def signal_stop(*_):raise RuntimeError('Signal at CPU selection; preserve evidence')
    signal.signal(signal.SIGTERM,signal_stop);signal.signal(signal.SIGINT,signal_stop)
    try:
        result=select(ctx,streamed_rows(ctx,check));check();boundary(ctx);outputs={}
        for name,value in (('selection.json',result),('family_selected_profiles.json',result['family_selected_profiles']),
            ('H_reference_profiles.json',result['H_reference_profiles']),('candidate_summaries.json',result['candidate_summaries']),

        done=dict(status=DONE,registration_sha256=regsha,config_sha256=cfgsha,outputs=outputs,
            source_bindings=reg['source_bindings'],input_bindings=reg['input_bindings'],worker_identity=me,supervision=admission,
            source_count=1000,frame_count=30000,noise_seeds=[4101,4102,4103],unique_candidate_snr_count=10,
            family_selected_count=4,H_reference_count=4,primary='dinov2_vitl14_cosine',selection_scope=result['selection_scope'],
            active_SNRs=[13,19],deferred_SNRs=[1,4,7,10],budget_before=ctx['before'],budget_after=ctx['before'],
            new_packet_decodes=0,new_metric_calls=0,noise_generated=0,GPU_used=False,development_used=False,holdout_used=False,
            shortlist_reselected=False,main_six_SNR_complete=False,automatic_successor_started=False)
        boundary(ctx);save(out/'completion.json',done);return done

def launch(ctx,folder):
    qualified(ctx);reg,regsha,cfgsha=load_registered(ctx);r=ctx['request'];e=Path(r['execution_dir'])

    import fcntl
    with Path(ctx['cpu']['cfg']['ledger']).with_suffix('.owner.lock').open('a+') as lock:
        fcntl.flock(lock,fcntl.LOCK_EX|fcntl.LOCK_NB);boundary(ctx)
        parent=ctx['owner'].ids.process_identity(os.getpid());require(parent['argv']==command(ctx,'launch'),'Exact CPU owner command required')

        require(done['supervision']==read(e/'worker_admission.json'),'Worker used a different admission');boundary(ctx)
        outputs=combine(outputs,done['outputs'],bind([Path(r['out'])/'completion.json',e/'owner_start.json',e/'worker_admission.json']))
        value=dict(status=CLOSED,registration_sha256=regsha,config_sha256=cfgsha,owner_identity=parent,
            outputs=outputs,budget_before=ctx['before'],budget_after=ctx['before'],source_count=1000,frame_count=30000,

    launch_rec=read(folder/'launch.json');end=read(folder/'exit.json');parent=start['identity'];child=end['identity']
    require(done['status']==CLOSED and done['registration_sha256']==regsha and done['config_sha256']==cfgsha
        and done['owner_identity']==parent and parent['argv']==command(ctx,'launch') and ctx['owner'].exited(parent)
        and start['registration_sha256']==regsha and start['config_sha256']==cfgsha,'Normal owner exit required')
    require(end['identity']==launch_rec['identity'] and child['argv']==command(ctx,'worker') and end['exit_code']==0
        and end['wait_completed'] is True and end['launch_sha256']==sha(folder/'launch.json')
        and end['log_sha256']==sha(folder/'process.log') and ctx['owner'].exited(child),'Waited worker exit0 and closed log required')
    science=science_receipt(ctx,regsha,cfgsha,child);a=read(e/'worker_admission.json')
    require(science['supervision']==a and a['identity']==child and a['parent_identity']==parent
        and a['registration_sha256']==regsha and a['config_sha256']==cfgsha
        and done['budget_before']==done['budget_after']==ctx['before'],'Normal selection supervision/budget differs')
    return dict(ctx=ctx,done=science,owner_done=done,bindings=combine(reg['source_bindings'],reg['input_bindings'],
        done['outputs'],bind([e/'completion.json',e/'execution_registration.json',e/'config.json',e/'registration_completion.json'])))


def main():
    p=argparse.ArgumentParser();p.add_argument('--request',required=True)
    p.add_argument('--stage',choices=('qualify','register','launch','worker'),required=True);a=p.parse_args()
    require(sys.platform.startswith('linux'),'Authentic Linux process identity required for operations')
    folder=None
    try:
        if a.stage!='worker':folder=operation(a.request,a.stage)
        ctx=context(a.request)
        result=worker(ctx) if a.stage=='worker' else globals()[a.stage](ctx,folder)
        print(result['status'],flush=True)
    except BaseException:
        if folder is not None:save(folder/'failure.json',dict(traceback=traceback.format_exc(),new_packet_decodes=0,GPU_used=False))
        raise


if __name__=='__main__':main()
