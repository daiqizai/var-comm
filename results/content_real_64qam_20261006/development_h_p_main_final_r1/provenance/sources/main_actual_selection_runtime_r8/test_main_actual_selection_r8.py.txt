def synthetic_stream(root):
    out=Path(root)/'visual';out.mkdir();data={};pins={};records=[];schedule=[]
    for snr in (13,19):
        for i in range(5):schedule.append(dict(slot=len(schedule),candidate_id=str(i),snr_db=snr))
    writer=visual.FrameWriter(out)
    for index in range(1000):
        sid=f'source-{index}';records.append(dict(source_id=sid,source_index=index));rows=[]
        archive=out/'images'/f'{index:04d}.npz';rp=out/'sources'/f'{index:04d}.json';cp=out/'source_checkpoints'/f'{index:04d}.json'
        for point in schedule:
            for j,seed in enumerate((4101,4102,4103)):
                rows.append(dict(point,source_index=index,source_id=sid,noise_seed=seed,population_role='calibration',
                    public_frame_counter=((4 if point['snr_db']==13 else 5)*1000+index)*3+j,
                    image_archive=str(archive),image_slot=0))
        writer.append(rows);data[str(rp)]=rows;pins[str(rp)]=hashlib.sha256(json.dumps(rows).encode()).hexdigest()
        pins[str(archive)]='a'*64
        data[str(cp)]=dict(status=visual.SOURCE_DONE,registration_sha256='b'*64,source_index=index,source_id=sid,
            frame_count=30,noise_seeds=[4101,4102,4103],new_packet_decodes=0,source_decode_complete=True,
            image_reconstruction_complete=True,images_scored=True,primary_metric=visual.PRIMARY,input_bindings={},
            outputs={str(rp):pins[str(rp)],str(archive):pins[str(archive)]})
        pins[str(cp)]=hashlib.sha256(json.dumps(data[str(cp)]).encode()).hexdigest()
    final=writer.finish(30000);pins[str(final)]=p.sha(final)
    ctx=dict(visual=dict(cfg={'out':str(out)},regsha='b'*64),cpu=dict(records=records,schedule=schedule),
        closed=dict(done={'outputs':pins},bindings={}),owner=SimpleNamespace(w=visual))
    return ctx,data,pins

    def test_only_r8_paths_and_original_deadline_resource_contract(self):
        with tempfile.TemporaryDirectory() as td:
            r=request(Path(td));p.validate_request(r)
            bad=copy.deepcopy(r);bad['visual_batch']['completion']=bad['visual_batch']['completion'].replace('_r8','_r2')
            with self.assertRaisesRegex(RuntimeError,'R8'):p.validate_request(bad)
            bad=copy.deepcopy(r);bad['visual_registration']['path']='unsealed'
            with self.assertRaisesRegex(RuntimeError,'R8'):p.validate_request(bad)
            bad=copy.deepcopy(r);bad['out']=bad['execution_dir']
            with self.assertRaisesRegex(RuntimeError,'disjoint'):p.validate_request(bad)
            # An actually registered but not completed visual stage cannot reach
            # imported predecessor validation, score reading, or registration.
            op=Path(r['visual_batch']['owner_config']);rp=Path(r['visual_registration']['path'])
            p.save(op,{'registration':str(rp)})
            p.save(rp,dict(owner_config_sha256=p.sha(op),source_bindings={r['visual_owner_module']:p.OWNER_SHA}))
            r['visual_registration']['sha256']=p.sha(rp);requestpath=Path(td)/'request.json';p.save(requestpath,r)
            actual_sha=p.sha
            with mock.patch.object(p,'sha',side_effect=lambda x:p.OWNER_SHA if str(x)==r['visual_owner_module']else actual_sha(x)),\
                 mock.patch.object(p,'module',side_effect=AssertionError('No predecessor import before normal completion')),\
                 self.assertRaisesRegex(RuntimeError,'Normal R8 visual owner completion'):
                p.context(requestpath)
        metadata=Path(os.environ.get('MAIN_SELECTION_R8_VISUAL_METADATA',Path(__file__).absolute().parent.parent/'current/main_r8'))
        oc,vc,vr,seal=[p.read(metadata/n)for n in ('owner_config.json','worker_config.json','execution_registration.json','registration_completion.json')]
        self.assertEqual(vr['owner_config_sha256'],p.sha(metadata/'owner_config.json'))
        self.assertEqual(vr['config_sha256'],p.sha(metadata/'worker_config.json'))
        self.assertEqual(seal['registration_sha256'],p.sha(metadata/'execution_registration.json'))
        self.assertEqual(oc['registration'],vc['registration']);self.assertTrue(oc['owner_out'].endswith('/actual_visual_execution_r8'))
        self.assertTrue(vc['runtime_dir'].endswith('/main_actual_visual_runtime_r7'))
        self.assertEqual(vr['source_bindings'][vc['visual_owner_module']],p.OWNER_SHA)
        self.assertEqual(vr['source_bindings'][vc['worker_module']],'a27820aac863cc970a73441ac35bb35b526bf42fdae6d4d0fb5ffbea268ffa18')
        self.assertEqual(vr['budget_before']['charged'],105176)

    def test_original_selector_stream_and_science_logic_are_unchanged(self):
        original=Path(os.environ.get('MAIN_SELECTION_R8_ORIGINAL_SOURCE',str(Path(__file__).with_name('main_actual_selection_r1.py'))))
        self.assertEqual(p.sha(original),'07e66831f2a78a19e952785bd118b7967e98d8ff80a0fad194c0c984ffe59cdd')
        old=ast.parse(original.read_text(encoding='utf-8'));new=ast.parse(Path(p.__file__).read_text(encoding='utf-8'))
        for name in ('select','streamed_rows','worker','science_receipt','launch','closed_batch'):
            before=next(n for n in old.body if isinstance(n,ast.FunctionDef) and n.name==name)
            after=next(n for n in new.body if isinstance(n,ast.FunctionDef) and n.name==name)
            self.assertEqual(ast.dump(before,include_attributes=False),ast.dump(after,include_attributes=False),name)
        self.assertEqual(p.SELECTOR_SHA,'aedd71d7a10774ec4d3e8f7c57704fc34c10bcb83d464277cef9e51deaec9b4b')

    def test_poll_then_wait_exit_log_closure_without_live_identity_race(self):
        with tempfile.TemporaryDirectory() as td:
            c=context(Path(td));folder=Path(td)/'process';folder.mkdir();argv=['original-UM','job'];child=mock.Mock(pid=45)
            child.poll.side_effect=[None,None,0];child.wait.return_value=0
