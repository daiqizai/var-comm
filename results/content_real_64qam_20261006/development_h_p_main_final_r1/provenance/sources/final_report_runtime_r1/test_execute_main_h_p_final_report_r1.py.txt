def emit(p,v):
    p=Path(p);p.parent.mkdir(parents=True,exist_ok=True)
    if p.suffix=='.csv':
        with p.open('w',encoding='utf-8',newline='') as f:
            names=list(v[0]) if v else ['empty'];w=csv.DictWriter(f,fieldnames=names);w.writeheader()
            for row in v:w.writerow({k:json.dumps(x) if isinstance(x,(list,dict)) else x for k,x in row.items()})
    else:p.write_text(json.dumps(v,ensure_ascii=False),encoding='utf-8')
    return str(p)

    def test_full_dataset_real_projection_and_exact_original_copies(self):

        with mock.patch.object(e,'preflight',return_value=c):e.register(rp)
        config=str(c['e']/'config.json');who=dict(pid=42,start_ticks=100,uid=2,argv=[r['python'],'-B',str(Path(e.__file__).absolute()),'--stage','run','--config',config])
        env={k:'2' for k in ('OMP_NUM_THREADS','MKL_NUM_THREADS','OPENBLAS_NUM_THREADS','NUMEXPR_NUM_THREADS')};env['CUDA_VISIBLE_DEVICES']=''
        with mock.patch.object(e,'preflight',return_value=c),mock.patch.object(e,'identity',return_value=who),mock.patch.object(e.sys,'platform','linux'),mock.patch.object(e.os,'sched_setaffinity',create=True),mock.patch.object(e.os,'setpriority',create=True),mock.patch.object(e.os,'PRIO_PROCESS',0,create=True),mock.patch.dict(os.environ,env):

        view=e.read(c['out']/'view.json');self.assertEqual(len(view['summary']),432);self.assertEqual(len(view['paired']),1044)
        inventory=e.read(c['out']/'table_inventory.json');self.assertEqual(len(inventory),32)
        for p,v in inventory.items():self.assertEqual(Path(p).read_bytes(),Path(v['original']).read_bytes())

        launch=emit(self.root/'launch.json',dict(identity=who));end=emit(self.root/'exit.json',dict(identity=who,exit_code=0,process_waited=True,log_sha256=e.sha(log)))
        spec=dict(config=config,completion=str(c['out']/'completion.json'),launch=launch,exit=end,log=str(log))
        with mock.patch.object(e,'absent',return_value=True):
            value=e.closed_report(spec);self.assertEqual(value['receipt'],done)
            self.assertIn(str(c['e']/'execution_registration.json'),value['bindings'])

        with mock.patch.object(e,'absent',return_value=False),self.assertRaises(RuntimeError):e.closed_report(spec)

    def test_actual_historical_metadata_sparse_ancestry_and_envelope(self):
