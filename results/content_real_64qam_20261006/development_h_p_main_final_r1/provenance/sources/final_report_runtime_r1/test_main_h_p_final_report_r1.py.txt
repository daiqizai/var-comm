class Report(unittest.TestCase):
    def test_complete_projection_preserves_values_roles_and_no_mutation(self):
        data=fixture();before=copy.deepcopy(data);out=r.project(**data)
        self.assertEqual(data,before);self.assertEqual(out['summary'],data['final']['summary'])
        self.assertEqual(out['paired'],data['final']['paired']);self.assertEqual(len(out['paired']),1044)
        self.assertEqual(len(out['H_full_metric_annex']['summary']),378)
        self.assertEqual(len(out['H_full_metric_annex']['paired']),546)
        self.assertEqual([x['role'] for x in out['H_catalog']].count('H_RAW_PARTIAL_SYSTEM'),2)
        self.assertEqual(len(out['H_success']['comparisons']),36)
        self.assertFalse(out['normal_closure_verified_by_this_module']);self.assertFalse(out['new_estimates'])

    def test_missing_duplicate_new_pair_or_altered_old_values_fail(self):
        for change in ('missing','duplicate','newpair','changed_old','missing_h21'):
            data=fixture()
            if change=='missing':data['final']['summary'].pop()
            elif change=='duplicate':data['final']['paired'][-1]=data['final']['paired'][0]
            elif change=='newpair':data['final']['comparison_plan']['comparisons'][-1]=('H18_SLOT_00','H18_SLOT_08')
            elif change=='changed_old':data['final']['summary'][0]['mean']=123
            else:data['h18']['summary'].pop()
            with self.subTest(change=change),self.assertRaises(ValueError):r.project(**data)

    def test_branch_noise_precision_and_original_success_bytes_remain(self):
        data=fixture();out=r.project(**data);text=r.markdown(out)
        for phrase in ('2001–2003','float64','float32','固定 m7','最强 raw64','DINO-L','F恢复误差未测','N/A','每点100源','每点300帧',
                       'fresh Encoder/VQ另列','MAIN实际能量','header_energy','| 18 | RX | GRAY | N/A | N/A | N/A | N/A | 3/300 |'):
            self.assertIn(phrase,text)
        self.assertEqual(out['missing_P_metrics'],list(r.old.MISSING))
        data['criteria_bytes']+=b'\n'
        with self.assertRaises(ValueError):r.project(**data)
        data=fixture();data['final']['paired'][0]['reference_noise_seeds']=[6201,6202,6203]
        with self.assertRaises(ValueError):r.project(**data)

    def test_gate_per_reference_and_whole_two_snr_without_winner(self):
        data=fixture()
        for row in data['final']['paired']:
            if row['method'].startswith('H18_') and row['reference'].startswith('MAIN_') and row['metric'] in r.success.METRICS:
                row.update(mean=-.5 if row['metric']=='lpips_alex' else .5,
                    ci_low=-1 if row['metric']=='lpips_alex' else .1,ci_high=-.1 if row['metric']=='lpips_alex' else 1.)
        out=r.project(**data);self.assertTrue(all(x['status']=='PASS' for x in out['H_success']['whole_arm_two_snr_gates']))
        row=next(x for x in data['final']['paired'] if x['method']=='H18_SLOT_00' and x['reference']=='MAIN_SLOT_19' and x['metric']=='clip_image_cosine')
        row['ci_low']=-.1;out=r.project(**data)
        self.assertEqual(out['H_success']['whole_arm_two_snr_gates'][0]['status'],'UNRESOLVED')
        self.assertEqual(sum(x['status']=='UNRESOLVED' for x in out['H_success']['comparisons']),1)
        self.assertEqual(len(out['display_order']),24);self.assertFalse(out['policy_selection'])

    def test_main_gray_keep_and_different_denominators(self):
        data=fixture();out=r.project(**data)
        self.assertIsNone(out['MAIN_distributions']['points'][0]['RX_endpoints'][0]['m'])
        self.assertFalse(out['MAIN_distributions']['points'][0]['RX_endpoints'][1]['body_crc_accept'])
        for change in ('gray_m0','omit_failure','wrong_tx','invent_energy'):
            data=fixture();p=data['main_distribution']['points'][0]
            if change=='gray_m0':p['RX_endpoints'][0]['m']=0
            elif change=='omit_failure':p['RX_endpoints'].pop(0)
            elif change=='wrong_tx':p['TX_source_denominator']=300
            else:p['energy']['idle_energy']=p['energy']['total_energy']
            with self.subTest(change=change),self.assertRaises(ValueError):r.project(**data)

    def test_pending_costs_and_missing_phy_never_become_zero_or_sum(self):
        data=fixture();data['main_timing']=None;out=r.project(**data)
        self.assertEqual(out['MAIN_costs']['status'],'PENDING_NORMAL_MAIN_TIMING_CLOSURE')
        self.assertIn('等待该计时阶段正常闭合',r.markdown(out));self.assertIsNone(out['MAIN_costs']['end_to_end_latency'])
        data=fixture();data['main_timing']['component_summary']['end_to_end_latency']=0
        with self.assertRaises(ValueError):r.project(**data)
        data=fixture();data['main_timing']['completion']['timing_design']['noise_seed']=6202
        with self.assertRaises(ValueError):r.project(**data)
        data=fixture();data['h_timing']['completion']['source_ids'][0]='different-source'
        with self.assertRaises(ValueError):r.project(**data)

    def test_incomplete_science_or_bridge_cannot_be_reported_as_final(self):
        for key,val in [('status','RUNNING'),('old_H_P_source_means_preserved',False),('policy_selection',True)]:
            data=fixture();data['final']['completion'][key]=val
            with self.assertRaises(ValueError):r.project(**data)
        data=fixture();data['hp']['normalization_proof_sha256']='d'*64
        with self.assertRaises(ValueError):r.project(**data)
        data=fixture();data['final']['completion']['input_bindings'].clear()
        with self.assertRaises(ValueError):r.project(**data)
