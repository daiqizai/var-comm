    def worker_patches(self,ctx,cp,reg,parent,child):
        stack=contextlib.ExitStack()
        for target,value in [('sealed',(self.cfg,reg)),('inspect',ctx),('budget',self.before)]:stack.enter_context(mock.patch.object(d,target,return_value=value))
        stack.enter_context(mock.patch.object(d,'process',side_effect=lambda p:child if p==4 else parent))
        stack.enter_context(mock.patch.object(d.os,'getpid',return_value=4));stack.enter_context(mock.patch.object(d.os,'getppid',return_value=1))
        stack.enter_context(mock.patch.object(d.os,'sched_getaffinity',return_value={0,1},create=True))
        stack.enter_context(mock.patch.object(d.os,'getpriority',return_value=15,create=True));stack.enter_context(mock.patch.object(d.os,'PRIO_PROCESS',0,create=True))
        stack.enter_context(mock.patch.dict(d.os.environ,dict(CUDA_VISIBLE_DEVICES='',**{k:'2' for k in d.THREADS})))
        return stack

    def test_worker_full_synthetic_grid_preserves_old_mean_values(self):
        ctx,cp,reg,parent,child=self.worker_setup();original=d.load_rows
        # All eighteen fake metrics intentionally have identical source arrays.
        # Execute the original10000-draw interval for each distinct array once;
        # reuse only exact duplicate fake inputs to keep CPU qualification small.
        intervals={}
        def interval(a,samples):
            key=a.tobytes()
            if key not in intervals:intervals[key]=f.s.interval(a,samples)
            return dict(intervals[key])
        ctx['statistics']=SimpleNamespace(original100=f.s.original100,point_contract=f.s.point_contract,
            source_means=f.s.source_means,draws=f.s.draws,interval=interval)
        def load(checked):
            self.assertTrue((self.out/'comparison_plan.json').is_file());return original(checked)
        with self.worker_patches(ctx,cp,reg,parent,child),mock.patch.object(d,'load_rows',side_effect=load):result=d.worker(cp)
        self.assertEqual(result['source_mean_cells'],43200);self.assertEqual(result['paired_cells'],1044)
        d.verify(result['outputs']);self.assertFalse(result['success_gate_evaluated']);self.assertEqual(result['new_metric_calls'],0)
        with (self.out/'source_means.csv').open(newline='') as handle:got=list(csv.DictReader(handle))
        old=[r for r in got if not r['point_id'].startswith('MAIN')]
        self.assertEqual(old,[{k:str(v) for k,v in r.items()} for r in self.means])
        self.assertTrue(d.read(self.out/'provenance.json')['old_means_reused'])
        self.assertTrue(intervals);self.assertTrue(all(v['bootstrap_replicates']==10000 for v in intervals.values()))

    def test_stop_write_failure_still_drains_child_before_unlock(self):
        self.e.mkdir();cp=put(self.e/'config.json',self.cfg);put(self.cfg['registration'],{})
        reg=dict(predecessors=[],lock_paths=[str(self.root/'H.owner.lock')]);order=[];parent=dict(pid=1,uid=2,argv=d.command(self.cfg,cp,'run'))
        class Locks(contextlib.ExitStack):
            def close(self):order.append('unlock');super().close()
        class Child:
            pid=9;returncode=1
            def poll(self):return None
            def wait(self):order.append('wait');return 1
        with mock.patch.object(d,'process',side_effect=[dict(pid=9,uid=2,argv=['pre_exec_parent']),dict(pid=9,uid=2,argv=['target'])]),mock.patch.object(d.time,'sleep'):
            self.assertEqual(d.capture(Child(),['target'],2)['argv'],['target'])
        with mock.patch.object(Child,'poll',return_value=1),self.assertRaisesRegex(RuntimeError,'before identity'):
            d.capture(Child(),['target'],2)
        original=d.save
        def save(path,value):
            if Path(path).name=='STOP':order.append('stop_failed');raise OSError('marker disk error')
            return original(path,value)
        def process(pid):
            if pid==9:raise RuntimeError('identity capture failed')
            return parent
        with mock.patch.object(d.sys,'platform','linux'),mock.patch.object(d,'sealed',return_value=(self.cfg,reg)),\
             mock.patch.object(d,'process',side_effect=process),mock.patch.object(d,'save',side_effect=save),\
             mock.patch.object(d.contextlib,'ExitStack',Locks),mock.patch.object(d.subprocess,'Popen',return_value=Child()),\
             mock.patch.object(d.os,'sched_setaffinity',create=True),mock.patch.object(d.os,'setpriority',create=True),\
             mock.patch.object(d.os,'PRIO_PROCESS',0,create=True),mock.patch.dict(sys.modules,{'fcntl':SimpleNamespace(flock=lambda *a:None,LOCK_EX=1,LOCK_NB=2)}),\
             self.assertRaisesRegex(RuntimeError,'identity capture failed'):d.run(cp)
        self.assertEqual(order,['stop_failed','wait','unlock']);self.assertTrue(d.read(self.e/'worker_exit.json')['wait_completed'])
        self.assertIn('STOP write failed',d.read(self.e/'failure.json')['traceback']);self.assertFalse((self.e/'completion.json').exists())
