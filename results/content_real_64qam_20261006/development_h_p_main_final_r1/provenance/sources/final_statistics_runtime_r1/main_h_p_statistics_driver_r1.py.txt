def require(ok,message):
    if not ok:raise RuntimeError(message)
def read(p):return json.loads(Path(p).read_text(encoding='utf-8-sig'))
def sha(p):
    h=hashlib.sha256()
    with Path(p).open('rb') as f:
        for b in iter(lambda:f.read(1048576),b''):h.update(b)
    return h.hexdigest()
def identity(v):return hashlib.sha256(json.dumps(v,sort_keys=True,separators=(',',':'),allow_nan=False).encode()).hexdigest()
def bind(paths):return {str(Path(p).absolute()):sha(p) for p in paths}
def verify(pins):
    for p,h in pins.items():require(Path(p).is_absolute() and sha(p)==h,'Changed bound input: '+p)
def combine(*maps):
    result={}
    for mapping in maps:
        for p,h in mapping.items():require(p not in result or result[p]==h,'Conflicting SHA: '+p);result[p]=h
    return result
def save(p,v):
    p=Path(p);p.parent.mkdir(parents=True,exist_ok=True)
    with p.open('x',encoding='utf-8',newline='\n') as f:json.dump(v,f,sort_keys=True,indent=2,allow_nan=False);f.write('\n');f.flush();os.fsync(f.fileno())
def module(p,name,bound):
    p=str(Path(p).absolute());require(bound.get(p)==sha(p),'Source must be pinned before import: '+p)
    s=importlib.util.spec_from_file_location(name,p);m=importlib.util.module_from_spec(s);sys.modules[name]=m;s.loader.exec_module(m);return m
def process(pid):
    p=Path('/proc')/str(pid);f=(p/'stat').read_text().rsplit(')',1)[1].split()
    return dict(pid=pid,start_ticks=int(f[19]),uid=p.stat().st_uid,argv=(p/'cmdline').read_bytes().decode().rstrip('\0').split('\0'))
def absent(old):
    try:now=process(old['pid'])
    except FileNotFoundError:return True
    return any(now[k]!=old[k] for k in ('start_ticks','uid'))
def waited(child):
    while True:
        try:return child.wait()
        except KeyboardInterrupt:continue
def capture(child,argv,uid):
    until=time.monotonic()+10
    while time.monotonic()<until:
        require(child.poll() is None,'CPU child exited before identity admission')
        try:
            candidate=process(child.pid)
            if candidate['argv']==argv and candidate['uid']==uid:return candidate
        except FileNotFoundError:pass
        time.sleep(.05)
    raise RuntimeError('CPU child command did not settle to the registered interpreter')
def stop(*_):
    global STOP
    STOP=True

def historical_hp(spec,visual):

    require(set(spec)=={'config','registration','owner_completion','completion'},'Exact four historical H/P control pins required')
    cc=visual['ctx']['cpu_ctx'];sealed=cc['predecessor']['bindings']
    for key in ('registration','owner_completion','completion'):
        p=spec[key];require(sealed.get(p)==sha(p),'H/P receipt absent from normal MAIN4 predecessor snapshot: '+p)
    # The normal CPU snapshot seals the original registration itself. Its
    # immutable input map authenticates the old worker config and private
    # admission/plan paths even when they were not flattened into that snapshot.
    reg=read(spec['registration']);ancestry=combine(sealed,reg['source_bindings'],reg['input_bindings'])
    require(ancestry.get(spec['config'])==sha(spec['config']),'H/P worker config is not bound by the sealed original registration')
    cfg,owner,done=[read(spec[k]) for k in ('config','owner_completion','completion')]
    require(cfg['schema']=='H_P_FIXED_STATISTICS_CONFIG_V1' and cfg['registration']==spec['registration']
        and reg['input_bindings'].get(spec['config'])==sha(spec['config'])
        and done['status']=='H_P_FIXED_SOURCE_COMPARISON_COMPLETE_MAIN_PENDING'
        and done['registration_sha256']==sha(spec['registration']) and done['config_sha256']==sha(spec['config']),
        'Original H/P statistics identity differs')
    require(owner['status']=='REGISTERED_H_STAGE_BATCH_COMPLETE' and owner['registration_sha256']==sha(spec['registration'])
        and owner['budget']==done['budget_before']==done['budget_after'] and owner['budget']['charged']==164760,
        'Historical H/P successful owner budget proof differs')
    require(done['source_ids']==cc['source_ids'] and done['source_count']==100 and done['comparison_count']==18
        and done['admitted_metric_count']==18 and done['missing_P_metric_count']==3
        and done['bootstrap_seed']==2026100605 and done['bootstrap_replicates']==10000
        and done['policy_selection'] is False and done['new_packet_decodes']==done['new_metric_calls']==0,
        'Original fixed H18/P2 statistics scope differs')
    old=combine(reg['source_bindings'],reg['input_bindings'],done['input_bindings'],done['source_bindings'],done['outputs'])
    paths={name:str(Path(cfg['out'])/name) for name in HP_FILES}
    paths.update(admission=cfg['admission'],comparison_plan=cfg['comparison_plan'])
    for p in paths.values():require(old.get(p)==sha(p) and ancestry.get(p)==sha(p),'Sealed historical table/metadata changed: '+p)
    points=read(paths['point_identity.json']);proof=read(paths['normalization_proof.json'])
    admission=read(paths['admission']);availability=read(paths['metric_availability.json']);plan=read(paths['comparison_plan'])
    require(proof['status']=='H_P_NORMALIZED_VIEW_NOT_STATISTICAL_RESULT' and proof['source_ids']==cc['source_ids']
        and proof['common_metric_admission_sha256']==identity(admission)
        and done['common_metric_admission_sha256']==sha(paths['admission'])
        and done['comparison_plan_sha256']==sha(paths['comparison_plan']),
        'Original canonical admission identity and file SHA domains differ')
    metrics={v['metric']:v['metric_identity'] for v in availability if v['status']=='ADMITTED'}
    missing={v['metric'] for v in availability if v['status']=='MISSING_IN_P'}
    require(len(metrics)==18 and set(proof['metrics'])==set(metrics)
        and missing=={'mse','resnet50_top1_probability','confidently_wrong'}
        and all(p['metric_identity']==metrics for p in points.values()),'Original18 admission/3 missing P metrics differ')
    for name,h in metrics.items():
        ent=admission['metrics'][name];require(ent['status']=='ADMITTED' and ent['branches']['H']['identity']==ent['branches']['P']['identity'],'Historical per-metric identity differs')
        expected=identity(dict(metric=name,identity=ent['branches']['H']['identity'],both_branch_numerics={k:{j:v[j] for j in ('precision','batch_execution')} for k,v in ent['branches'].items()},admission_sha256=identity(admission)))
        require(h==expected,'Historical metric identity digest differs')
    return dict(cfg=cfg,reg=reg,done=done,paths=paths,points=points,proof=proof,metrics=metrics,
        availability=availability,pairs=[(r['method'],r['reference']) for r in plan['comparisons']],bindings=combine(bind(spec.values()),bind(paths.values())))

    hp=ctx['hp'];ids=ctx['ids'];bridge=read(hp['paths']['target_pixel_bridge.json']);rows=read(ctx['main_rows'])
    require(ctx['visual']['science']['outputs'].get(ctx['main_rows'])==sha(ctx['main_rows']),'Actual MAIN rows changed')

    for h in read(ctx['H_rows']):
        i=h['source_index'];require(type(i)is int and 0<=i<100 and h['source_id']==ids[i],'Historical H source row identity differs')
        if h['development_slot']==0 and h['noise_seed']==6201:
            require(references[i] is None,'Duplicate fixed H reference row');references[i]=h
    require(all(r is not None for r in references),'Original H row required for every source prediction/reference hash')

def run(config_path):
    require(sys.platform.startswith('linux'),'Original Linux identity required');cfg,reg=sealed(config_path);check_stop(cfg)
    e=Path(cfg['execution_dir']);out=Path(cfg['out']);require(not out.exists() and not any((e/n).exists() for n in ('owner_start.json','failure.json','completion.json','worker.log')),'No automatic retry')
    require(all(absent(p) for p in reg['predecessors']),'Original visual owner/worker must be reaped')
    require(process(os.getpid())['argv']==command(cfg,config_path,'run'),'Registered supervisor argv required')
    os.sched_setaffinity(0,cfg['affinity']);os.setpriority(os.PRIO_PROCESS,0,15)
    child=None;log=None;locks=contextlib.ExitStack();began=time.monotonic();me=process(os.getpid())
    try:
        import fcntl
        for p in reg['lock_paths']:
            f=locks.enter_context(Path(p).open('a+'));fcntl.flock(f,fcntl.LOCK_EX|fcntl.LOCK_NB)
        save(e/'owner_start.json',dict(identity=me,registration_sha256=sha(cfg['registration'])))
        argv=command(cfg,config_path,'worker');logpath=e/'worker.log';log=logpath.open('xb')
        env=dict(os.environ,PYTHONPATH=os.pathsep.join(cfg['pythonpath']),CUDA_VISIBLE_DEVICES='',PYTHONDONTWRITEBYTECODE='1',**{k:'2' for k in THREADS})
        child=subprocess.Popen(argv,cwd=cfg['root'],env=env,stdin=subprocess.DEVNULL,stdout=log,stderr=subprocess.STDOUT)
        save(e/'spawn_pending_identity.json',dict(pid=child.pid,argv=argv,identity_verified=False))
        ident=capture(child,argv,me['uid'])
        save(e/'worker_launch.json',dict(identity=ident,owner_identity=me,registration_sha256=sha(cfg['registration'])))
        while child.poll() is None:
            check_stop(cfg);require(time.monotonic()-began<cfg['max_seconds'],'Registered CPU owner deadline');time.sleep(.2)
        code=waited(child);log.flush();os.fsync(log.fileno());log.close();require(absent(ident),'Worker must be reaped')
        cp=out/'completion.json';exit=dict(exit_code=code,wait_completed=True,identity=ident,log_sha256=sha(logpath),completion_sha256=sha(cp) if cp.exists() else None)
        save(e/'worker_exit.json',exit);require(code==0 and cp.exists() and not(out/'failure.json').exists(),'Actual successful CPU exit required')
