This supervisor never constructs a decoder/model or changes any packet ledger.
The original native GPU guard remains inside the worker. The monitoring parent
does not turn a completed child's brief zombie state into a foreign-PID error.
"""

SCHEMA='MAIN_ACTUAL_VISUAL_OWNER_CONFIG_R1'
DONE='MAIN_ACTUAL_VISUAL_NORMALLY_COMPLETE_R1'
REGISTERED='MAIN_ACTUAL_VISUAL_REGISTERED_NOT_LAUNCHED_R1'
ADMITTED='MAIN_VISUAL_WORKER_ADMITTED'
THREAD_KEYS=('OMP_NUM_THREADS','MKL_NUM_THREADS','OPENBLAS_NUM_THREADS','NUMEXPR_NUM_THREADS')
require=w.require;read=w.read;sha=w.sha;save=w.save;verify=w.verify;combine=w.combine


def bind(paths):return {str(Path(p).absolute()):sha(p) for p in paths}
def argv(owner):return [owner['python'],'-B',str(Path(__file__).absolute()),'--config',owner['_path']]


def exited(identity):
    try:return not ids.same_identity(identity,ids.process_identity(identity['pid']))
    except (FileNotFoundError,ProcessLookupError):return True


def atomic_save(path,value):
    path=Path(path);tmp=path.with_name(path.name+'.pending-'+str(os.getpid()));save(tmp,value)
    try:os.link(tmp,path)
    finally:tmp.unlink()


def validate_shape(owner,ctx):
    cfg=ctx['cfg'];e=Path(owner['owner_out']);study=Path(cfg['root'])/'outputs/CONTENT-REAL-64QAM-20261006'
    require(owner['schema']==SCHEMA and owner['registration']==cfg['registration']
        and owner['worker_config']==cfg['_path'] and owner['_path']==cfg['visual_owner_config'], 'Exact visual owner/config required')
    require(owner['worker_argv']==w.command(cfg) and owner['python']==cfg['python']
        and cfg['visual_owner_module']==str(Path(__file__).absolute()),'Registered original GPU interpreter/entry required')
    require(owner['gpu_device']==cfg['gpu_device']==0 and owner['gpu_threads']==cfg['gpu_threads']==6
        and owner['gpu_affinity']==cfg['gpu_affinity'] and owner['cpu_affinity']==sorted(cfg['gpu_affinity'])
        and len(set(owner['cpu_affinity']))==6 and owner['nice']==cfg['nice']==15,
        'One GPU and exact original six CPU threads required')
    require(owner['max_seconds']==cfg['max_seconds'] and 0<owner['max_seconds']<=86400
        and owner['deadline_unix']==cfg['deadline_unix'] and owner['automatic_successor'] is False,
        'Bounded visual-only owner required')
    require(e.is_absolute() and e.parent==study/'MAIN' and e!=Path(cfg['out'])
        and owner['visual_lock_path']==str(study/'shared_visual.lock')
        and Path(owner['nvidia_smi']).is_absolute(), 'Original shared study GPU lock and separate execution directory required')
    require(ctx['bound'].get(owner['nvidia_smi'])==sha(owner['nvidia_smi']), 'Actual GPU inventory executable must be bound')


def load(config,require_seal=True):
    path=str(Path(config).absolute());owner=read(path);owner['_path']=path
    ctx=w.load_registered(owner['worker_config']);validate_shape(owner,ctx)
    require(ctx['reg']['owner_config_sha256']==sha(path) and ctx['reg']['owner_scope']=='MAIN_ACTUAL_VISUAL_ONLY',
        'Owner must belong to the independent visual registration')
    if require_seal:
        e=Path(owner['owner_out']);require(not(e/'registration_failure.json').exists(),'Registration failure retained')
        seal=read(e/'registration_completion.json');require(seal['status']==REGISTERED
            and seal['registration_sha256']==ctx['regsha'] and seal['owner_config_sha256']==sha(path), 'Successful visual registration seal required')
        verify(seal['outputs'])
    require(w.budget_snapshot(ctx)==ctx['before'],'Readonly MAIN budget differs before visual stage')
    return dict(ctx,owner=owner)


def check_stop(ctx,*,enforce_deadline=True):
    cfg=ctx['cfg'];study=Path(cfg['root'])/'outputs/CONTENT-REAL-64QAM-20261006'
    require((not enforce_deadline or time.time()<cfg['deadline_unix']) and not any(p.exists() for p in
        (Path(cfg['stop_file']),Path(cfg['out'])/'STOP',study/'STOP',study/'H/STOP')), 'STOP or original deadline blocks visual stage')


def request_stop(ctx,cause):
    p=Path(ctx['cfg']['stop_file'])
    if not p.exists():
        try:save(p,dict(status='MAIN_VISUAL_STOP_PRESERVE',cause=cause,registration_sha256=ctx['regsha']))
        except FileExistsError:pass


def gpu_idle(owner,runner=subprocess.run):
    exe=owner['nvidia_smi']
    def query(fields,kind='gpu'):
        return runner([exe,'--query-'+kind+'='+fields,'--format=csv,noheader,nounits'],
            check=True,capture_output=True,text=True,timeout=15).stdout.strip()
    lines=query('index,uuid').splitlines()
    require(len(lines)==1,'Exactly one physical GPU required')
    fields=[x.strip() for x in lines[0].split(',')]
    require(len(fields)==2 and fields[0]=='0' and fields[1].startswith('GPU-'),'Unexpected GPU identity')
    pids=query('pid','compute-apps')
    require(not pids,'GPU compute PID present; no historical or foreign exception')
    return dict(status='GPU_IDLE_CONFIRMED',gpu_device=0,uuid=fields[1],compute_pids=[],
        nvidia_smi_sha256=sha(exe),observed_unix=time.time())


@contextlib.contextmanager
def visual_lock(owner):
    import fcntl
    p=Path(owner['visual_lock_path']);p.parent.mkdir(parents=True,exist_ok=True)
    with p.open('a+') as f:
        fcntl.flock(f,fcntl.LOCK_EX|fcntl.LOCK_NB)
        try:yield
        finally:fcntl.flock(f,fcntl.LOCK_UN)


def lock_is_held(path):
    """Worker proves another process holds the shared lock; never acquires it."""
    import fcntl
    with Path(path).open('r+') as f:
        try:fcntl.flock(f,fcntl.LOCK_EX|fcntl.LOCK_NB)
        except BlockingIOError:return True
        fcntl.flock(f,fcntl.LOCK_UN);return False


def environment():
    return dict(os.environ,CUDA_VISIBLE_DEVICES='0',PYTHONDONTWRITEBYTECODE='1',PYTHONUNBUFFERED='1',
        **{k:'6' for k in THREAD_KEYS})


def admission_paths(owner):
    e=Path(owner['owner_out']);return e/'worker_admission.json',e/'worker/launch.json',e/'gpu_admission.json'


def check_live_admission(ctx,supervision):
    """Called by the still-running worker only; never by parent's poll loop."""
    cfg=ctx['cfg'];owner=read(cfg['visual_owner_config']);owner['_path']=cfg['visual_owner_config'];validate_shape(owner,ctx)
    p,lp,gp=admission_paths(owner);require(supervision==read(p),'Worker admission changed')
    me=ids.process_identity(os.getpid());parent=supervision['owner_identity']
    require(supervision['status']==ADMITTED and supervision['registration_sha256']==ctx['regsha']
        and supervision['config_sha256']==sha(cfg['_path']) and ids.same_identity(me,supervision['worker_identity'])
        and me['argv']==w.command(cfg) and me['uid']==parent['uid'] and os.getppid()==parent['pid']
        and parent['argv']==argv(owner) and ids.same_identity(parent,ids.process_identity(parent['pid'])),
        'Exact registered live parent/worker required')
    require(os.environ.get('CUDA_VISIBLE_DEVICES')=='0' and _preflight_environment.check(ctx['bound'],cfg['root'])
        and os.sched_getaffinity(0)==set(cfg['gpu_affinity']) and os.getpriority(os.PRIO_PROCESS,0)==15,
        'Exact admitted GPU resources required')
    verify(supervision['bindings']);launch=read(lp);gpu=read(gp)
    require(launch['identity']==supervision['worker_identity'] and launch['owner_identity']==parent
        and launch['registration_sha256']==ctx['regsha'] and launch['config_sha256']==sha(cfg['_path'])
        and gpu==supervision['gpu_admission'] and gpu['status']=='GPU_IDLE_CONFIRMED'
        and gpu['owner_identity']==parent and gpu['registration_sha256']==ctx['regsha']
        and gpu['visual_lock_path']==owner['visual_lock_path'] and lock_is_held(owner['visual_lock_path']),
        'Live lock/launch/GPU admission proof differs')
    check_stop(ctx)


def admit_worker(config_path,ctx):
    require(str(Path(config_path).absolute())==ctx['cfg']['_path'],'Wrong visual worker config')
    owner=read(ctx['cfg']['visual_owner_config']);owner['_path']=ctx['cfg']['visual_owner_config'];validate_shape(owner,ctx)
    path,_,_=admission_paths(owner);began=time.monotonic()
    while not path.is_file():
        check_stop(ctx);require(time.monotonic()-began<120,'Owner admission timed out before any GPU loading');time.sleep(.1)
    value=read(path);check_live_admission(ctx,value)
    # The parent holds the MAIN accounting-owner lock throughout this stage.
    # Audit the budget at admission/terminal boundaries, not every image frame.
    require(w.budget_snapshot(ctx)==ctx['before'],'MAIN budget changed before admitted visual worker')
    return value


def spawn(ctx,parent):
    owner=ctx['owner'];cfg=ctx['cfg'];folder=Path(owner['owner_out'])/'worker';folder.mkdir()
    logpath=folder/'worker.log';log=logpath.open('xb');command=w.command(cfg)
    def resources():os.sched_setaffinity(0,owner['gpu_affinity']);os.setpriority(os.PRIO_PROCESS,0,15)
    try:
        p=subprocess.Popen(command,cwd=cfg['root'],env=environment(),stdin=subprocess.DEVNULL,stdout=log,
            stderr=subprocess.STDOUT,preexec_fn=resources)
        save(folder/'spawn_pending_identity.json',dict(pid=p.pid,argv=command,identity_verified=False))
        child=ids.process_identity(p.pid)
        require(child['argv']==command and child['uid']==parent['uid'],'Actual visual child identity differs')
        launch=dict(status='MAIN_VISUAL_WORKER_LAUNCHED',registration_sha256=ctx['regsha'],config_sha256=sha(cfg['_path']),
            owner_identity=parent,identity=child,gpu_device=0,threads=6,affinity=owner['gpu_affinity'])
        save(folder/'launch.json',launch)
        return dict(process=p,log=log,folder=folder,logpath=logpath,launch=launch)
    except BaseException:
        request_stop(ctx,'Spawn/identity failure; preserve child evidence')
        if 'p' in locals():p.wait()
        log.close();raise


def wait_child(ctx,child,began):
    """Poll only the process handle; wait/reap before inspecting terminal proof."""
    failed=False;cfg=ctx['cfg']
    while child['process'].poll() is None:
        if time.monotonic()-began>=ctx['owner']['max_seconds'] or time.time()>=cfg['deadline_unix']:
            request_stop(ctx,'Registered visual deadline');failed=True
        if Path(cfg['stop_file']).exists() or (Path(cfg['out'])/'STOP').exists():failed=True
        time.sleep(.2)
    code=child['process'].wait();child['log'].flush();child['log'].close()
    cp=Path(cfg['out'])/'completion.json';folder=child['folder'];identity=child['launch']['identity']
    result=dict(status='MAIN_VISUAL_WORKER_EXITED',registration_sha256=ctx['regsha'],config_sha256=sha(cfg['_path']),
        identity=identity,exit_code=code,wait_completed=True,launch_sha256=sha(folder/'launch.json'),
        log_sha256=sha(child['logpath']),completion_sha256=sha(cp) if cp.is_file() else None)
    save(folder/'exit.json',result)
    require(exited(identity),'Waited worker identity must no longer be live')
    require(not failed and code==0 and cp.is_file() and not Path(cfg['stop_file']).exists()
        and not(Path(cfg['out'])/'STOP').exists(), 'Worker/STOP failure preserved; no normal completion')
    return result


def receipt(ctx):
    cfg=ctx['cfg'];p=Path(cfg['out'])/'completion.json';done=read(p)
    require(done['status']==w.DONE and done['registration_sha256']==ctx['regsha'] and done['config_sha256']==sha(cfg['_path'])
        and done['source_count']==1000 and done['frame_count']==3000*len(ctx['cpu_ctx']['schedule'])
        and done['source_ids']==[r['source_id'] for r in ctx['cpu_ctx']['records']]
        and done['schedule']==ctx['cpu_ctx']['schedule'] and done['noise_seeds']==list(w.NOISE_SEEDS), 'Complete original visual population/grid required')
    require(all(done[k] is False for k in ('policy_selection','development_used','holdout_used','automatic_successor_started','main_final_selection_complete'))
        and all(done[k] is True for k in ('source_decode_complete','image_reconstruction_complete','images_scored','GPU_used'))
        and done['new_packet_decodes']==done['noise_generated']==0 and done['receiver_rule']=='KEEP'
        and done['metric_objective']==w.PRIMARY, 'Visual-only KEEP/L14 completion required')
    require(done['source_bindings']==ctx['reg']['source_bindings'] and done['input_bindings']==ctx['reg']['input_bindings']
        and done['budget_before']==done['budget_after']==ctx['before']==w.budget_snapshot(ctx), 'Dependency or zero-PHY budget changed')
    require(done['visual_source_bindings']==ctx['static_bindings'] and done['numerical_runtime']==ctx['expected_flags']
        and done['frozen_visual_identity']==ctx['calibration']['identity'] and done['score_identity']==ctx['metric_identity'],
        'Actual native/metric model/source/FP32 identity differs')
    verify(done['outputs']);return done


def closed_batch(spec):
    require(set(spec)=={'owner_config','completion'},'Exact normal visual predecessor specification')
    ctx=load(spec['owner_config']);owner=ctx['owner'];cfg=ctx['cfg'];e=Path(owner['owner_out'])
    require(spec['completion']==str(e/'completion.json') and not any((p/n).exists() for p in (e,Path(cfg['out']))
        for n in ('failure.json','STOP')),'Original visual failure cannot be accepted as normal success')
    check_stop(ctx,enforce_deadline=False);done=read(spec['completion']);verify(done['outputs']);start=read(e/'owner_start.json')
    parent=start['identity'];launch=read(e/'worker/launch.json');end=read(e/'worker/exit.json');science=receipt(ctx)
    require(done['status']==DONE and done['registration_sha256']==ctx['regsha'] and done['owner_config_sha256']==sha(spec['owner_config'])
        and start['registration_sha256']==ctx['regsha'] and start['owner_config_sha256']==sha(spec['owner_config'])
        and parent['argv']==argv(owner) and parent==done['owner_identity']==launch['owner_identity'] and exited(parent),
        'Original successful owner must normally exit')
    require(end['status']=='MAIN_VISUAL_WORKER_EXITED' and end['identity']==launch['identity'] and exited(end['identity'])
        and launch['identity']['argv']==w.command(cfg) and end['exit_code']==0 and end['wait_completed'] is True
        and end['launch_sha256']==sha(e/'worker/launch.json') and end['log_sha256']==sha(e/'worker/worker.log')
        and end['completion_sha256']==sha(Path(cfg['out'])/'completion.json'), 'Original worker waited exit0 and closed log required')
    admission=read(e/'worker_admission.json')
    require(science['supervision']==admission and admission['owner_identity']==parent and admission['worker_identity']==end['identity']
        and admission['registration_sha256']==ctx['regsha'] and admission['config_sha256']==sha(cfg['_path']), 'Original supervision differs')
    verify(admission['bindings'])
    require(done['budget_before']==done['budget_after']==ctx['before'] and done['frame_count']==science['frame_count']
        and done['new_packet_decodes']==0 and done['automatic_successor_started'] is False, 'Owner scope/budget differs')
    bindings=combine(ctx['bound'],done['outputs'],science['outputs'],bind([spec['completion'],Path(cfg['out'])/'completion.json']))
    return dict(ctx=ctx,done=science,owner_done=done,bindings=bindings)


def run(config):
    require(sys.platform.startswith('linux'),'Authentic Linux process identity required')
    ctx=load(config);owner=ctx['owner'];cfg=ctx['cfg'];e=Path(owner['owner_out']);check_stop(ctx)
    require(not any((e/n).exists() for n in ('attempt.json','owner_start.json','failure.json','completion.json','worker_admission.json','worker'))
        and not Path(cfg['out']).exists(),'Fresh registered visual batch only; no retry')
    me=ids.process_identity(os.getpid());require(me['argv']==argv(owner),'Actual owner argv differs')
    require(set(owner['cpu_affinity'])<=os.sched_getaffinity(0),'Registered parent CPU union unavailable')
    os.sched_setaffinity(0,owner['cpu_affinity']);os.setpriority(os.PRIO_PROCESS,0,15)
    child=None;started=time.monotonic();handlers={};locks=contextlib.ExitStack()
    for n in (signal.SIGTERM,signal.SIGINT):handlers[n]=signal.signal(n,lambda *_:request_stop(ctx,'Owner signal; safe worker drain'))
    try:
        locks.enter_context(visual_lock(owner))
        # Exclude a new MAIN packet owner until this zero-PHY stage closes.
        import fcntl
        mainlock=locks.enter_context(Path(ctx['cpu_ctx']['cfg']['ledger']).with_suffix('.owner.lock').open('a+'))
        fcntl.flock(mainlock,fcntl.LOCK_EX|fcntl.LOCK_NB)
        save(e/'attempt.json',dict(status='MAIN_VISUAL_OWNER_ATTEMPT_NO_RETRY',identity=me,registration_sha256=ctx['regsha']))
        save(e/'owner_start.json',dict(identity=me,registration_sha256=ctx['regsha'],owner_config_sha256=sha(config)))
        gpu=dict(gpu_idle(owner),owner_identity=me,registration_sha256=ctx['regsha'],visual_lock_path=owner['visual_lock_path'])
        save(e/'gpu_admission.json',gpu);require(w.budget_snapshot(ctx)==ctx['before'],'Budget changed before worker')
        Path(cfg['out']).mkdir();child=spawn(ctx,me)
        supervision=dict(status=ADMITTED,registration_sha256=ctx['regsha'],config_sha256=sha(cfg['_path']),
            owner_identity=me,worker_identity=child['launch']['identity'],gpu_device=0,threads=6,affinity=owner['gpu_affinity'],
            gpu_admission=gpu,bindings=bind([e/'owner_start.json',e/'gpu_admission.json',e/'worker/launch.json']))
        atomic_save(e/'worker_admission.json',supervision)
        wait_child(ctx,child,started);child=None
        science=receipt(ctx);require(science['supervision']==supervision,'Scientific worker used a different admission')
        verify(ctx['bound']);check_stop(ctx)
        outputs=combine(science['outputs'],bind([Path(cfg['out'])/'completion.json',e/'attempt.json',e/'owner_start.json',
            e/'gpu_admission.json',e/'worker_admission.json',e/'worker/launch.json',e/'worker/exit.json',e/'worker/worker.log']))
        done=dict(status=DONE,registration_sha256=ctx['regsha'],owner_config_sha256=sha(config),owner_identity=me,
            frame_count=science['frame_count'],source_count=1000,budget_before=ctx['before'],budget_after=w.budget_snapshot(ctx),
            source_bindings=ctx['reg']['source_bindings'],input_bindings=ctx['reg']['input_bindings'],outputs=outputs,
            new_packet_decodes=0,noise_generated=0,GPU_used=True,automatic_successor_started=False,
            policy_selection=False,development_used=False,elapsed_seconds=time.monotonic()-started)
        require(done['budget_after']==ctx['before'],'Visual owner changed MAIN ledger');save(e/'completion.json',done);return done
    except BaseException:
        failure=traceback.format_exc();request_stop(ctx,'Visual owner failed; preserve all evidence')
        if child is not None:
            try:wait_child(ctx,child,started)
            except BaseException:pass
        if not(e/'failure.json').exists():save(e/'failure.json',dict(status='MAIN_VISUAL_OWNER_FAILED_PRESERVE',registration_sha256=ctx['regsha'],identity=me,traceback=failure))
        raise
    finally:
        # Locks remain held while an exceptional child drains and logs close.
        locks.close()
        for n,handler in handlers.items():signal.signal(n,handler)


if __name__=='__main__':
    p=argparse.ArgumentParser();p.add_argument('--config',required=True);a=p.parse_args();print(run(a.config)['status'],flush=True)
