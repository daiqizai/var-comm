        outputs=p.bind([f/n for n in ('process.log','launch.json','exit.json')]))
    p.save(c['loc']['q']/'completion.json',q);return q

    def test_identity_capture_failure_keeps_pending_actual_pid(self):

            with mock.patch.object(p.subprocess,'Popen',return_value=child),mock.patch.object(p,'capture',side_effect=RuntimeError('identity failed')):

    def test_parent_gpu_union_and_no_native_load_no_hidden_successor(self):
        text=Path(p.__file__).read_text(encoding='utf-8');tree=ast.parse(text)
        launch=next(n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name=='launch')
        self.assertIn('resources(list(range(4, 10)))',ast.unparse(launch))
        calls={n.func.attr for n in ast.walk(tree) if isinstance(n,ast.Call) and isinstance(n.func,ast.Attribute)}
        self.assertTrue({'build_native','decode_once','MetricScorer','resolve'}.isdisjoint(calls))
        self.assertNotIn('while True',text);self.assertEqual(p.TEST_MODULES[-1],'test_register_main_actual_visual_r1')

    def test_real_four_directory_layout_runs_complete_formal_inventory(self):
        # These are source/metadata fixtures only; no actual context is called.
        visual=Path(os.environ.get('MAIN_VISUAL_TEST_SOURCE_RUNTIME',HERE))
        actual=Path(os.environ.get('MAIN_VISUAL_TEST_ACTUAL_RUNTIME',HERE))
        physical=Path(os.environ.get('MAIN_VISUAL_TEST_PHYSICAL_RUNTIME',HERE))
        screen=Path(os.environ.get('MAIN_VISUAL_TEST_PRESCREEN_RUNTIME',HERE))
        renderer=Path(os.environ.get('MAIN_TEST_SOURCE_RENDERER',BASE/'h_source_driver.py'))
        plan=Path(os.environ.get('MAIN_TEST_PLAN_SUMMARY',BASE/'current/main_plan_only/MAIN/plan_only_v1/result/summary'))
        uep=Path(os.environ.get('MAIN_TEST_UEP_RUNTIME',BASE.parent/'prior_aware_uep_20261004/runtime'))
        with tempfile.TemporaryDirectory() as td:
            root=Path(td);paths={k:root/k for k in ('visual','actual','physical','prescreen','private','unrelated')}
            for path in paths.values():path.mkdir()
            for name in p.NEW_SOURCES:shutil.copyfile(visual/name,paths['visual']/name)
            cpu=p.module(actual/'operate_main_actual_calibration_r1.py','_visual_test_cpu_ops')
            for name in cpu.NEW_SOURCES:shutil.copyfile(actual/name,paths['actual']/name)
            old=p.module(physical/'operate_main_cpu_phase_r2_remote.py','_visual_test_old_ops')
            for name in list(old.PINS)+['operate_main_cpu_phase_r2_remote.py']:shutil.copyfile(physical/name,paths['physical']/name)
            prior=p.module(screen/'operate_main_psnr_freeze_r1.py','_visual_test_psnr_ops')
            for name in prior.PURE_PINS:shutil.copyfile(screen/name,paths['prescreen']/name)
            # Private imports must not accidentally find a same-directory engine.
            shutil.copyfile(p.__file__,paths['private']/Path(p.__file__).name)
            env=dict(os.environ,PYTHONPATH=os.pathsep.join(str(paths[k]) for k in ('visual','actual','physical','prescreen')),
                CUDA_VISIBLE_DEVICES='',PYTHONDONTWRITEBYTECODE='1',MAIN_TEST_SOURCE_RENDERER=str(renderer),
                MAIN_TEST_PLAN_SUMMARY=str(plan),MAIN_TEST_UEP_RUNTIME=str(uep))
            env.update({k:'2' for k in p.THREAD_KEYS})
            result=subprocess.run([sys.executable,'-B','-m','unittest','-v',*p.TEST_MODULES],cwd=paths['unrelated'],env=env,
                capture_output=True,text=True,timeout=180)
            self.assertEqual(result.returncode,0,result.stdout+result.stderr)
            self.assertRegex(result.stderr,rf'Ran {p.TEST_COUNT} tests in');self.assertNotIn('skipped',result.stderr)
