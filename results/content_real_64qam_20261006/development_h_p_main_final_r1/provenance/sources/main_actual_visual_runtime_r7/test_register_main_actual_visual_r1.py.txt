class Tests(unittest.TestCase):
    def test_actual_stage_key_and_normal_original_visual_closure(self):
        with tempfile.TemporaryDirectory() as td:
            spec=visual_fixture(td)
            with mock.patch.object(r.cpu_owner.w,'exited',return_value=True):v=r.visual_closed(spec)
            self.assertEqual(v['python'],'originalUM');self.assertEqual(v['science']['source_count'],1000)
            self.assertIn(spec['config'],v['bindings']);self.assertEqual(len(v['source_bindings']),2)

    def test_live_original_owner_blocks_reference(self):
        with tempfile.TemporaryDirectory() as td:
            spec=visual_fixture(td)
            with mock.patch.object(r.cpu_owner.w,'exited',return_value=False),self.assertRaisesRegex(RuntimeError,'owner'):r.visual_closed(spec)

    def test_zero_exit_and_original_stage_receipt_are_both_required(self):
        with tempfile.TemporaryDirectory() as td:
            spec=visual_fixture(td);e=Path(spec['completion']).parent;f=e/'stages/source/workers/full1000_source/exit_receipt.json'
            value=r.read(f);value['exit_code']=1;rewrite(f,value)
            with mock.patch.object(r.cpu_owner.w,'exited',return_value=True),self.assertRaises(RuntimeError):r.visual_closed(spec)

    def test_changed_original_source_configuration_refused(self):
