        self.views.append(copy.deepcopy(view));image=np.full((3,256,256),.5,np.float32)
        hit=bool(self.cache)

            receiver_view_sha256=a.wire.identity(view),visual_cache_key='c'*64,visual_cache_hit=hit,
            target_used_for_generation=False,new_packet_decodes=0)
        return dict(image=image,summary=summary)


class FakeScorer:
    def __init__(self,mutate=None):self.prepared=[];self.calls=[];self.mutate=mutate
    def prepare_source(self,record,target,index):
        assert target.dtype==np.float32 and target.shape==(3,256,256)
        self.prepared.append((copy.deepcopy(record),target.copy(),index))
    def __call__(self,record,images):
        self.calls.append((record,len(images)))
        rows=[dict(dinov2_vitl14_cosine=.4,psnr_db=12.,lpips_alex=.2,dino_cosine=.5) for _ in images]
        if self.mutate:
            for row in rows:self.mutate(row)
        return rows

    def test_real_receiver_receive_full_frame_then_visual_phase_completion(self):
        # Exercise the frozen real receive method, not a handwritten subset of
        # its output. Only the PHY/ledger/model callbacks are synthetic.
        rec,_,target,points=fixtures();cat=object.__new__(a.wire.PublicCatalogue)
        cat._profiles=[dict(profile_id=0,m=1,K=0,wire_key='e'*64,groups=[dict(phy_key='d'*64,source_bits=12,
            information_bits=28,transmitted_bits=100,symbols=50,modulation='QPSK',layout={'fake':True})])]
        cat._aliases=[];cat.digest=a.wire.identity(cat._profiles);cat.aliases_digest=a.wire.identity([]);cat.backend_identity={'fake':True}

        backend=SimpleNamespace(identity=cat.backend_identity,device='cpu',torch=fake_torch)
        ledger=SimpleNamespace(branch='MAIN',register_configuration=lambda *args:None,
            decode_once=lambda phase,event_id,**kwargs:kwargs['decode']())
        header=SimpleNamespace(receive=lambda *args:dict(profile_id=0,header_crc_ok=True,header_fields_legal=True,header_ok=True,score=1.))

        crc=lambda bits:bool(np.asarray(bits)[-1])
        phy=SimpleNamespace(receive_packet=physical,crc_accept_batch=crc)
        physical_rx=a.wire.Receiver(cat,header=header,backend=backend,phy=phy,ledger=ledger)
