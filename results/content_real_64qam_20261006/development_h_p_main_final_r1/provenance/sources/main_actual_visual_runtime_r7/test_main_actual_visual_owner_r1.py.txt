class Tests(unittest.TestCase):
    def test_registered_shape_single_gpu_original_union_lock(self):
        with tempfile.TemporaryDirectory() as td:
            c=fixture(td);o.validate_shape(c['owner'],c)
            for field,value in [('cpu_affinity',[4,5]),('gpu_threads',2),('visual_lock_path',str(Path(td)/'newlock')),('automatic_successor',True)]:
                bad=dict(c['owner'],**{field:value})
                with self.assertRaises(RuntimeError):o.validate_shape(bad,c)

    def test_gpu_inventory_one_empty_only_no_zombie_exception(self):
        with tempfile.TemporaryDirectory() as td:
            c=fixture(td)
            def runner_with(values):return mock.Mock(side_effect=[SimpleNamespace(stdout=s) for s in values])
            ok=o.gpu_idle(c['owner'],runner_with(['0, GPU-fixture\n','']))
            self.assertEqual(ok['status'],'GPU_IDLE_CONFIRMED')
            for values in (['0, GPU-a\n1, GPU-b\n'],['0, GPU-a\n','999\n']):
                with self.assertRaises(RuntimeError):o.gpu_idle(c['owner'],runner_with(values))

    def test_worker_environment_keeps_original_six_threads(self):
        with mock.patch.dict(os.environ,{'CUDA_VISIBLE_DEVICES':'','OMP_NUM_THREADS':'2'}):
            env=o.environment();self.assertEqual(env['CUDA_VISIBLE_DEVICES'],'0')
            self.assertTrue(all(env[k]=='6' for k in o.THREAD_KEYS))

    def test_atomic_admission_never_overwrites(self):
        with tempfile.TemporaryDirectory() as td:
            p=Path(td)/'admission';o.atomic_save(p,{'x':1})
            with self.assertRaises(FileExistsError):o.atomic_save(p,{'x':2})
            self.assertEqual(o.read(p),{'x':1})

    def test_current_worker_admission_requires_real_parent_lock_resources(self):
        with tempfile.TemporaryDirectory() as td:
            c=fixture(td);a=admission(c)
            with mock.patch.object(o.os,'getpid',return_value=11),mock.patch.object(o.os,'getppid',return_value=10),\
                 mock.patch.object(o.ids,'process_identity',side_effect=lambda p:a['worker_identity'] if p==11 else a['owner_identity']),\
                 mock.patch.object(o.os,'sched_getaffinity',return_value=set(c['cfg']['gpu_affinity']),create=True),\
                 mock.patch.object(o.os,'getpriority',return_value=15,create=True),mock.patch.object(o.os,'PRIO_PROCESS',0,create=True),\
                 mock.patch.dict(os.environ,o.environment()),mock.patch.object(o,'lock_is_held',return_value=True),\
                 mock.patch.object(o.w,'budget_snapshot',side_effect=AssertionError('Do not scan ledger per frame')):
                o.check_live_admission(c,a)
                with mock.patch.object(o.os,'getppid',return_value=12),self.assertRaisesRegex(RuntimeError,'parent/worker'):o.check_live_admission(c,a)
                with mock.patch.object(o,'lock_is_held',return_value=False),self.assertRaisesRegex(RuntimeError,'lock/launch'):o.check_live_admission(c,a)

    def test_changed_admission_or_budget_blocks_live_worker(self):

    def test_poll_zombie_window_waits_before_identity_check(self):
        with tempfile.TemporaryDirectory() as td:
            c=fixture(td);a=admission(c);folder=Path(c['owner']['owner_out'])/'worker';lp=folder/'worker.log';log=lp.open('xb');log.write(b'complete\n')
            o.save(Path(c['cfg']['out'])/'completion.json',{'fixture':True});events=[]
            p=mock.Mock();p.poll.side_effect=[None,0];p.wait.side_effect=lambda:events.append('wait') or 0
            child=dict(process=p,log=log,folder=folder,logpath=lp,launch={'identity':a['worker_identity']})
            def reaped(x):self.assertTrue(log.closed);self.assertEqual(events,['wait']);events.append('reaped');return True
            with mock.patch.object(o.time,'sleep'),mock.patch.object(o,'exited',side_effect=reaped),\
                 mock.patch.object(o.ids,'process_identity',side_effect=AssertionError('No parent live check while monitoring')):
                result=o.wait_child(c,child,o.time.monotonic())
            self.assertEqual(events,['wait','reaped']);self.assertEqual(result['exit_code'],0)
            self.assertTrue((folder/'exit.json').is_file());p.terminate.assert_not_called();p.kill.assert_not_called()

    def test_stop_and_deadline_drain_keep_exit_and_logs(self):
        for why in ('stop','deadline'):
            with self.subTest(why=why),tempfile.TemporaryDirectory() as td:
                c=fixture(td);a=admission(c);folder=Path(c['owner']['owner_out'])/'worker';lp=folder/'worker.log';log=lp.open('xb');log.write(b'checkpoint retained')
                o.save(Path(c['cfg']['out'])/'completion.json',{'fixture':True})
                if why=='stop':o.request_stop(c,'fixture STOP')
                p=mock.Mock();p.poll.side_effect=[None,0];p.wait.return_value=0
                child=dict(process=p,log=log,folder=folder,logpath=lp,launch={'identity':a['worker_identity']})
                with mock.patch.object(o.time,'sleep'),mock.patch.object(o,'exited',return_value=True),self.assertRaises(RuntimeError):
                    o.wait_child(c,child,o.time.monotonic()-(101 if why=='deadline' else 0))
                self.assertTrue(log.closed);self.assertEqual(o.read(folder/'exit.json')['exit_code'],0)
                self.assertTrue(Path(c['cfg']['stop_file']).exists());p.terminate.assert_not_called();p.kill.assert_not_called()

    def test_closed_batch_requires_normal_owner_waited_worker_and_exact_receipt(self):
        with tempfile.TemporaryDirectory() as td:
            c=fixture(td);a=admission(c);e=Path(c['owner']['owner_out']);lp=e/'worker/worker.log';lp.write_bytes(b'COMPLETE\n')
            d=science(c);d['supervision']=a;cp=Path(c['cfg']['out'])/'completion.json';o.save(cp,d)
            end=dict(status='MAIN_VISUAL_WORKER_EXITED',identity=a['worker_identity'],exit_code=0,wait_completed=True,
                launch_sha256=o.sha(e/'worker/launch.json'),log_sha256=o.sha(lp),completion_sha256=o.sha(cp));o.save(e/'worker/exit.json',end)

    def test_supervisor_never_mutates_packet_ledger_or_loads_models(self):
        import ast
        tree=ast.parse(Path(o.__file__).read_text(encoding='utf-8'))
        calls={n.func.attr for n in ast.walk(tree) if isinstance(n,ast.Call) and isinstance(n.func,ast.Attribute)}
        self.assertTrue({'BudgetLedger','decode_once','build_native','MetricScorer','kill','terminate'}.isdisjoint(calls))
        source=Path(o.__file__).read_text(encoding='utf-8');self.assertLess(source.index('if child is not None:'),source.index('locks.close()'))
