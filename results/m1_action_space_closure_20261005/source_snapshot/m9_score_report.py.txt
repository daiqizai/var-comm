def require(ok,message):
    if not ok:raise RuntimeError(message)
def sha(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as f:
        for b in iter(lambda:f.read(8*1024*1024),b''):h.update(b)
    return h.hexdigest()
def read(path):return json.loads(Path(path).read_text(encoding='utf-8-sig'))
def ident(v):return hashlib.sha256(json.dumps(v,sort_keys=True,separators=(',',':'),allow_nan=False).encode()).hexdigest()
def write(path,value):
    p=Path(path);p.parent.mkdir(parents=True,exist_ok=True);tmp=p.with_name(p.name+'.tmp')
    tmp.write_text(json.dumps(value,indent=2,ensure_ascii=False,allow_nan=False)+'\n',encoding='utf-8');os.replace(tmp,p)
def seal(path,value):
    if Path(path).exists():require(read(path)==value,'Immutable receipt differs: '+str(path))
    else:write(path,value)
def verify(bindings):
    for p,h in bindings.items():require(sha(p)==h,'Bound input changed: '+str(p))
def pixels(a):
    a=np.asarray(a);require(a.dtype==np.float32 and a.shape==(3,256,256) and np.isfinite(a).all()
        and a.min()>=0 and a.max()<=1,'Expected unchanged float32 CHW256 RGB');return np.ascontiguousarray(a)
def rgb_sha(a):return hashlib.sha256(b'float32:3,256,256:RGB\0'+pixels(a).tobytes()).hexdigest()
def read_csv(path):
    opener=gzip.open if str(path).endswith('.gz') else open
    with opener(path,'rt',encoding='utf-8-sig',newline='') as f:return list(csv.DictReader(f))
def csv_write(path,rows):
    p=Path(path);p.parent.mkdir(parents=True,exist_ok=True);fields=list(dict.fromkeys(k for r in rows for k in r))
    opener=gzip.open if str(path).endswith('.gz') else open
    with opener(path,'wt',encoding='utf-8',newline='') as f:
        w=csv.DictWriter(f,fieldnames=fields,lineterminator='\n');w.writeheader();w.writerows(rows)

def validate_seal(cp,binding=None,index=None):
    require(cp['payload_sha256']==ident({k:v for k,v in cp.items() if k!='payload_sha256'}),'Source checkpoint seal differs')
    if binding is not None:require(cp['binding']==binding,'Source registration binding differs')
    if index is not None:require(int(cp['source_index'])==index,'Source index differs')

def validate_source(cp,index,excluded=()):
    validate_seal(cp,index=index);rows=cp['rows'];expected=expected_source(index,excluded)
    require(len(rows)==len(expected) and {row_key(r) for r in rows}==expected,'Development omits a legal frame or fabricates an unavailable oracle')
    proof=cp['float_reconstructions'];require(sha(proof['path'])==proof['sha256'],'Development float archive changed')
    with np.load(proof['path'],allow_pickle=False) as z:
        require(set(z.files)=={'images','source_rgb','row_ids','image_slots'},'Float archive schema differs')
        images=z['images'].copy();target=pixels(z['source_rgb']).copy();slots=z['image_slots'].tolist();ids=z['row_ids'].tolist()
    require(images.dtype==np.float32 and images.ndim==4 and images.shape[1:]==(3,256,256),'Invalid float archive images')
    require(len(slots)==len(expected) and ids==[r['replay_row_id'] for r in rows] and len(set(ids))==len(expected),'Development row mapping differs')
    hashes=[rgb_sha(a) for a in images];reference=rgb_sha(target)
    for row,slot in zip(rows,slots):
        require(type(slot) is int and 0<=slot<len(images) and row.get('image_slot',slot)==slot,'Invalid image slot')
        require(row['image_sha256']==hashes[slot] and row['reference_sha256']==reference,'Development RGB/source hash differs')
        require(row.get('label_conditioned',False) is False,'Unexpected class conditioning')
        require(row['source_id']==cp['record']['image_id'] and row['preprocessing_id']==cp['record']['preprocessing_id'],'Development source identity differs')
    return rows,images,target,slots

def policy_index(policy):
    import m9_phy as phy
    require(policy['sources']==1000 and policy['noise_seeds']==[4101,4102,4103]
        and policy['full_grid'] is True and policy['development_read'] is False,'Complete calibration-only policy required')
    result={};legal={(N,p):{(a.m,a.q,a.order) for a in phy.action_grid(N,p)} for N,p in CELLS}
    for r in policy['cells']:
        a=r['action'];key=(int(r['N']),r['phy_family'],int(r['snr_db']),r['method'])
        require(key not in result and (key[0],key[1]) in legal,'Duplicated/outside policy cell')
        require((int(a['N']),a['phy'])==key[:2] and (int(a['m']),int(a['q']),a['order']) in legal[key[:2]],'Unregistered policy action')
        result[key]=r
    require(set(result)=={(N,p,s,m) for N,p in CELLS for s in SNRS for m in ('whole','raster','random','entropy','oracle')},'Policy scope incomplete')
    return result

def action_for(policy,N,phy,snr,method):
    if method.endswith('_at_entropy'):
        a=dict(policy[N,phy,snr,'entropy']['action']);a['order']='whole' if a['q']==0 else method.removesuffix('_at_entropy');return a
    return dict(policy[N,phy,snr,method.removesuffix('_policy')]['action'])

def action_interpretation(a):
    if int(a['q'])==0:return 'whole scale; no partial-token or ordering contribution'
    return 'partial scale '+str(int(a['m'])+1)

def unavailable_cells(policy,receipt):
    import m9_phy as phy
    expected={}
    for N,p in CELLS:
        for s in SNRS:
            a=action_for(policy,N,p,s,'oracle_at_entropy')
            try:phy.Action(**a)
            except ValueError:
                maximum=phy.max_legal_q(N,p,int(a['m']),'oracle')
                require(int(a['q'])>maximum,'Unexpected invalid paid oracle')
                expected[N,p,s,'oracle_at_entropy']=(int(a['m']),int(a['q']),maximum)
    seen=set()
    for r in receipt['rows']:
        key=(int(r['N']),r['phy_family'],int(r['snr_db']),r['method'])
        require(key in expected and key not in seen and r['status']=='NOT_FEASIBLE'
            and isinstance(r['reason'],str) and r['reason'],'Unregistered unavailable action')
        require((int(r['m']),int(r['q']),int(r['maximum_legal_q']))==expected[key],'Unavailable oracle silently changed K or budget')
        seen.add(key)
    require(seen==set(expected),'Unavailable same-K receipt differs from physical legality');return seen

def interval(values,draws):
    a=np.asarray(values,np.float64);require(a.shape==(100,) and np.isfinite(a).all(),'Complete finite source vector required')
    lo,hi=np.quantile(a[draws].mean(1),[.025,.975])
    return dict(mean=float(a.mean()),ci_low=float(lo),ci_high=float(hi),n_sources=100,n_frames=300,
        bootstrap_unit='source_mean_over_three_noise_repeats',bootstrap_replicates=10000,bootstrap_seed=20261002)

def analyze(rows,excluded=()):
    groups=defaultdict(dict)
    for r in rows:
        g=group_key(r);key=(int(r['source_index']),int(r['noise_seed']))
        require(g[2] in SNRS and key not in groups[g],'Unexpected or duplicate comparison frame')
        groups[g][key]=r
    expanded={g for g in groups if g[3]=='expanded'}
    require(expanded=={(N,p,s,'expanded',m) for N,p in CELLS for s in SNRS for m in METHODS if (N,p,s,m) not in excluded},'Expanded group coverage differs')
    for N in (1024,2048):
        require(all((N,'continuous',s,'P',f'P{N}') in groups for s in SNRS),'Missing same-budget historical P group')
    draws=np.random.default_rng(20261002).integers(0,100,(10000,100));means={};summary=[]
    for g,by in sorted(groups.items()):
        require(set(by)==population(),'Incomplete source/noise comparison group')
        metadata=dict(N=g[0],phy_family=g[1],snr_db=g[2],comparison_scope=g[3],method=g[4])
        for metric in METRICS+EXTRA:
            values=[by[i,n].get(metric) for i in range(100) for n in SEEDS]
            if any(v in (None,'') for v in values):
                require(g[3]!='expanded' or metric not in METRICS,'Missing mandatory new metric: '+metric)
                # History can lack a later diagnostic; it is absent, never imputed.
                require(metric not in METRICS[:8]+METRICS[9:11],'Missing mandatory historical image metric: '+metric)
                continue
            arr=np.asarray(values,np.float64).reshape(100,3).mean(1);means[g+(metric,)]=arr
            summary.append(dict(**metadata,metric=metric,**interval(arr,draws)))
        for field in EVENTS:
            if not all(field in r and type(r[field]) is bool for r in by.values()):continue
            arr=np.array([by[i,n][field] for i in range(100) for n in SEEDS],float).reshape(100,3).mean(1)
            summary.append(dict(**metadata,metric=field+'_fraction',**interval(arr,draws)))
    paired=[]
    for N,p in CELLS:
        for snr in SNRS:
            base=(N,p,snr);new=lambda m:base+('expanded',m)
            contrasts=[(new(m),new('whole_policy'),'partial_policy_minus_whole') for m in METHODS[1:]]
            contrasts += [(new('entropy_policy'),new(m),'same_m_K_order_control') for m in METHODS[5:]]
            contrasts += [(new(m),(N,'continuous',snr,'P',f'P{N}'),'digital_minus_continuous') for m in METHODS]
            for scope in ('legacy_low_screened','legacy69','legacy82_m9'):
                contrasts += [(new(m),base+(scope,m),'expanded_minus_historical_grid') for m in METHODS]
            for ga,gb,contrast in contrasts:
                if ga not in groups or gb not in groups:continue
                a,b=groups[ga],groups[gb]
                for key in population():
                    require(all(a[key][f]==b[key][f] for f in ('source_id','preprocessing_id','reference_sha256')),'Paired methods use different sources or float targets')
                for metric in METRICS+EXTRA:
                    if ga+(metric,) not in means or gb+(metric,) not in means:continue
                    paired.append(dict(N=N,phy_family=p,snr_db=snr,scope_A=ga[3],method_A=ga[4],scope_B=gb[3],method_B=gb[4],
                        contrast=contrast,metric=metric,delta='A_minus_B',**interval(means[ga+(metric,)]-means[gb+(metric,)],draws),
                        pairing='same source and nominal seed; distinct P noise namespace' if gb[3]=='P' else 'same source and registered M1 noise convention; waveform depends on action',
                        inference_scope='descriptive reused development after action-range revision'))
    return summary,paired

def markdown(headers,rows):
    def cell(x):return str(x).replace('|','\\|').replace('\n',' ')
    return '\n'.join(['| '+' | '.join(map(cell,headers))+' |','| '+' | '.join('---' for _ in headers)+' |']+['| '+' | '.join(map(cell,r))+' |' for r in rows])
