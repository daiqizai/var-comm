class FakeCommands:
    def __init__(self, root):
        self.calls=[];self.root=root;self.head='a'*40;self.remote=self.head
        self.staged=set();self.changed=False;self.didcommit=False;self.tree='b'*40

    def run(self, argv, codes=(0,)):
        a=[str(v) for v in argv];self.calls.append(a)
        if a[:2] == ['git', 'ls-remote']:return self.remote+'\trefs/heads/main',0
        if a[:4] == ['git','diff','--cached','--name-only']:return '\n'.join(sorted(self.staged)),0
        if a[:3] == ['git','diff','--name-only']:return 'unrelated.py' if self.changed else '',0
        if a[:2] == ['git','branch']:return 'main',0
        if a[:2] == ['git','rev-parse']:
            ref=a[2]
            if ref=='HEAD':return self.head,0
            if ref=='HEAD^':return 'a'*40,0
            if ref=='HEAD^{tree}':return self.tree,0
            if ref.startswith(':'):return 'f'*40,0
        if a[:2] == ['git','hash-object']:return 'f'*40,0
        if a[:2] == ['git','write-tree']:return self.tree,0
        if a[:2] == ['git','add']:self.staged.update(a[3:]);return '',0
        if a[:2] == ['git','commit']:self.head='c'*40;self.didcommit=True;return '',0
        if a[:2] == ['git','push']:self.remote=self.head;return '',0
        if 'tools/update_repository_manifest.py' in a:
            (self.root/'release_manifest.json').write_text('{}\n');return '',0
        return '',0

    def test_payload_bit_length_is_safe_but_array_is_raw(self):
        p.reject_raw({'payload_bits':3060, 'groups':[{'source_bits':400}]})
        p.inspect_text(b'payload_bits,symbols\n3060,1024\n','.csv')
        for bad in ({'payload_bits':[0,1]}, {'images':[[0]]}, {'tokens':[3]}):
            with self.assertRaises(RuntimeError):p.reject_raw(bad)
        with self.assertRaises(RuntimeError):p.inspect_text(b'payload_bits,symbols\n"[0,1]",2\n','.csv')

    def test_metric_provenance_fingerprints_and_msssim_are_not_tensors(self):
        p.reject_raw({'assets':{'weights':{'path':'weights.pt','sha256':'a'*64}},
                      'preprocessing':{'weights':[.0448,.2856,.3001,.2363,.1333]}})
        for bad in ({'weights':[.1,.2]}, {'weights':{'tensor':[[1,2]]}},
                    {'weights':{'path':'weights.pt','sha256':'not-a-digest'}}):
            with self.assertRaisesRegex(RuntimeError,'arrays forbidden'):p.reject_raw(bad)

    def test_csv_record_split_exact_including_multiline_utf8(self):
        data='method,note\n甲,"first\nsecond"\n乙,ok\n丙,end\n'.encode()
        parts=p.csv_parts(data,35)
        self.assertGreater(len(parts),1)
        self.assertEqual(b''.join(parts),data)
        self.assertTrue(all(len(x)<35 for x in parts))
        self.assertEqual(list(csv.reader(io.StringIO(b''.join(parts).decode())))[1],['甲','first\nsecond'])

    def test_alive_recycled_zombie_process_identity(self):
        wanted=dict(pid=55,start_ticks=100,uid=1002)
        with patch.object(p,'identity',return_value=dict(wanted,state='R')):
            self.assertFalse(p.exited(wanted))
        with patch.object(p,'identity',return_value=dict(wanted,state='Z')):
            self.assertTrue(p.exited(wanted))
        with patch.object(p,'identity',return_value=dict(pid=55,start_ticks=101,uid=1002,state='R')):
            self.assertTrue(p.exited(wanted))
        with patch.object(p,'identity',return_value=dict(pid=55,start_ticks=100,uid=1003,state='R')):
            with self.assertRaisesRegex(RuntimeError,'UID'):p.exited(wanted)

    def test_explicit_normal_publish_checks_and_remote_verification(self):
        inv=self.fx.prepare();cmd=FakeCommands(self.fx.root)
        env=self.fx.root/'outputs/UNIFIED-METRICS-20261002/environment/bin/python'
        env.parent.mkdir(parents=True);env.write_text('mock python')
        with patch.object(p,'old_delivery_gate',return_value={'status':'VERIFIED'}):
            result=p.publish(self.fx.root,self.fx.out,cmd)
        self.assertEqual(result['status'],'PUBLISHED')
        self.assertEqual(result['commit'],result['remote_commit'])
        commands=cmd.calls
        self.assertIn(['git','push','origin','HEAD:main'],commands)
        self.assertTrue(any('tools/verify_repository.py' in c for c in commands))
        self.assertTrue(any('tools/run_cpu_checks.py' in c for c in commands))
        self.assertIn(['git','diff','--cached','--check'],commands)
        self.assertIn(['git','merge-base','--is-ancestor',result['commit'],result['remote_commit']],commands)
        for c in commands:
            self.assertFalse(any(x in c for x in ('--force','reset','clean','config')))
            if c[:2]==['git','add']:
                self.assertTrue(set(c[3:]) <= set(inv['files'])|{'release_manifest.json'})

    def test_remote_may_advance_only_when_it_contains_commit(self):
        class AncestorCommands:
            def __init__(self):self.calls=[]
            def run(self,a,codes=(0,)):
                self.calls.append(a)
                if a[:3]==['git','merge-base','--is-ancestor']:
                    raise RuntimeError('not an ancestor')
                return '',0
        with self.assertRaisesRegex(RuntimeError,'not an ancestor'):
            p.remote_contains(AncestorCommands(),'a'*40,'b'*40)
