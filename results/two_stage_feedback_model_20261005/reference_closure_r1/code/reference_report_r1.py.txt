"""Independent report for symmetric exact-layout reference closure.

The registered original search/report modules and outputs remain unchanged.
Only FIXED/QCSI references are replaced; no GPU or physical decoder is loaded.
"""

import sys

os.environ['CUDA_VISIBLE_DEVICES'] = ''
for _key in ('OMP_NUM_THREADS', 'MKL_NUM_THREADS', 'OPENBLAS_NUM_THREADS', 'NUMEXPR_NUM_THREADS'):
    os.environ[_key] = '2'
import numpy as np

VERSION = 'TWO_STAGE_SYMMETRIC_REFERENCE_CLOSURE_REPORT_R1'
SCOPE = 'TWO_STAGE_FEEDBACK_REFERENCE_CLOSURE_REPORT_R1'
REFERENCES = ('FIXED', 'QCSI')
UNCHANGED = ('TWO_STAGE', 'TWO_STAGE_C8')
TOLERANCE = 1e-9


def require(ok, message):
    if not ok:
        raise RuntimeError(message)


def load_runtime(path):
    path = Path(path).resolve()
    for name in ('report', 'auxiliary'):
        source = path / (name + '.py')
        spec = importlib.util.spec_from_file_location(name, source)
        module = importlib.util.module_from_spec(spec)
        sys.modules[name] = module
        spec.loader.exec_module(module)
    return sys.modules['report'], sys.modules['auxiliary']


def canonical_sha(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True, separators=(',', ':'), allow_nan=False).encode()).hexdigest()


def reference_paths(policy, row):
    """Compute every publicly identified forward layout, with feedback charged once."""
    require(policy['kind'] == row['method'] and row['method'] in REFERENCES, 'Wrong reference policy kind')
    require(policy['reference_closure'] == 'SYMMETRIC_EXACT_FORMATS_R1', 'Incomplete reference catalogue')
    require(policy['pilot_symbols'] == row['pilot_symbols'], 'Reference pilot budget changed')
    branches = policy['branches']
    count = 1 if row['method'] == 'FIXED' else 2 ** row['feedback_bits']
    require(len(branches) == count, 'Reference public branch count differs from feedback bins')
    require([b['public_header_id'] for b in branches] == list(range(count)), 'Reference branch IDs are not public first-header IDs')
    base = row['pilot_symbols'] + 68
    if row['method'] == 'QCSI':
        require(policy['feedback_bits'] == row['feedback_bits']
                and policy['feedback_symbols'] == row['feedback_symbols'], 'Reference feedback cost differs')
        require(len(policy['thresholds_db']) == count - 1, 'Reference quantizer bin count differs')
        base += row['feedback_symbols']
    allowed_headers = {0, row['second_header_symbols']}
    if row['header_mode'] == 'measured_headers' and row['second_header_symbols'] == 16:
        allowed_headers = {0}
    paths = []
    for branch in branches:
        if branch['format'] == 'CATALOG':
            used = base + sum(g['symbols'] for g in branch['profile']['groups'])
        else:
            require(branch['format'] == 'PAIR', 'Unregistered reference layout family')
            first, second = branch['first'], branch['second']
            require(first['state_id'] in ('m4_K0', 'm5_K0', 'm6_K0')
                    and second['start_state'] == first['state_id'], 'Reference pair prefix is not contiguous')
            header = branch['second_header_symbols']
            require(header in allowed_headers, 'Reference uses an unavailable or unmatched second header format')
            extra = 0 if second['stop'] else header + second['symbols']
            used = base + first['symbols'] + extra
        require(0 <= used <= 1024, 'Reference public path exceeds N1024')
        paths.append(used)
    require(abs(max(paths) - row['max_used_symbols']) <= TOLERANCE, 'Reference summary resource maximum differs')
    return paths


def validate_policy_catalog(report, grid, policies):
    require(policies.get('public_header_bits') == 12
            and policies.get('public_tree_ids_local_to_frozen_operating_cell') is True,
            'Reference first-header identification contract is missing')
    for pid, policy in policies['policies'].items():
        require(canonical_sha(policy) == pid, 'Reference policy hash differs')
    original_style = [row for row in grid if row['method'] in UNCHANGED]
    report.validate_policies(original_style, policies)
    count = 0
    for row in grid:
        if row['method'] in REFERENCES and row['status'] == 'MODEL_ONLY':
            reference_paths(policies['policies'][row['policy_id']], row)
            count += 1
    return dict(status='PASS', reference_method_rows=count, all_paths_at_most_1024=True,
                first_header_symbols=68, public_header_bits=12, reference_feedback_paid_once=True)


def merge_checks(original, revised):
    """Validate reference monotonicity only on the construction population."""
    old = {(r['cell_id'], r['method']): r for r in original}
    new = {(r['cell_id'], r['method']): r for r in revised}
    require(len(old) == len(original) and len(new) == len(revised) and set(old) == set(new),
            'Reference closure changed the registered cell grid')
    changes, dominance = [], []
    for key, row in new.items():
        prior = old[key]
        require(all(row[k] == prior[k] for k in ('average_snr_db', 'header_mode', 'pilot_symbols', 'feedback_bits',
                'feedback_symbols', 'second_header_symbols', 'first_m', 'status')), 'Reference closure changed grid or availability')
        if row['method'] in UNCHANGED:
            for field in ('policy_id', 'train_mean', 'review_mean', 'max_used_symbols',
                          'bad_probability_train', 'bad_probability_review', 'mean_used_symbols', 'mean_used_symbols_train'):
                require(row.get(field) == prior.get(field), 'Original two-stage result changed: ' + field)
            continue
        if row['status'] != 'MODEL_ONLY':
            continue
        gain = row['train_mean'] - prior['train_mean']
        require(gain >= -TOLERANCE, 'Expanded reference is worse than its original construction-set optimum')
        changes.append(dict(cell_id=row['cell_id'], method=row['method'], average_snr_db=row['average_snr_db'],
                            header_mode=row['header_mode'], pilot_symbols=row['pilot_symbols'],
                            feedback_bits=row['feedback_bits'], feedback_symbols=row['feedback_symbols'],
                            second_header_symbols=row['second_header_symbols'], first_m=row['first_m'],
                            old_policy_id=prior['policy_id'], new_policy_id=row['policy_id'],
                            old_train_mean=prior['train_mean'], new_train_mean=row['train_mean'], train_difference=gain,
                            old_review_mean=prior['review_mean'], new_review_mean=row['review_mean'],
                            review_difference=row['review_mean'] - prior['review_mean'],
                            review_used_for_selection=False, original_results_preserved=True))
        if row['method'] == 'QCSI':
            two = old[row['cell_id'], 'TWO_STAGE']
            if two['status'] == 'MODEL_ONLY':
                delta = row['train_mean'] - two['train_mean']
                require(delta >= -TOLERANCE, 'Symmetric reference closure failed QCSI >= TWO_STAGE construction dominance')
                dominance.append(dict(cell_id=row['cell_id'], average_snr_db=row['average_snr_db'],
                                      qcsi_train_mean=row['train_mean'], two_stage_train_mean=two['train_mean'],
                                      qcsi_minus_two_stage=delta, tolerance=TOLERANCE, status='PASS',
                                      review_dominance_required=False))
    return changes, dominance


def verify_unchanged_arrays(report, old_folder, new_folder, old_search, new_search, bindings):
    require(old_search['source_ids'] == new_search['source_ids'] and old_search['state_ids'] == new_search['state_ids'],
            'Reference closure changed source/state alignment')
    checked = 0
    # Both receipts bind each archive; compare the unchanged methods numerically
    # without rewriting containers or mistaking a recompressed NPZ for new science.
    old_cells = {p.stem: p for p in (Path(old_folder) / 'cells').glob('*.npz')}
    new_cells = {p.stem: p for p in (Path(new_folder) / 'cells').glob('*.npz')}
    require(set(old_cells) == set(new_cells), 'Reference closure array cell inventory changed')
    for cid in sorted(old_cells):
        old_path, new_path = old_cells[cid], new_cells[cid]
        require(str(old_path.resolve()) in bindings and str(new_path.resolve()) in bindings,
                'Unchanged array comparison is not receipt-bound')
        with np.load(old_path, allow_pickle=False) as a, np.load(new_path, allow_pickle=False) as b:
            for method in UNCHANGED:
                for suffix in ('quality_point', 'bad_probability', 'mean_used_symbols', 'tree_id'):
                    name = method + '_' + suffix
                    require((name in a) == (name in b), 'Unchanged method array availability differs')
                    if name in a:
                        require(np.array_equal(a[name], b[name]), 'Two-stage source array changed: ' + cid + '/' + name)
                        checked += 1
    return dict(status='PASS', array_pairs_checked=checked, cells=len(old_cells), exact_numerical_equality=True)

def build(args):
    report, cached = load_runtime(args.runtime_dir)
    original_root, revised_root, out = map(lambda p: Path(p).resolve(), (args.original_model, args.reference_dir, args.out))
    protocol = Path(args.protocol).resolve()
    require(report.read_json(protocol)['N'] == 1024, 'Unregistered resource budget')
    bindings = {str(protocol): report.sha(protocol)}
    for path in (Path(args.runtime_dir) / 'report.py', Path(args.runtime_dir) / 'auxiliary.py'):
        bindings[str(path.resolve())] = report.sha(path)
    grid, selected, primary, historical, changes, dominance, coverage, numerical, reference_numerical = [], [], [], [], [], [], [], [], []
    scores = {}
    review = None
    for snr in report.SNRS:
        old_folder, folder = original_root / ('search_snr' + str(snr)), revised_root / ('search_snr' + str(snr))
        old_done, old_bind = report.verify_receipt(old_folder / 'completion.json')
        done, new_bind = report.verify_receipt(folder / 'completion.json')
        require(done.get('reference_scope') == 'SYMMETRIC_EXACT_FORMATS_R1', 'Revised reference scope is not registered')
        bindings.update(old_bind); bindings.update(new_bind)
        for receipt in (old_done, done):
            require(receipt.get('real_decodes') == 0 and receipt.get('new_visual_inference') is False
                    and receipt.get('development_read') is False and receipt.get('GPU_used') is False,
                    'Reference reporting requires CPU-only table receipts')
        for name in ('search.json', 'split.json', 'grid_summary.csv', 'selected_summary.csv', 'primary_summary.csv',
                     'per_source.csv', 'policies.json', 'coverage.json'):
            require(str((old_folder / name).resolve()) in old_bind and str((folder / name).resolve()) in new_bind,
                    'Original or reference consumed file is not receipt-bound: ' + name)
        old_grid = [report.normalize_summary(r) for r in report.read_csv(old_folder / 'grid_summary.csv')]
        new_grid = [report.normalize_summary(r) for r in report.read_csv(folder / 'grid_summary.csv')]
        new_selected = [report.normalize_summary(r) for r in report.read_csv(folder / 'selected_summary.csv')]
        new_primary = [report.normalize_summary(r) for r in report.read_csv(folder / 'primary_summary.csv')]
        old_primary = [report.normalize_summary(r) for r in report.read_csv(old_folder / 'primary_summary.csv')]
        require(all(r['average_snr_db'] == snr for r in old_grid + new_grid + new_selected + new_primary), 'Foreign SNR in shard')
        report.validate_grids(new_grid, new_selected, expected_snrs=(snr,))
        report.validate_primary(new_grid, new_primary)
        require({r['cell_id'] for r in old_primary} == {r['cell_id'] for r in new_primary}, 'Primary control point changed')
        old_policies, policies = report.read_json(old_folder / 'policies.json'), report.read_json(folder / 'policies.json')
        for row in new_grid:
            if row['method'] in UNCHANGED and row['status'] == 'MODEL_ONLY':
                require(policies['policies'][row['policy_id']] == old_policies['policies'][row['policy_id']],
                        'Original two-stage policy changed')
        audit = validate_policy_catalog(report, new_grid, policies)
        delta, proof = merge_checks(old_grid, new_grid)
        changes.extend(delta); dominance.extend(proof)
        old_search, new_search = report.read_json(old_folder / 'search.json'), report.read_json(folder / 'search.json')
        exact = verify_unchanged_arrays(report, old_folder, folder, old_search, new_search, bindings)
        split = report.validate_split(report.read_json(folder / 'split.json'))
        require(review is None or review == split, 'Reference shards changed paired sources')
        review = split
        current = report.load_source_scores(folder / 'per_source.csv', new_selected, review)
        require(not set(scores) & set(current), 'Repeated reference cell IDs')
        scores.update(current)
        grid.extend(new_grid); selected.extend(new_selected); primary.extend(new_primary); historical.extend(old_primary)
        coverage.append(dict(average_snr_db=snr, original=report.read_json(old_folder / 'coverage.json'),
                             revised=report.read_json(folder / 'coverage.json'), policy_path_audit=audit, unchanged_arrays=exact))
        diag = folder / 'primary_reference_numerical_diagnostic.json'
        require(str(diag.resolve()) in new_bind, 'New reference numerical diagnostic is not receipt-bound')
        diagnostic = report.read_json(diag)
        require(diagnostic.get('status') == 'SAME_FROZEN_PRIMARY_REFERENCE_ORDER16_VS32'
                and diagnostic.get('reselected') is False
                and {r['method'] for r in diagnostic['rows']} == set(REFERENCES),
                'Reference numerical diagnostic scope differs')
        expected_ids = {r['method']: r['policy_id'] for r in new_primary}
        require(all(r['policy_id'] == expected_ids[r['method']] and r.get('reselected') is False
                    for r in diagnostic['rows']), 'Numerical diagnostic used another reference policy')
        reference_numerical.append(dict(average_snr_db=snr, **diagnostic))
        old_diag = old_folder / 'primary_numerical_diagnostic.json'
        require(str(old_diag.resolve()) in old_bind, 'Original numerical diagnostic not bound')
        original = report.read_json(old_diag)
        numerical.append(dict(average_snr_db=snr,
                              unchanged_two_stage_rows=[r for r in original['rows'] if r['method'] in UNCHANGED],
                              original_receipt_path=str(old_diag), original_receipt_sha256=report.sha(old_diag),
                              scope='Only unchanged TWO_STAGE/C8; original FIXED/QCSI diagnostics do not apply to new references'))
    grouped = report.validate_grids(grid, selected)
    weights = report.bootstrap_weights()
    paired = report.paired_comparisons(grouped, scores, weights)
    effects = report.matched_axis_effects(grouped, scores, weights)
    overview = report.headlines(grouped)
    auxiliary = cached.build(revised_root, args.quality_csv, args.auxiliary_manifest, bindings, weights)
    bindings.update(auxiliary['input_bindings'])
    source_bindings = {str(Path(__file__).resolve()): report.sha(__file__),
                       **{str((Path(args.runtime_dir) / name).resolve()): report.sha(Path(args.runtime_dir) / name)
                          for name in ('report.py', 'auxiliary.py')}}
    summary = dict(status='SYMMETRIC_REFERENCE_CLOSURE_REPORT_COMPLETE', version=VERSION, N=1024,
                   metric='dinov2_vitl14_cosine', full_grid_method_rows=len(grid), selected_method_rows=len(selected),
                   selected_missing_method_rows=sum(r['status'] == 'MISSING_ASSET' for r in selected),
