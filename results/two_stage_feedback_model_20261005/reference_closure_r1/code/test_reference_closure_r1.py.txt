import importlib.util
import itertools
from pathlib import Path
import unittest
import numpy as np
import reference_closure_r1 as r

frozen_search=next(p for p in (Path(__file__).parents[1]/'runtime'/'search.py',Path(__file__).parents[2]/'code'/'search.py') if p.exists())
spec=importlib.util.spec_from_file_location('original_search_for_test',frozen_search)
search=importlib.util.module_from_spec(spec);spec.loader.exec_module(search)


def bundle(coefficients):
    c=np.asarray(coefficients,dtype=float)
    return dict(coefficients=c,prefix_indices=np.ones(len(c),dtype=int),
                target_indices=np.full(len(c),2,dtype=int),gray_index=0)


class ReferenceClosureTests(unittest.TestCase):
    def test_interval_envelope_matches_exhaustive_candidates_and_budgets(self):
        rng=np.random.default_rng(17)
        c=rng.uniform(size=(7,8,4)); c/=c.sum((1,2))[:,None,None]
        Q=np.array([[.1,.2,.3,.4], [.3,.4,.5,.6], [.8,.7,.6,.5]])
        costs=np.array([100,150,300,500,650,700,900]); budgets=[200,600,1000]
        envelope=r.Envelope(budgets); envelope.add(bundle(c),Q,[0,1,2],costs,lambda i:dict(candidate=i))
        means=Q[:,:3].mean(1)
        for budget in budgets:
            values,records=envelope.at(budget)
            for col,(a,b) in enumerate(r.INTERVALS):
                possible=[float((c[i,a:b].sum(0)[:3]*means).sum()) for i in range(len(c)) if costs[i]<=budget]
                self.assertAlmostEqual(values[col],max(possible),places=13)
                chosen=records[col]['branch']['candidate'];self.assertLessEqual(costs[chosen],budget)

    def test_qcsi_matches_exhaustive_thresholds_and_covers_uniform_tree(self):
        c=np.zeros((3,8,4));c[:,:,0]=1/8
        c[0,:4,0]=0;c[0,:4,1]=1/8
        c[1,4:,0]=0;c[1,4:,2]=1/8
        c[2,:,0]=0;c[2,:,1]=1/8
        Q=np.array([[0,0,0,0],[.4,.5,.6,.7],[.7,.8,.9,1.]])
        envelope=r.Envelope([1000]); envelope.add(bundle(c),Q,[0,1,2],[100,200,250],lambda i:dict(format='CATALOG',profile={'groups':[]},candidate=i))
        result=r.select_reference([envelope],Q,[0,1,2],16,search,bits=1,feedback=8)
        expected=-np.inf
        for partition in search.partitions(1):
            for choices in itertools.product(range(3),repeat=2):
                value=0
                for (a,b),i in zip(partition,choices):
                    value+=float((c[i,a:b].sum(0)[:3]*Q[:,:3].mean(1)).sum())
                expected=max(expected,value)
        self.assertAlmostEqual(result['scores'][:3].mean(),expected,places=13)
        self.assertGreaterEqual(result['scores'][:3].mean(),Q[1,:3].mean())
        self.assertEqual([b['public_header_id'] for b in result['policy']['branches']],[0,1])
        self.assertLessEqual(result['max_used_symbols'],1024)

    def test_same_format_replay_kept_when_signed_quality_makes_extra_header_better(self):
        Q=np.array([[0.,0.],[.8,.8],[.2,.2]])
        success=np.zeros((1,8,4)); success[:,:,2]=1/8
        reject=np.zeros((1,8,4)); reject[:,:,1]=1/8
        zero=r.Envelope([1008]);zero.add(bundle(success),Q,[0],[200],lambda i:dict(format='PAIR',second_header_symbols=0))
        paid=r.Envelope([1008]);paid.add(bundle(reject),Q,[0],[268],lambda i:dict(format='PAIR',second_header_symbols=68))
        selected=r.select_reference([zero,paid],Q,[0],16,search)
        self.assertEqual(selected['policy']['branches'][0]['second_header_symbols'],68)
        self.assertAlmostEqual(selected['scores'][0],.8)
        self.assertEqual(selected['max_used_symbols'],284)
        selected_review=r.select_reference([zero,paid],np.array([[0.,0.],[.8,-.8],[.2,1.]]),[0],16,search)
        self.assertEqual(selected_review['policy']['branches'][0]['second_header_symbols'],68)
        self.assertAlmostEqual(selected_review['scores'][1],-.8)


if __name__=='__main__':unittest.main()
