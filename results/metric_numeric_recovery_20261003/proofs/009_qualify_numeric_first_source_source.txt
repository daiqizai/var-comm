import csv,importlib.util,json,sys,time
from pathlib import Path
ROOT=Path('/home/liulu/projects/VAR_COMM');OUT=ROOT/'outputs/METRIC-NUMERIC-RECOVERY-20261003'
sys.path.insert(0,str(ROOT/'experiments/unified-metrics-20261002'))
import runner,replay
sys.path.insert(0,str(ROOT/'outputs/METRIC-FINAL-CACHE-20261002/runtime'));import cache_assembly as assembly
sys.path.insert(0,str(ROOT/'outputs/METRIC-CONCURRENT-R4-20261002/runtime'));import cache_handoff as cache

    for key,value in original.items():assert cache.same_json(row[key],value),(key,row[key],value)

qual=dict(status='REAL_FIRST_SOURCE_PARITY_PASS',parity_passed=True,synthetic=False,source_index=0,

    total_rows=2370,fresh_m2_replay_rows=780,historical_cache_rows=1590,parity_origin_counts=counts,
    checkpoints_bound={str(checkpoint):sha(checkpoint)},inputs=inputs,metric_batch_sizes_used=done['metric_batch_sizes_used'],
    original_parity_tolerances=replay.PARITY_TOLERANCES,training_updates=0,policy_selection_updates=0,time=time.time())
p=OUT/'real_first_source_qualification.json';assert not p.exists();write(p,qual)
