def require(ok,message):
    if not ok:raise RuntimeError(message)

def sha(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as f:
        for b in iter(lambda:f.read(8*1024*1024),b''):h.update(b)
    return h.hexdigest()

def read(path):return json.loads(Path(path).read_text(encoding='utf-8-sig'))
def ident(value):return hashlib.sha256(json.dumps(value,sort_keys=True,separators=(',',':'),allow_nan=False).encode()).hexdigest()
def write(path,value):
    p=Path(path);p.parent.mkdir(parents=True,exist_ok=True);tmp=p.with_name(p.name+'.tmp')
    tmp.write_text(json.dumps(value,indent=2,ensure_ascii=False,allow_nan=False)+'\n',encoding='utf-8');os.replace(tmp,p)
def seal(path,value):
    if Path(path).exists():require(read(path)==value,'Immutable receipt differs: '+str(path))
    else:write(path,value)
def verify(bindings):
    for p,h in bindings.items():require(sha(p)==h,'Bound input changed: '+str(p))
def pixels(a):
    a=np.asarray(a);require(a.dtype==np.float32 and a.shape==(3,256,256) and np.isfinite(a).all()
        and a.min()>=0 and a.max()<=1,'Expected unchanged float32 CHW256 RGB');return np.ascontiguousarray(a)
def rgb_sha(a):return hashlib.sha256(b'float32:3,256,256:RGB\0'+pixels(a).tobytes()).hexdigest()
def read_csv(path):
    opener=gzip.open if str(path).endswith('.gz') else open
    with opener(path,'rt',encoding='utf-8-sig',newline='') as f:return list(csv.DictReader(f))
def csv_write(path,rows):
    p=Path(path);p.parent.mkdir(parents=True,exist_ok=True);fields=list(dict.fromkeys(k for r in rows for k in r))
    opener=gzip.open if str(path).endswith('.gz') else open
    with opener(path,'wt',encoding='utf-8',newline='') as f:
        w=csv.DictWriter(f,fieldnames=fields);w.writeheader();w.writerows(rows)

def row_key(r):return(int(r['source_index']),int(r['snr_db']),int(r['noise_seed']),r['method'])


def legacy_header_compatibility(rows):
    for row in rows:
        require(type(row.get('header_crc_ok')) is bool and type(row.get('header_fields_legal')) is bool,
            'Original target development rows lack explicit CRC/header-legality diagnostics')
        require(not(row['header_crc_ok'] and not row['header_fields_legal']),
            'Original target development has a CRC-passed illegal header; extended header dictionary may change its output')


def validate_source(cp,index,excluded=()):
    require(cp['source_index']==index and cp['payload_sha256']==ident({k:v for k,v in cp.items() if k!='payload_sha256'}),
        'Source checkpoint seal differs')
    rows=cp['rows'];expected={(index,s,n,m) for s in SNRS for n in SEEDS for m in METHODS if (s,m) not in excluded}
    require(len(rows)==len(expected) and {row_key(r) for r in rows}==expected,'Expanded source omits a legal frame or fabricates an unavailable oracle')
    proof=cp['float_reconstructions'];require(sha(proof['path'])==proof['sha256'],'Expanded float archive changed')
    with np.load(proof['path'],allow_pickle=False) as z:
        require(set(z.files)=={'images','source_rgb','row_ids','image_slots'},'Float archive schema differs')
        images=z['images'].copy();target=pixels(z['source_rgb']).copy();slots=z['image_slots'].tolist();ids=z['row_ids'].tolist()
    require(images.dtype==np.float32 and images.ndim==4 and images.shape[1:]==(3,256,256),'Invalid float archive images')
    require(len(slots)==len(expected) and ids==[r['replay_row_id'] for r in rows] and len(set(ids))==len(expected),'Expanded row mapping differs')
    hashes=[rgb_sha(a) for a in images];reference=rgb_sha(target)
    for row,slot in zip(rows,slots):
        require(type(slot) is int and 0<=slot<len(images) and row.get('image_slot',slot)==slot,'Invalid image slot')
        require(row['image_sha256']==hashes[slot] and row['reference_sha256']==reference,'Expanded RGB/source hash differs')
        require(int(row['N'])==2048 and row['phy_family']==PHY and row.get('label_conditioned',False) is False,
                'Expanded N/PHY/class conditioning differs')
        require(row['source_id']==cp['record']['image_id'] and row['preprocessing_id']==cp['record']['preprocessing_id'],
                'Expanded source identity differs')
    return rows,images,target,slots


def policy_index(policy,expanded):
    require(policy['sources']==1000 and policy['noise_seeds']==[4101,4102,4103]
        and policy['full_grid'] is True and policy['development_read'] is False,'Original full calibration policy required')
    rows=[r for r in policy['cells'] if r['phy_family']==PHY and int(r['snr_db']) in SNRS]
    result={(int(r['snr_db']),r['method']):r for r in rows}
    require(len(rows)==len(result)==15 and set(result)=={(s,m) for s in SNRS for m in ('whole','raster','random','entropy','oracle')},
            'Policy cells are missing or duplicated')
    for row in rows:
        action=row['action'];m,k=int(action['m']),int(action['q'])
        require(4<=m<=(9 if expanded else 8) and k>=0,'Policy outside registered scale range')
        if m==9:
            legal=(0,64,128,144) if action['order']=='oracle' else (0,64,128,166)
            require(k in legal,'Unregistered m9 partial count')
        require((k==0)==(action['order']=='whole'),'Whole/partial action annotation differs')
    return result


def action_for(policy,snr,method):
    if method.endswith('_at_entropy'):
        action=dict(policy[snr,'entropy']['action'])
        action['order']='whole' if action['q']==0 else method.removesuffix('_at_entropy')
        return action
    return dict(policy[snr,method.removesuffix('_policy')]['action'])


def action_interpretation(action):
    m,k=int(action['m']),int(action['q'])
    if k==0:return 'm9 whole-scale extension; no partial-token or ordering contribution' if m==9 else 'whole-scale; no partial-token or ordering contribution'
    return 'partial scale 10' if m==9 else 'partial next scale within previous range'


def unavailable_cells(policy,receipt):
    expected={(s,'oracle_at_entropy') for s in SNRS if int(policy[s,'entropy']['action']['m'])==9
              and int(policy[s,'entropy']['action']['q'])==166}
    rows=receipt['rows'];seen=set()
    for row in rows:
        key=(int(row['snr_db']),row['method'])
        require(key not in seen and key in expected and row['phy_family']==PHY and isinstance(row['reason'],str) and row['reason'],
                'Unregistered unavailable action')
        require(int(row['N'])==2048 and int(row['m'])==9 and int(row['q'])==166
                and row['maximum_legal_q']==144 and row['status']=='NOT_FEASIBLE',
                'Unavailable action must preserve exact selected entropy m/K and paid-mask limit')
        seen.add(key)
    require(seen==expected,'Unavailable same-K receipt differs from paid-mask legality')
    return expected


def interval(values,draws):
    a=np.asarray(values,np.float64);require(a.shape==(100,) and np.isfinite(a).all(),'Complete finite source vector required')
    lo,hi=np.quantile(a[draws].mean(1),[.025,.975])
    return dict(mean=float(a.mean()),ci_low=float(lo),ci_high=float(hi),n_sources=100,n_frames=300,
        bootstrap_unit='source_mean_over_three_noise_repeats',bootstrap_replicates=10000,bootstrap_seed=20261002)


def analyze(rows,excluded=()):
    expected={(scope,s,m) for scope in ('expanded','legacy69') for s in SNRS for m in METHODS
              if scope!='expanded' or (s,m) not in excluded}
    expected|={('P2048',s,'P2048') for s in SNRS};groups=defaultdict(dict)
    for r in rows:
        g=(r['comparison_scope'],int(r['snr_db']),r['method']);key=(int(r['source_index']),int(r['noise_seed']))
        require(g in expected and key not in groups[g],'Unexpected or duplicated scored comparison frame')
        groups[g][key]=r
    require(set(groups)==expected and len(rows)==15300-300*len(excluded),'All legal expanded +7200 old +900 P rows required')
    population={(i,n) for i in range(100) for n in SEEDS};draws=np.random.default_rng(20261002).integers(0,100,(10000,100))
    means={};summary=[]
    for (scope,snr,method),by in sorted(groups.items()):
        require(set(by)==population,'Incomplete source/noise comparison group')
        for metric in METRICS+EXTRA:
            values=[by[i,n].get(metric) for i in range(100) for n in SEEDS]
            if any(x in (None,'') for x in values):
                require(metric not in METRICS,'Missing mandatory unified comparison metric: '+metric);continue
            arr=np.asarray(values,np.float64).reshape(100,3).mean(1);means[scope,snr,method,metric]=arr
            summary.append(dict(N=2048,phy_family='continuous' if scope=='P2048' else PHY,comparison_scope=scope,
                snr_db=snr,method=method,metric=metric,**interval(arr,draws)))
        for field in EVENTS:
            if not all(field in r for r in by.values()):continue
            require(all(type(r[field]) is bool for r in by.values()),'Event fractions require actual boolean decisions')
            arr=np.array([by[i,n][field] for i in range(100) for n in SEEDS],float).reshape(100,3).mean(1)
            summary.append(dict(N=2048,phy_family=PHY,comparison_scope=scope,snr_db=snr,method=method,
                metric=field+'_fraction',**interval(arr,draws)))
    contrasts=[('expanded',m,'legacy69',m,'expanded_action_set_minus_original69') for m in METHODS]
    contrasts += [('expanded',m,'expanded','whole_policy','partial_policy_minus_expanded_whole') for m in METHODS[1:]]
    contrasts += [('expanded','entropy_policy','expanded',m,'same_m_K_order_control') for m in METHODS[5:]]
    contrasts += [(scope,m,'P2048','P2048','digital_minus_historical_continuous') for scope in ('expanded','legacy69') for m in METHODS]
    paired=[]
    for snr in SNRS:
        for sa,ma,sb,mb,contrast in contrasts:
            if (sa,snr,ma) not in groups or (sb,snr,mb) not in groups:continue
            a,b=groups[sa,snr,ma],groups[sb,snr,mb]
            for key in population:
                require(all(a[key][f]==b[key][f] for f in ('source_id','preprocessing_id','reference_sha256')),
                    'Paired methods use different sources or float reference inputs')
            for metric in METRICS+EXTRA:
                ka,kb=(sa,snr,ma,metric),(sb,snr,mb,metric)
                if ka not in means or kb not in means:continue
                paired.append(dict(N=2048,phy_family=PHY,snr_db=snr,scope_A=sa,method_A=ma,scope_B=sb,method_B=mb,
                    contrast=contrast,metric=metric,delta='A_minus_B',**interval(means[ka]-means[kb],draws),
                    pairing='same source and nominal noise seed; P uses its original different noise namespace' if sb=='P2048'
                    else 'same source and registered M1 channel-noise convention; waveform depends on action'))
    return summary,paired

            and self.cal['newly_evaluated_frames']==117000,'Expanded calibration must combine 69 old and13 new actions on1000 sources')

def fill_p_missing(rows,images,features,index,mismatch,evaluator,metric,native,dino,torch,batch,record):
    """Compute only absent P metrics; existing numbers are copied unchanged."""
    from batch_speed import qualified_chunks
    target=images.pop('source_rgb');prepared=evaluator.prepare_reference(torch.from_numpy(target[None]));source_pred=int(prepared['resnet50'][0])
    need=[r for r in rows if any(r.get(m) in (None,'') for m in METRICS)]
    allowed={'dino_specificity','confidently_wrong'}
    require(all({m for m in METRICS if r.get(m) in (None,'')}<=allowed for r in need),'Unexpected missing historical P metric')
    rgb=[images[int(r['snr_db']),int(r['noise_seed'])][0] for r in need];computed=[];offset=0
    for count in qualified_chunks(len(rgb),batch['chosen_batch_size'],batch['qualified_batch_sizes']):
        with torch.inference_mode():
            tensor=torch.from_numpy(np.stack(rgb[offset:offset+count])).to('cuda:0')
            embed=native.dino_features(dino,tensor)
            negative=torch.from_numpy(features[mismatch[index]][None]).to('cuda:0').expand_as(embed)
            wrongcos=torch.nn.functional.cosine_similarity(embed,negative,dim=1).cpu().tolist()
            prob,pred=evaluator.models['resnet50'](metric.preprocess_resnet50(tensor)).float().softmax(-1).max(-1)
            computed.extend(zip(wrongcos,prob.cpu().tolist(),pred.cpu().tolist()))
        offset+=count
    for r,(wrongcos,prob,pred) in zip(need,computed):
        require(int(float(r['resnet50_top1_source_prediction']))==int(pred==source_pred)
            and int(float(r['resnet50_top1_label']))==int(pred==int(record['class_index'])),
            'Historical P classifier prediction differs while supplementing missing metrics')
        supplied=[]
        if r.get('dino_specificity') in (None,''):
            r['dino_specificity']=float(r['dino_cosine'])-wrongcos;supplied.append('dino_specificity')
        if r.get('confidently_wrong') in (None,''):
            r['confidently_wrong']=int(prob>=.5 and pred!=source_pred);supplied.append('confidently_wrong')
        r['supplemented_metrics']=','.join(supplied)
        r['supplement_origin']='existing admitted historical float RGB; no reconstruction, training or selection'
    for r in rows:
        require(all(r.get(m) not in (None,'') and math.isfinite(float(r[m])) for m in METRICS),'P mandatory metric unavailable')
        rgb,old=images[int(r['snr_db']),int(r['noise_seed'])]
        require(r['image_sha256']==rgb_sha(rgb) and r['reference_sha256']==rgb_sha(target),'P copied metric row does not match admitted RGB')
        r.update(source_index=index,snr_db=int(r['snr_db']),noise_seed=int(r['noise_seed']),N=2048,
            historical_model_seed=2026092304,source_class_index=int(record['class_index']))
    return rows

def markdown(headers,rows):
    def cell(x):return str(x).replace('|','\\|').replace('\n',' ')
    return '\n'.join(['| '+' | '.join(map(cell,headers))+' |','| '+' | '.join('---' for _ in headers)+' |']+
        ['| '+' | '.join(map(cell,r))+' |' for r in rows])
def fmt(x):return 'N/A' if x in (None,'') else f'{float(x):.4f}'
def ci(r):return 'N/A' if r is None else f"{float(r['mean']):.4f} [{float(r['ci_low']):.4f}, {float(r['ci_high']):.4f}]"
