def read(path): return json.loads(Path(path).read_text(encoding='utf-8-sig'))
def sha(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as f:
        for b in iter(lambda:f.read(1048576),b''):h.update(b)
    return h.hexdigest()
def save(path,value):
    path=Path(path);path.parent.mkdir(parents=True,exist_ok=True);tmp=path.with_name(path.name+'.tmp.'+str(os.getpid()))
    tmp.write_text(json.dumps(value,ensure_ascii=False,indent=2,allow_nan=False)+'\n',encoding='utf-8');os.replace(tmp,path)
def csv_save(path,rows):
    with Path(path).open('w',encoding='utf-8',newline='') as f:
        w=csv.DictWriter(f,fieldnames=list(rows[0]),lineterminator='\n');w.writeheader();w.writerows(rows)
def load_module(path):
    spec=importlib.util.spec_from_file_location('frozen_probability_model',path);module=importlib.util.module_from_spec(spec);spec.loader.exec_module(module);return module


def split_sources(source_ids):
    require(len(source_ids)==len(set(source_ids))==1000,'Original1000 unique calibration sources required')
    key=lambda s:(hashlib.sha256(json.dumps(['C/SOURCE_SPLIT',2026100501,s],ensure_ascii=False,separators=(',',':')).encode()).hexdigest(),s)
    ordered=sorted(source_ids,key=key);lookup={s:i for i,s in enumerate(source_ids)}
    return dict(ordered_source_ids=ordered,construction_source_ids=ordered[:800],verification_source_ids=ordered[800:],
                construction_indices=[lookup[s] for s in ordered[:800]],verification_indices=[lookup[s] for s in ordered[800:]],
                source_split_namespace='C/SOURCE_SPLIT',source_split_seed=2026100501,internal_verification_is_holdout=False)


def validate_protocol(p):
    require(p['schema']==SCHEMA,'Wrong content screening protocol')
    for k,v in dict(N=1024,pilot_symbols=16,feedback_bits=3,feedback_symbols=16,header_symbols=68,body_symbols_max=924,
                    dictionary_max=8,expected_candidate_count=792747,construction_sources=800,verification_sources=200,
                    gate_gain=.03,gate_min_snrs=2,per_snr_seconds=7200,max_rss_bytes=2147483648).items():
        require(p[k]==v,'Protocol constant differs: '+k)
    require(p['average_snrs_db']==[4,7,10,13] and p['thresholds_db']==THRESHOLDS,'Registered SNR grid or unique quantizer changed')
    require(p['greedy_compare_U_dictionary'] is True and p['second_header_symbols']==0 and p['intermediate_feedback'] is False,'Single-forward dictionary contract differs')
    require(p['sensitivity_scenarios']==['primary_clip','low_tail_all_R','low_tail_all_U','interpolation_left_hold','interpolation_right_hold'],
            'Frozen-policy sensitivity scenarios changed')
    require(p['failure_rule']=='consecutive_crc_accepted_prefix_v1_DROP_extension' and p['generation']=='unconditional_class1000_greedy_argmax_no_CFG_no_sampling',
            'Cached receiver/generation protocol changed')
    require(p['new_PHY_decodes']==0 and p['new_GPU_jobs']==0 and p['development_read'] is False and p['automatic_asset_or_link_followup'] is False,
            'Model-only stopping boundary changed')


def update_best(value,index,best_value,best_index):
    return value>best_value+EPS or (abs(value-best_value)<=EPS and (best_index<0 or index<best_index))


def uniform_and_fixed(coeff,rows,Q,gray,chunk=8192,progress=None):
    uvalue=np.full(8,-np.inf);uid=np.full(8,-1,dtype=np.int64);fixed_value=-np.inf;fixed=-1
    for begin in range(0,len(rows),chunk):
        end=min(begin+chunk,len(rows));scores=population_scores(coeff[begin:end],rows[begin:end],Q,gray)
        for b in range(8):
            local=int(np.argmax(scores[:,b]));value=float(scores[local,b]);index=begin+local
            if update_best(value,index,uvalue[b],int(uid[b])):uvalue[b]=value;uid[b]=index
        totals=scores.sum(1);local=int(np.argmax(totals));value=float(totals[local]);index=begin+local
        if update_best(value,index,fixed_value,fixed):fixed_value=value;fixed=index
        if progress:progress('population_reference_scan',end,len(rows))
    return uid,fixed,dict(per_bin_construction_contributions=uvalue.tolist(),U_construction_mean=float(uvalue.sum()),fixed_construction_mean=fixed_value)


def dictionary_scores(coeff,rows,Q,gray,indices):
    ids=np.asarray(sorted(set(map(int,indices))),dtype=np.int64)
    scores=score_block(coeff[ids],rows[ids],Q,gray)
    choice=np.argmax(scores,axis=0)
    return ids,ids[choice],np.take_along_axis(scores,choice[None,:,:],axis=0)[0]


def exact_greedy(coeff,rows,Q,gray,first,maximum=8,chunk=256,progress=None,lazy=True):
    """Exact diminishing-return upper bounds accelerate the registered greedy.

    Every bound is a previously evaluated marginal for the same candidate at a
    subset of the current dictionary. This preserves the full candidate set.
    """
    selected=[int(first)];current=score_block(coeff[[first]],rows[[first]],Q,gray)[0]
    bounds=np.full(len(rows),np.inf);bounds[first]=-np.inf;history=[];evaluations=0
    while len(selected)<maximum and len(selected)<len(rows):
        if lazy: order=np.lexsort((np.arange(len(rows)), -bounds))
        else: order=np.arange(len(rows))
        best=-1;best_gain=-np.inf;examined=0
        for begin in range(0,len(order),chunk):
            if lazy and np.isfinite(best_gain) and bounds[order[begin]]+BOUND_EPS<best_gain:break
            ids=order[begin:begin+chunk];ids=ids[np.isfinite(bounds[ids])|np.isposinf(bounds[ids])]
            if not len(ids):continue
            score=score_block(coeff[ids],rows[ids],Q,gray)
            gains=np.maximum(score-current[None,:,:],0).sum((1,2))/Q.shape[1]
            require(np.isfinite(gains).all() and gains.min()>=0,'Nonfinite content marginal')
            bounds[ids]=gains;evaluations+=len(ids);examined+=len(ids)
            for index,gain in zip(ids,gains):
                if update_best(float(gain),int(index),best_gain,best):best_gain=float(gain);best=int(index)
            if progress and begin%(chunk*32)==0:progress('greedy_round_'+str(len(selected)+1),examined,len(rows))
        if best<0 or best_gain<=EPS:break
        selected.append(best);bounds[best]=-np.inf
        current=np.maximum(current,score_block(coeff[[best]],rows[[best]],Q,gray)[0])
        history.append(dict(added_candidate=best,marginal_gain=best_gain,construction_mean=float(current.sum(0).mean()),evaluated_candidates=examined))
    return selected,history,evaluations


def choose_dictionary(coeff,rows,Q,gray,U,first,maximum=8,chunk=256,progress=None):
    greedy,history,count=exact_greedy(coeff,rows,Q,gray,first,maximum,chunk,progress)
    uids,_,uvalues=dictionary_scores(coeff,rows,Q,gray,U)
    gids,_,gvalues=dictionary_scores(coeff,rows,Q,gray,greedy)
    alternatives=[dict(name='U_BIN_WINNERS_DICTIONARY',indices=uids.tolist(),construction_mean=float(uvalues.sum(0).mean())),
                  dict(name='EXACT_GREEDY_MAX8',indices=gids.tolist(),construction_mean=float(gvalues.sum(0).mean()))]
    winner=alternatives[0]
    other=alternatives[1]
    if other['construction_mean']>winner['construction_mean']+EPS or (abs(other['construction_mean']-winner['construction_mean'])<=EPS
        and (len(other['indices']),other['indices'])<(len(winner['indices']),winner['indices'])):winner=other
    require(len(winner['indices'])<=maximum,'Dictionary capacity exceeded')
    return winner,dict(alternatives=alternatives,selected_alternative=winner['name'],greedy_history=history,
        marginal_candidate_evaluations=count,greedy_is_global_dictionary_optimum=False,
        greedy_exact_for_registered_stepwise_rule=True,construction_only=True)


def frozen_source_weights(coeff,rows,Q,gray,ids,method,bin_masses,states_count):
    if method=='QCSI-C8':_,choice,_=dictionary_scores(coeff,rows,Q,gray,ids)
    elif method=='QCSI-U':choice=np.broadcast_to(np.asarray(ids,dtype=np.int64)[:,None],(8,Q.shape[1])).copy()
    else:choice=np.full((8,Q.shape[1]),int(ids[0]),dtype=np.int64)
    weights=np.zeros((Q.shape[1],states_count));bad=np.zeros(Q.shape[1]);mean_used=np.zeros(Q.shape[1]);maximum=np.zeros(Q.shape[1],dtype=int)
    for b in range(8):
        picked=choice[b];c=np.asarray(coeff[picked,b]);sources=np.arange(Q.shape[1])
        np.add.at(weights,(sources,np.full(Q.shape[1],gray)),c[:,0])
        np.add.at(weights,(sources,np.asarray(rows['prefix'][picked],dtype=int)),c[:,1])
        np.add.at(weights,(sources,np.asarray(rows['target'][picked],dtype=int)),c[:,2])
        bad+=c[:,3];cost=100+rows['body'][picked].astype(int)
        mean_used+=float(bin_masses[b])*cost;maximum=np.maximum(maximum,cost)
    require(np.allclose(weights.sum(1)+bad,1,atol=1e-9,rtol=0) and np.max(maximum)<=1024,'Frozen policy probability/resource inconsistency')
    score=np.sum(weights*Q.T,axis=1)
    return dict(weights=weights,bad_probability=bad,quality_point=score,mean_used_symbols=mean_used,max_used_symbols=maximum,profile_indices_by_bin=choice.T)


def evaluate_with_frozen_choices(coeff_subset,selected_ids,rows,Q,gray,choice,bin_masses):
    lookup={int(i):k for k,i in enumerate(selected_ids)};score=np.zeros(Q.shape[1]);bad=np.zeros(Q.shape[1])
    for b in range(8):
        ids=choice[:,b];local=np.asarray([lookup[int(i)] for i in ids]);c=coeff_subset[local,b];source=np.arange(Q.shape[1])
        score+=c[:,0]*Q[gray]+c[:,1]*Q[rows['prefix'][ids],source]+c[:,2]*Q[rows['target'][ids],source];bad+=c[:,3]
    return score,bad


def scenario_model(model,scenario):
    require(scenario in ('primary_clip','low_tail_all_R','low_tail_all_U','interpolation_left_hold','interpolation_right_hold'),'Unknown frozen-policy sensitivity scenario')
    if scenario=='primary_clip':return model
    result=copy.copy(model);result.probabilities={}
    for key,(knots,probabilities) in model.curves.items():
        values=model.probabilities[key].copy()
        if scenario.startswith('low_tail_'):
            where=model.locations<knots[0];values[where]=[0.,1.,0.] if scenario=='low_tail_all_R' else [0.,0.,1.]
        else:
            side='right' if scenario=='interpolation_left_hold' else 'left'
            indices=np.searchsorted(knots,model.locations,side=side)-(1 if side=='right' else 0)
            values=probabilities[np.clip(indices,0,len(knots)-1)].copy()
        result.probabilities[key]=values
    return result
