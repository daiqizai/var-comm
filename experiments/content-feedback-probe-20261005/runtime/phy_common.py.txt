SIZES=(1,2,3,4,5,6,8,10,13,16)
HEADER_KEY='uep-v1-profile12-crc16-tail6-conv-ratematch136-qpsk-Es2'
RNG_NAMESPACE='CONTENT_FEEDBACK_20261005_V1'
SEEDS={'planning':2026100502,'verification':2026100503,'bootstrap':2026100504}


def require(value,message):
    if not value:raise RuntimeError(message)
def read(path):return json.loads(Path(path).read_text(encoding='utf-8-sig'))
def sha(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as f:
        for b in iter(lambda:f.read(8*1024*1024),b''):h.update(b)
    return h.hexdigest()
def save(path,value):
    path=Path(path);path.parent.mkdir(parents=True,exist_ok=True);temp=path.with_name(path.name+'.tmp.'+str(os.getpid()))
    temp.write_text(json.dumps(value,indent=2,allow_nan=False)+'\n');os.replace(temp,path)
def seal(path,value):
    if Path(path).exists():require(read(path)==value,'Frozen output changed: '+str(path))
    else:save(path,value)
def verify(bindings):
    for path,h in bindings.items():require(sha(path)==h,'Frozen input changed: '+str(path))
def array_sha(value):return hashlib.sha256(np.ascontiguousarray(value).tobytes()).hexdigest()
def token_count(m):return sum(s*s for s in SIZES[:m]) if m else 0
def raw_bits(tokens):return ((np.asarray(tokens,dtype=np.int64)[:,None]>>np.arange(11,-1,-1))&1).astype(np.uint8).ravel()
def decode_tokens(bits):
    bits=np.asarray(bits,dtype=np.uint8);require(len(bits)%12==0,'12-bit token alignment required')
    return (bits.reshape(-1,12).astype(np.int64)*(1<<np.arange(11,-1,-1))).sum(1)
def pack_bits(bits):
    bits=np.asarray(bits,dtype=np.uint8);return dict(bits=len(bits),hex=np.packbits(bits,bitorder='big').tobytes().hex())
def unpack_bits(value):return np.unpackbits(np.frombuffer(bytes.fromhex(value['hex']),dtype=np.uint8),bitorder='big')[:value['bits']]


def rng(phase,*identity):
    require(phase in SEEDS,'Separate registered RNG phase required')
    key=json.dumps([RNG_NAMESPACE,SEEDS[phase],phase,*identity],separators=(',',':')).encode()
    return np.random.Generator(np.random.PCG64(int.from_bytes(hashlib.sha256(key).digest()[:16],'little')))


class Channel:
    """A channel object belongs only to the simulator, never to the TX policy."""
    def __init__(self,phase,source_id,snr_db,rep):
        self.phase=phase;self.source_id=str(source_id);self.snr_db=float(snr_db);self.rep=int(rep)
        z=rng(phase,self.source_id,self.snr_db,self.rep,'h').standard_normal(2)/np.sqrt(2.)
        self.h=complex(*z);self.variance=10.**(-self.snr_db/10)
    def apply(self,wave,round_index,slot_start=0):
        wave=np.asarray(wave,dtype=np.float64);require(wave.ndim==2 and wave.shape[1]==2,'Complex Cartesian waveform required')
        require(0<=round_index<3 and 0<=slot_start and slot_start+len(wave)<=1024,'Registered waveform slots exceeded')
        noise=rng(self.phase,self.source_id,self.snr_db,self.rep,'noise',int(round_index)).standard_normal((1024,2))[slot_start:slot_start+len(wave)]
        x=wave[:,0]+1j*wave[:,1];y=self.h*x+np.sqrt(self.variance)*(noise[:,0]+1j*noise[:,1])
        return np.stack((y.real,y.imag),1),array_sha(noise)


def logsumexp(x,axis):
    m=np.max(x,axis=axis,keepdims=True);return np.squeeze(m,axis)+np.log(np.exp(x-m).sum(axis=axis))


def rayleigh_logits(received,h,variance,q):
    """Exact fixed-constellation LLR in y-space. h=0 is not clipped or inverted."""
    y=np.asarray(received,dtype=np.float64);h=complex(h)
    require(y.ndim==2 and y.shape[1]==2 and np.isfinite(y).all() and np.isfinite([h.real,h.imag,variance]).all()
            and variance>0 and q in (2,4),'Finite received samples and actual RX CSI required')
    labels=((np.arange(1<<q)[:,None]>>np.arange(q-1,-1,-1))&1).astype(np.uint8)
    if q==2:points=1.-2.*labels.astype(float)
    else:
        pam=np.array([-3.,-1.,3.,1.])/np.sqrt(5)
        points=np.stack((pam[2*labels[:,0]+labels[:,1]],pam[2*labels[:,2]+labels[:,3]]),1)
    expected=h*(points[:,0]+1j*points[:,1]);actual=y[:,0]+1j*y[:,1]
    lp=-np.abs(actual[:,None]-expected[None])**2/(2*variance)
    return np.stack([logsumexp(lp[:,labels[:,b]==1],1)-logsumexp(lp[:,labels[:,b]==0],1) for b in range(q)],1).ravel()


class PacketPHY:
    def __init__(self,backend,old_phy,scale_channel,planner,ledger):
        self.backend=backend;self.old=old_phy;self.scale=scale_channel;self.planner=planner;self.ledger=ledger
        require(backend.device=='cpu' and backend.qualified,'Pinned qualified CPU LDPC required')
        ledger.register_configuration(HEADER_KEY,dict(source_bits=12,CRC=16,tail=6,coded_bits=136,symbols=68,modulation='QPSK'),False)

    def plan_group(self,source_bits,modulation,symbols,nominal_rate='1/2'):
        return self.planner.group(int(source_bits),self.mcs_class(modulation,nominal_rate),int(symbols))

    def admit(self,group,new=None):
        q=2 if group['modulation']=='QPSK' else 4
        actual=self.backend.plan(int(group['source_bits'])+16,int(group['symbols'])*q,q)
        require(actual==group['layout'],'Actual backend layout differs from registered body')
        expected=digest(dict(backend=self.backend.identity,source_bits=group['source_bits'],CRC=16,layout=actual,modulation=group['modulation'],modulation_padding_bits=0))
        require(group['phy_key']==expected,'Physical key does not describe this exact code block')
        if hasattr(self,'known_phy_keys'):
            is_new=group['phy_key'] not in self.known_phy_keys
            require(new is not True or is_new,'An existing configuration was falsely declared new')
            new=is_new
        self.ledger.register_configuration(group['phy_key'],{k:group[k] for k in ('source_bits','modulation','symbols','layout')},new)

    def header_wave(self,header_id):
        require(type(header_id)is int and 0<=header_id<4096,'12-bit public header ID required')
        bits=self.scale.indices_to_bits([header_id],12)
        return np.asarray(self.scale.encode_packet(bits,68)['symbols'],dtype=np.float64)

    def body_wave(self,payload,group,counter,group_index,session):
        bits=np.asarray(payload,dtype=np.uint8);require(bits.shape==(group['source_bits'],),'Registered source payload length required')
        q=2 if group['modulation']=='QPSK' else 4
        wave,_=self.old.transmit_packet(self.backend,bits[None],group['symbols']*q,q,[int(counter)],int(group_index),session)
        return wave.detach().cpu().numpy()[0].astype(np.float64)

    def header_receive(self,y,h,variance,codebook,event,phase):
        def decode():
            llr=rayleigh_logits(y,h,variance,2);mapping=self.scale.rate_match_indices(68,136)
            bits,_=self.scale.decode_map(np.bincount(mapping,weights=-.5*llr,minlength=68))
            crc=bool(self.scale.crc_accepts(bits[:-6]));pid=int(self.scale.bits_to_indices(bits[:12],12)[0]);legal=str(pid) in codebook
            return dict(profile_id=pid,header_crc_accept=crc,header_fields_legal=legal,header_ok=bool(crc and legal),decoded=pack_bits(bits))
        request=dict(received_sha256=array_sha(y),h=[h.real,h.imag],variance=variance,codebook_sha256=digest(codebook))
        return self.ledger.decode_once(phase,event,'header',HEADER_KEY,request,decode)

    def body_receive(self,y,h,variance,group,counter,group_index,session,event,phase):
        q=2 if group['modulation']=='QPSK' else 4
        require(np.asarray(y).shape==(group['symbols'],2),'Received body length mismatch')
        def decode():
            logits=rayleigh_logits(y,h,variance,q)
            mask=self.old.mask_batch(group['symbols']*q,[counter],group_index,session)[0]
            logits=logits*(1-2*mask.astype(np.float64))
            decoded=self.backend.decode(logits[None].astype(np.float32),group['source_bits']+16,group['symbols']*q,q).detach().cpu().numpy()[0].astype(np.uint8)
            return dict(payload=pack_bits(decoded[:-16]),crc_accept=bool(self.old.crc_accept_batch(decoded)),decoded_sha256=array_sha(decoded))
        request=dict(received_sha256=array_sha(y),h=[h.real,h.imag],variance=variance,group=group['phy_key'],counter=int(counter),group_index=int(group_index),session=session)
        return self.ledger.decode_once(phase,event,'body',group['phy_key'],request,decode)


def load_runtime(snapshot,ledger):
    runtime=snapshot['runtime'];require(os.environ.get('CUDA_VISIBLE_DEVICES')=='','New CPU branch requires CUDA_VISIBLE_DEVICES empty')
    for key in ('OMP_NUM_THREADS','MKL_NUM_THREADS','OPENBLAS_NUM_THREADS'):
        require(0<int(os.environ.get(key,'999'))<=2,'CPU thread cap must be explicit: '+key)
    roots=[Path(runtime['uep_code']).resolve(),Path(runtime['src']).resolve()]
    for entry in snapshot.get('entries',{}).values():
        path=Path(entry['path']).resolve()
        if (path.suffix=='.py' and any(path.is_relative_to(root) for root in roots)) or path==Path(runtime['ldpc_qualification']).resolve():
            require(sha(path)==entry['sha256'],'Bound physical dependency changed before import')
    sys.path.insert(0,str(runtime['uep_code']));sys.path.insert(0,str(runtime['src']))
    import ldpc_backend,uep_phy,profiles
    from var_comm import scale_channel
    backend=ldpc_backend.SionnaBackend('cpu',runtime['ldpc_qualification']);backend.torch.set_num_threads(2)
    backend.torch.set_num_interop_threads(2)
    require(backend.torch.get_num_threads()<=2 and backend.torch.get_num_interop_threads()<=2,'CPU intra/inter-op thread limits exceeded')
    require(Path(ldpc_backend.__file__).resolve()==Path(runtime['uep_code']).resolve()/'ldpc_backend.py','LDPC source path differs')
    require(Path(scale_channel.__file__).resolve().is_relative_to(Path(runtime['src']).resolve()),'Source PHY must use bound snapshot, not live repository')
    phy=PacketPHY(backend,uep_phy,scale_channel,profiles.Planner(backend),ledger);phy.mcs_class=profiles.MCS
    profiles_asset=snapshot['assets']['profiles'];require(sha(profiles_asset['path'])==profiles_asset['sha256'],'Original physical catalogue changed')
    phy.known_phy_keys={g['phy_key'] for row in read(profiles_asset['path']) for g in row['groups']}
    return phy
