import copy
import unittest
import feedback_resources as r


def fixtures():
    profiles=[dict(N=1024,G=1 if i<209 else 2,stable_id=str(i),used_body_symbols=900 if i%2 else 956) for i in range(4041)]
    configurations={}
    pairs=[(0,m) for m in (4,5,6)]+[(m,m+d) for m in range(4,10) for d in (1,2) if m+d<=10]
    for start,end in pairs:
        bits=12*(r.tokens(end)-r.tokens(start))
        for mod in r.MODULATIONS:
            for rate in r.RATES:
                body=r.symbols(bits,mod,rate)
                if body+68>1024:continue
                sig=r.signature(bits,mod,body)
                configurations[sig]=dict(phy_key=r.identity(sig),source_bits=bits,modulation=mod,transmitted_bits=sig[2],
                    layout=dict(power_protocol='fixed_constellation_average_Es2',code_blocks=1,k=bits+16,n=sig[2]))
    rows=[dict(kind='body',phy_key=c['phy_key'],physical_configuration=c) for c in configurations.values()]
    quality=dict(rows=[dict(state_id=f'm{m}_K0',receiver='VAR') for m in range(4,10)])
    return profiles,dict(rows=rows),quality


class ResourceTests(unittest.TestCase):
    def test_CRC_and_symbol_rounding_exact(self):
        self.assertEqual(r.tokens(4),30);self.assertEqual(r.tokens(8),255)
        self.assertEqual(r.symbols(360,'QPSK','1/3'),564)
        self.assertEqual(r.symbols(3060,'16QAM','5/6'),923)
        self.assertEqual(3*68+2*32,268)

    def test_whole_nominal_grid_and_complete_QCSI_old_scope(self):
        profiles,bler,q=fixtures();cat=r.build_catalog(profiles,bler,q)
        self.assertEqual(len(cat['R_actions']),66)
        self.assertTrue(all(a['status']=='READY' for a in cat['R_actions']))
        self.assertTrue(all(a['minimum_prior_symbols']+a['forward_symbols']<=1024 for a in cat['R_actions']))
        self.assertFalse(any(a['end_m']>=8 for a in cat['R_actions']))
        self.assertEqual(len(cat['QCSI']['0']['old_profile_ids']),4041)
        self.assertEqual(cat['QCSI']['32']['old_grouped_count'],sum(p['G']==2 and p['used_body_symbols']<=924 for p in profiles))
        self.assertGreater(cat['QCSI']['32']['old_grouped_count'],0)
        self.assertEqual(len(cat['QCSI_new_actions']),23)
        self.assertLessEqual(len(cat['new_physical_requests']),23)
        self.assertLessEqual(cat['planning']['body_decodes_max'],41216)

    def test_QCSI_paid_body_is_never_reused_at_old_full_length(self):
        profiles,bler,q=fixtures();cat=r.build_catalog(profiles,bler,q)
        for action in cat['QCSI_new_actions']:
            self.assertEqual(action['body_symbols'],956-action['feedback_symbols'])
            self.assertEqual(action['forward_symbols']+action['feedback_symbols'],1024)
            if action['group']:
                self.assertEqual(action['group']['transmitted_bits'],action['body_symbols']*r.MODULATIONS[action['modulation']])

    def test_backend_rejections_are_retained_without_replacement(self):
        profiles,bler,q=fixtures();cat=r.build_catalog(profiles,bler,q)
        rejection=dict(complete=True,configs=[],rejections=[dict(request_id=x['request_id'],reason='Pinned backend unsupported') for x in cat['new_physical_requests']])
        final=r.build_catalog(profiles,bler,q,rejection)
        self.assertEqual(len(final['new_physical_requests']),len(cat['new_physical_requests']))
        self.assertTrue(all(x['qualification_status']=='UNAVAILABLE_BACKEND_REJECTION' for x in final['new_physical_requests']))
        self.assertEqual(len(final['R_actions']),66)

    def test_qualification_cannot_drop_request_or_change_layout(self):
        profiles,bler,q=fixtures();cat=r.build_catalog(profiles,bler,q)
        with self.assertRaisesRegex(ValueError,'every frozen request'):
            r.build_catalog(profiles,bler,q,dict(complete=True,configs=[],rejections=[]))
        x=cat['new_physical_requests'][0]
        group=dict(phy_key='new',source_bits=x['source_bits'],modulation=x['modulation'],symbols=x['body_symbols'],
            information_bits=x['source_bits']+16,transmitted_bits=x['transmitted_bits'],
            layout=dict(k=x['source_bits']+16,n=x['transmitted_bits']+4,code_blocks=1,power_protocol='fixed_constellation_average_Es2'))
        with self.assertRaisesRegex(ValueError,'changed requested length'):
            r.build_catalog(profiles,bler,q,dict(complete=True,configs=[group],rejections=[]))


if __name__=='__main__':unittest.main()
