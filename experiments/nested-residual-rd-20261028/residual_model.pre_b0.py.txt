"""Hard-forward nested scalar quantization; frozen decoder is external."""
import torch
from torch import nn
import torch.nn.functional as F

class ResidualCodec(nn.Module):
    def __init__(self,order):
        super().__init__();assert order in('scalar','progressive');self.order=order
        self.analysis=nn.Sequential(nn.Conv2d(64,96,3,2,1),nn.GELU(),nn.Conv2d(96,96,3,2,1),nn.GELU(),nn.Conv2d(96,4,3,1,1))
        self.synthesis_up=nn.Sequential(nn.Conv2d(8,96,3,1,1),nn.GELU(),nn.ConvTranspose2d(96,64,4,2,1),nn.GELU(),nn.ConvTranspose2d(64,64,4,2,1),nn.GELU())
        self.synthesis_out=nn.Sequential(nn.Conv2d(96,64,3,1,1),nn.GELU(),nn.Conv2d(64,32,3,1,1))
        self.entropy=nn.Sequential(nn.Conv2d(32,64,3,2,1),nn.GELU(),nn.Conv2d(64,64,3,2,1),nn.GELU(),nn.Conv2d(64,64,1))
        self.register_buffer('centers',-1+(2*torch.arange(16).float()+1)/16)

    def encode(self,full,coarse):return torch.tanh(self.analysis(torch.cat((full-coarse,coarse),1))).flatten(1)
    def probabilities(self,coarse):
        z=self.entropy(coarse).reshape(-1,4,16,4,4).permute(0,1,3,4,2).reshape(-1,64,16)
        return torch.softmax(z.float(),dim=-1)
    def positions(self):
        return [(i,d)for d in range(4)for i in range(64)]if self.order=='progressive'else[(i,d)for i in range(64)for d in range(4)]
    def quantize(self,u,j,gradient=True):
        assert 0<=j<=256
        v=torch.floor((u+1)*8).clamp(0,15).long();depth=torch.zeros(64,device=u.device,dtype=torch.long)
        for i,d in self.positions()[:j]:depth[i]=d+1
        scale=2.0**depth;prefix=torch.bitwise_right_shift(v,4-depth)
        hard=-1+(2*prefix+1)/scale;mask=(depth>0).to(u.dtype)
        hard=hard*mask
        q=hard+(u-u.detach())*mask if gradient else hard
        return q,depth.expand(len(u),-1),v
    def decode(self,q,depth,coarse):
        if not torch.any(depth):return coarse+q.sum()*0
        z=torch.cat((q.reshape(-1,4,4,4),depth.to(q.dtype).reshape(-1,4,4,4)/4),1)
        correction=self.synthesis_out(torch.cat((self.synthesis_up(z),coarse),1))
        return coarse+correction
    def estimated_bits(self,u,probabilities,j):
        # Hard bits forward, soft-assignment gradients to analysis. Previous
        # prefix conditioning is the actual hard prefix, never unsent residual.
        v=torch.floor((u+1)*8).clamp(0,15).long()
        assignment=torch.softmax(-((u[...,None]-self.centers)**2)/0.01,dim=-1)
        total=u.new_zeros(len(u));values=torch.arange(16,device=u.device)
        for i,d in self.positions()[:j]:
            prefix=v[:,i]>>(4-d) if d else torch.zeros_like(v[:,i])
            eligible=(values[None,:]>>(4-d))==prefix[:,None]if d else torch.ones((len(u),16),dtype=torch.bool,device=u.device)
            ones=((values>>(3-d))&1).bool()
            probs=probabilities[:,i];den=(probs*eligible).sum(-1).clamp_min(1e-9)
            p1=(probs*(eligible&ones)).sum(-1)/den;p1=p1.clamp(1e-7,1-1e-7)
            hard=((v[:,i]>>(3-d))&1).to(u.dtype)
            soft=(assignment[:,i]*ones).sum(-1);bit=hard+soft-soft.detach()
            total+=-bit*torch.log2(p1)-(1-bit)*torch.log2(1-p1)
        return total
    def forward(self,full,coarse,j):
        u=self.encode(full,coarse);q,d,_=self.quantize(u,j)
        return self.decode(q,d,coarse),self.estimated_bits(u,self.probabilities(coarse),j)
